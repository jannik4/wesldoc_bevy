29/**
30 *                  _______  ___  ___       ___           ___
31 *                 /       ||   \/   |     /   \         /   \
32 *                |   (---- |  \  /  |    /  ^  \       /  ^  \
33 *                 \   \    |  |\/|  |   /  /_\  \     /  /_\  \
34 *              ----)   |   |  |  |  |  /  _____  \   /  _____  \
35 *             |_______/    |__|  |__| /__/     \__\ /__/     \__\
36 *
37 *                               E N H A N C E D
38 *       S U B P I X E L   M O R P H O L O G I C A L   A N T I A L I A S I N G
39 *
40 *                         http://www.iryoku.com/smaa/
41 *
42 * Hi, welcome aboard!
43 *
44 * Here you'll find instructions to get the shader up and running as fast as
45 * possible.
46 *
47 * IMPORTANT NOTICE: when updating, remember to update both this file and the
48 * precomputed textures! They may change from version to version.
49 *
50 * The shader has three passes, chained together as follows:
51 *
52 *                           |input|------------------�
53 *                              v                     |
54 *                    [ SMAA*EdgeDetection ]          |
55 *                              v                     |
56 *                          |edgesTex|                |
57 *                              v                     |
58 *              [ SMAABlendingWeightCalculation ]     |
59 *                              v                     |
60 *                          |blendTex|                |
61 *                              v                     |
62 *                [ SMAANeighborhoodBlending ] <------�
63 *                              v
64 *                           |output|
65 *
66 * Note that each [pass] has its own vertex and pixel shader. Remember to use
67 * oversized triangles instead of quads to avoid overshading along the
68 * diagonal.
69 *
70 * You've three edge detection methods to choose from: luma, color or depth.
71 * They represent different quality/performance and anti-aliasing/sharpness
72 * tradeoffs, so our recommendation is for you to choose the one that best
73 * suits your particular scenario:
74 *
75 * - Depth edge detection is usually the fastest but it may miss some edges.
76 *
77 * - Luma edge detection is usually more expensive than depth edge detection,
78 *   but catches visible edges that depth edge detection can miss.
79 *
80 * - Color edge detection is usually the most expensive one but catches
81 *   chroma-only edges.
82 *
83 * For quickstarters: just use luma edge detection.
84 *
85 * The general advice is to not rush the integration process and ensure each
86 * step is done correctly (don't try to integrate SMAA T2x with predicated edge
87 * detection from the start!). Ok then, let's go!
88 *
89 *  1. The first step is to create two RGBA temporal render targets for holding
90 *     |edgesTex| and |blendTex|.
91 *
92 *     In DX10 or DX11, you can use a RG render target for the edges texture.
93 *     In the case of NVIDIA GPUs, using RG render targets seems to actually be
94 *     slower.
95 *
96 *     On the Xbox 360, you can use the same render target for resolving both
97 *     |edgesTex| and |blendTex|, as they aren't needed simultaneously.
98 *
99 *  2. Both temporal render targets |edgesTex| and |blendTex| must be cleared
100 *     each frame. Do not forget to clear the alpha channel!
101 *
102 *  3. The next step is loading the two supporting precalculated textures,
103 *     'areaTex' and 'searchTex'. You'll find them in the 'Textures' folder as
104 *     C++ headers, and also as regular DDS files. They'll be needed for the
105 *     'SMAABlendingWeightCalculation' pass.
106 *
107 *     If you use the C++ headers, be sure to load them in the format specified
108 *     inside of them.
109 *
110 *     You can also compress 'areaTex' and 'searchTex' using BC5 and BC4
111 *     respectively, if you have that option in your content processor pipeline.
112 *     When compressing then, you get a non-perceptible quality decrease, and a
113 *     marginal performance increase.
114 *
115 *  4. All samplers must be set to linear filtering and clamp.
116 *
117 *     After you get the technique working, remember that 64-bit inputs have
118 *     half-rate linear filtering on GCN.
119 *
120 *     If SMAA is applied to 64-bit color buffers, switching to point filtering
121 *     when accessing them will increase the performance. Search for
122 *     'SMAASamplePoint' to see which textures may benefit from point
123 *     filtering, and where (which is basically the color input in the edge
124 *     detection and resolve passes).
125 *
126 *  5. All texture reads and buffer writes must be non-sRGB, with the exception
127 *     of the input read and the output write in
128 *     'SMAANeighborhoodBlending' (and only in this pass!). If sRGB reads in
129 *     this last pass are not possible, the technique will work anyway, but
130 *     will perform antialiasing in gamma space.
131 *
132 *     IMPORTANT: for best results the input read for the color/luma edge
133 *     detection should *NOT* be sRGB.
134 *
135 *  6. Before including SMAA.h you'll have to setup the render target metrics,
136 *     the target and any optional configuration defines. Optionally you can
137 *     use a preset.
138 *
139 *     You have the following targets available:
140 *         SMAA_HLSL_3
141 *         SMAA_HLSL_4
142 *         SMAA_HLSL_4_1
143 *         SMAA_GLSL_3 *
144 *         SMAA_GLSL_4 *
145 *
146 *         * (See SMAA_INCLUDE_VS and SMAA_INCLUDE_PS below).
147 *
148 *     And four presets:
149 *         SMAA_PRESET_LOW          (60% of the quality)
150 *         SMAA_PRESET_MEDIUM       (80% of the quality)
151 *         SMAA_PRESET_HIGH         (95% of the quality)
152 *         SMAA_PRESET_ULTRA        (99% of the quality)
153 *
154 *     For example:
155 *         #define SMAA_RT_METRICS float4(1.0 / 1280.0, 1.0 / 720.0, 1280.0, 720.0)
156 *         #define SMAA_HLSL_4
157 *         #define SMAA_PRESET_HIGH
158 *         #include "SMAA.h"
159 *
160 *     Note that SMAA_RT_METRICS doesn't need to be a macro, it can be a
161 *     uniform variable. The code is designed to minimize the impact of not
162 *     using a constant value, but it is still better to hardcode it.
163 *
164 *     Depending on how you encoded 'areaTex' and 'searchTex', you may have to
165 *     add (and customize) the following defines before including SMAA.h:
166 *          #define SMAA_AREATEX_SELECT(sample) sample.rg
167 *          #define SMAA_SEARCHTEX_SELECT(sample) sample.r
168 *
169 *     If your engine is already using porting macros, you can define
170 *     SMAA_CUSTOM_SL, and define the porting functions by yourself.
171 *
172 *  7. Then, you'll have to setup the passes as indicated in the scheme above.
173 *     You can take a look into SMAA.fx, to see how we did it for our demo.
174 *     Checkout the function wrappers, you may want to copy-paste them!
175 *
176 *  8. It's recommended to validate the produced |edgesTex| and |blendTex|.
177 *     You can use a screenshot from your engine to compare the |edgesTex|
178 *     and |blendTex| produced inside of the engine with the results obtained
179 *     with the reference demo.
180 *
181 *  9. After you get the last pass to work, it's time to optimize. You'll have
182 *     to initialize a stencil buffer in the first pass (discard is already in
183 *     the code), then mask execution by using it the second pass. The last
184 *     pass should be executed in all pixels.
185 *
186 *
187 * After this point you can choose to enable predicated thresholding,
188 * temporal supersampling and motion blur integration:
189 *
190 * a) If you want to use predicated thresholding, take a look into
191 *    SMAA_PREDICATION; you'll need to pass an extra texture in the edge
192 *    detection pass.
193 *
194 * b) If you want to enable temporal supersampling (SMAA T2x):
195 *
196 * 1. The first step is to render using subpixel jitters. I won't go into
197 *    detail, but it's as simple as moving each vertex position in the
198 *    vertex shader, you can check how we do it in our DX10 demo.
199 *
200 * 2. Then, you must setup the temporal resolve. You may want to take a look
201 *    into SMAAResolve for resolving 2x modes. After you get it working, you'll
202 *    probably see ghosting everywhere. But fear not, you can enable the
203 *    CryENGINE temporal reprojection by setting the SMAA_REPROJECTION macro.
204 *    Check out SMAA_DECODE_VELOCITY if your velocity buffer is encoded.
205 *
206 * 3. The next step is to apply SMAA to each subpixel jittered frame, just as
207 *    done for 1x.
208 *
209 * 4. At this point you should already have something usable, but for best
210 *    results the proper area textures must be set depending on current jitter.
211 *    For this, the parameter 'subsampleIndices' of
212 *    'SMAABlendingWeightCalculationPS' must be set as follows, for our T2x
213 *    mode:
214 *
215 *    @SUBSAMPLE_INDICES
216 *
217 *    | S# |  Camera Jitter   |  subsampleIndices    |
218 *    +----+------------------+---------------------+
219 *    |  0 |  ( 0.25, -0.25)  |  float4(1, 1, 1, 0)  |
220 *    |  1 |  (-0.25,  0.25)  |  float4(2, 2, 2, 0)  |
221 *
222 *    These jitter positions assume a bottom-to-top y axis. S# stands for the
223 *    sample number.
224 *
225 * More information about temporal supersampling here:
226 *    http://iryoku.com/aacourse/downloads/13-Anti-Aliasing-Methods-in-CryENGINE-3.pdf
227 *
228 * c) If you want to enable spatial multisampling (SMAA S2x):
229 *
230 * 1. The scene must be rendered using MSAA 2x. The MSAA 2x buffer must be
231 *    created with:
232 *      - DX10:     see below (*)
233 *      - DX10.1:   D3D10_STANDARD_MULTISAMPLE_PATTERN or
234 *      - DX11:     D3D11_STANDARD_MULTISAMPLE_PATTERN
235 *
236 *    This allows to ensure that the subsample order matches the table in
237 *    @SUBSAMPLE_INDICES.
238 *
239 *    (*) In the case of DX10, we refer the reader to:
240 *      - SMAA::detectMSAAOrder and
241 *      - SMAA::msaaReorder
242 *
243 *    These functions allow to match the standard multisample patterns by
244 *    detecting the subsample order for a specific GPU, and reordering
245 *    them appropriately.
246 *
247 * 2. A shader must be run to output each subsample into a separate buffer
248 *    (DX10 is required). You can use SMAASeparate for this purpose, or just do
249 *    it in an existing pass (for example, in the tone mapping pass, which has
250 *    the advantage of feeding tone mapped subsamples to SMAA, which will yield
251 *    better results).
252 *
253 * 3. The full SMAA 1x pipeline must be run for each separated buffer, storing
254 *    the results in the final buffer. The second run should alpha blend with
255 *    the existing final buffer using a blending factor of 0.5.
256 *    'subsampleIndices' must be adjusted as in the SMAA T2x case (see point
257 *    b).
258 *
259 * d) If you want to enable temporal supersampling on top of SMAA S2x
260 *    (which actually is SMAA 4x):
261 *
262 * 1. SMAA 4x consists on temporally jittering SMAA S2x, so the first step is
263 *    to calculate SMAA S2x for current frame. In this case, 'subsampleIndices'
264 *    must be set as follows:
265 *
266 *    | F# | S# |   Camera Jitter    |    Net Jitter     |   subsampleIndices   |
267 *    +----+----+--------------------+-------------------+----------------------+
268 *    |  0 |  0 |  ( 0.125,  0.125)  |  ( 0.375, -0.125) |  float4(5, 3, 1, 3)  |
269 *    |  0 |  1 |  ( 0.125,  0.125)  |  (-0.125,  0.375) |  float4(4, 6, 2, 3)  |
270 *    +----+----+--------------------+-------------------+----------------------+
271 *    |  1 |  2 |  (-0.125, -0.125)  |  ( 0.125, -0.375) |  float4(3, 5, 1, 4)  |
272 *    |  1 |  3 |  (-0.125, -0.125)  |  (-0.375,  0.125) |  float4(6, 4, 2, 4)  |
273 *
274 *    These jitter positions assume a bottom-to-top y axis. F# stands for the
275 *    frame number. S# stands for the sample number.
276 *
277 * 2. After calculating SMAA S2x for current frame (with the new subsample
278 *    indices), previous frame must be reprojected as in SMAA T2x mode (see
279 *    point b).
280 *
281 * e) If motion blur is used, you may want to do the edge detection pass
282 *    together with motion blur. This has two advantages:
283 *
284 * 1. Pixels under heavy motion can be omitted from the edge detection process.
285 *    For these pixels we can just store "no edge", as motion blur will take
286 *    care of them.
287 * 2. The center pixel tap is reused.
288 *
289 * Note that in this case depth testing should be used instead of stenciling,
290 * as we have to write all the pixels in the motion blur pass.
291 *
292 * That's it!
293 */

381// Configurable Defines

383/**
384 * SMAA_THRESHOLD specifies the threshold or sensitivity to edges.
385 * Lowering this value you will be able to detect more edges at the expense of
386 * performance.
387 *
388 * Range: [0, 0.5]
389 *   0.1 is a reasonable value, and allows to catch most visible edges.
390 *   0.05 is a rather overkill value, that allows to catch 'em all.
391 *
392 *   If temporal supersampling is used, 0.2 could be a reasonable value, as low
393 *   contrast edges are properly filtered by just 2x.
394 */
395// (In the WGSL version of this shader, `SMAA_THRESHOLD` is set above, in "SMAA
396// Presets".)
397
398/**
399 * SMAA_MAX_SEARCH_STEPS specifies the maximum steps performed in the
400 * horizontal/vertical pattern searches, at each side of the pixel.
401 *
402 * In number of pixels, it's actually the double. So the maximum line length
403 * perfectly handled by, for example 16, is 64 (by perfectly, we meant that
404 * longer lines won't look as good, but still antialiased).
405 *
406 * Range: [0, 112]
407 */
408// (In the WGSL version of this shader, `SMAA_MAX_SEARCH_STEPS` is set above, in
409// "SMAA Presets".)
410
411/**
412 * SMAA_MAX_SEARCH_STEPS_DIAG specifies the maximum steps performed in the
413 * diagonal pattern searches, at each side of the pixel. In this case we jump
414 * one pixel at time, instead of two.
415 *
416 * Range: [0, 20]
417 *
418 * On high-end machines it is cheap (between a 0.8x and 0.9x slower for 16
419 * steps), but it can have a significant impact on older machines.
420 *
421 * Define SMAA_DISABLE_DIAG_DETECTION to disable diagonal processing.
422 */
423// (In the WGSL version of this shader, `SMAA_MAX_SEARCH_STEPS_DIAG` is set
424// above, in "SMAA Presets".)
425
426/**
427 * SMAA_CORNER_ROUNDING specifies how much sharp corners will be rounded.
428 *
429 * Range: [0, 100]
430 *
431 * Define SMAA_DISABLE_CORNER_DETECTION to disable corner processing.
432 */
433// (In the WGSL version of this shader, `SMAA_CORNER_ROUNDING` is set above, in
434// "SMAA Presets".)
435
436/**
437 * If there is a neighbor edge that has SMAA_LOCAL_CONTRAST_FACTOR times
438 * bigger contrast than current edge, current edge will be discarded.
439 *
440 * This allows to eliminate spurious crossing edges, and is based on the fact
441 * that, if there is too much contrast in a direction, that will hide
442 * perceptually contrast in the other neighbors.
443 */
444const SMAA_LOCAL_CONTRAST_ADAPTATION_FACTOR: f32 = 2.0;
445

459//-----------------------------------------------------------------------------
460// WGSL-Specific Functions
461
462// This vertex shader produces the following, when drawn using indices 0..3:
463//
464//  1 |  0-----x.....2
465//  0 |  |  s  |  . ´
466// -1 |  x_____x´
467// -2 |  :  .´
468// -3 |  1´
469//    +---------------
470//      -1  0  1  2  3
471//
472// The axes are clip-space x and y. The region marked s is the visible region.
473// The digits in the corners of the right-angled triangle are the vertex
474// indices.
475//
476// The top-left has UV 0,0, the bottom-left has 0,2, and the top-right has 2,0.
477// This means that the UV gets interpolated to 1,1 at the bottom-right corner
478// of the clip-space rectangle that is at 1,-1 in clip space.
479fn calculate_vertex_varyings(vertex_index: u32) -> VertexVaryings {
480    // See the explanation above for how this works
481    let uv = vec2<f32>(f32(vertex_index >> 1u), f32(vertex_index & 1u)) * 2.0;
482    let clip_position = vec2<f32>(uv * vec2<f32>(2.0, -2.0) + vec2<f32>(-1.0, 1.0));
483
484    return VertexVaryings(clip_position, uv);
485}

490@if(SMAA_EDGE_DETECTION)
491
492/**
493 * Edge Detection Vertex Shader
494 */
495@vertex
496fn edge_detection_vertex_main(@builtin(vertex_index) vertex_index: u32) -> EdgeDetectionVaryings {
497    let varyings = calculate_vertex_varyings(vertex_index);
498
499    var edge_detection_varyings = EdgeDetectionVaryings();
500    edge_detection_varyings.position = vec4(varyings.clip_coord, 0.0, 1.0);
501    edge_detection_varyings.tex_coord = varyings.tex_coord;
502
503    edge_detection_varyings.offset_0 = smaa_info.rt_metrics.xyxy * vec4(-1.0, 0.0, 0.0, -1.0) +
504        varyings.tex_coord.xyxy;
505    edge_detection_varyings.offset_1 = smaa_info.rt_metrics.xyxy * vec4(1.0, 0.0, 0.0, 1.0) +
506        varyings.tex_coord.xyxy;
507    edge_detection_varyings.offset_2 = smaa_info.rt_metrics.xyxy * vec4(-2.0, 0.0, 0.0, -2.0) +
508        varyings.tex_coord.xyxy;
509
510    return edge_detection_varyings;
511}

514@if(SMAA_BLENDING_WEIGHT_CALCULATION)
515
516/**
517 * Blend Weight Calculation Vertex Shader
518 */
519@vertex
520fn blending_weight_calculation_vertex_main(@builtin(vertex_index) vertex_index: u32)
521        -> BlendingWeightCalculationVaryings {
522    let varyings = calculate_vertex_varyings(vertex_index);
523
524    var weight_varyings = BlendingWeightCalculationVaryings();
525    weight_varyings.position = vec4(varyings.clip_coord, 0.0, 1.0);
526    weight_varyings.tex_coord = varyings.tex_coord;
527
528    // We will use these offsets for the searches later on (see @PSEUDO_GATHER4):
529    weight_varyings.offset_0 = smaa_info.rt_metrics.xyxy * vec4(-0.25, -0.125, 1.25, -0.125) +
530        varyings.tex_coord.xyxy;
531    weight_varyings.offset_1 = smaa_info.rt_metrics.xyxy * vec4(-0.125, -0.25, -0.125, 1.25) +
532        varyings.tex_coord.xyxy;
533
534    // And these for the searches, they indicate the ends of the loops:
535    weight_varyings.offset_2 =
536        smaa_info.rt_metrics.xxyy * vec4(-2.0, 2.0, -2.0, 2.0) * f32(SMAA_MAX_SEARCH_STEPS) +
537        vec4(weight_varyings.offset_0.xz, weight_varyings.offset_1.yw);
538
539    return weight_varyings;
540}

564@if(SMAA_EDGE_DETECTION)
565
566/**
567 * Luma Edge Detection
568 *
569 * IMPORTANT NOTICE: luma edge detection requires gamma-corrected colors, and
570 * thus 'color_texture' should be a non-sRGB texture.
571 */
572@fragment
573fn luma_edge_detection_fragment_main(in: EdgeDetectionVaryings) -> @location(0) vec4<f32> {
574    // Calculate the threshold:
575    // TODO: Predication.
576    let threshold = vec2(SMAA_THRESHOLD);
577
578    // Calculate luma:
579    let weights = vec3(0.2126, 0.7152, 0.0722);
580    let L = dot(textureSample(color_texture, color_sampler, in.tex_coord).rgb, weights);
581
582    let Lleft = dot(textureSample(color_texture, color_sampler, in.offset_0.xy).rgb, weights);
583    let Ltop  = dot(textureSample(color_texture, color_sampler, in.offset_0.zw).rgb, weights);
584
585    // We do the usual threshold:
586    var delta: vec4<f32> = vec4(abs(L - vec2(Lleft, Ltop)), 0.0, 0.0);
587    var edges = step(threshold, delta.xy);
588
589    // Then discard if there is no edge:
590    if (dot(edges, vec2(1.0)) == 0.0) {
591        discard;
592    }
593
594    // Calculate right and bottom deltas:
595    let Lright  = dot(textureSample(color_texture, color_sampler, in.offset_1.xy).rgb, weights);
596    let Lbottom = dot(textureSample(color_texture, color_sampler, in.offset_1.zw).rgb, weights);
597    delta = vec4(delta.xy, abs(L - vec2(Lright, Lbottom)));
598
599    // Calculate the maximum delta in the direct neighborhood:
600    var max_delta = max(delta.xy, delta.zw);
601
602    // Calculate left-left and top-top deltas:
603    let Lleftleft = dot(textureSample(color_texture, color_sampler, in.offset_2.xy).rgb, weights);
604    let Ltoptop   = dot(textureSample(color_texture, color_sampler, in.offset_2.zw).rgb, weights);
605    delta = vec4(delta.xy, abs(vec2(Lleft, Ltop) - vec2(Lleftleft, Ltoptop)));
606
607    // Calculate the final maximum delta:
608    max_delta = max(max_delta.xy, delta.zw);
609    let final_delta = max(max_delta.x, max_delta.y);
610
611    // Local contrast adaptation:
612    edges *= step(vec2(final_delta), SMAA_LOCAL_CONTRAST_ADAPTATION_FACTOR * delta.xy);
613
614    return vec4(edges, 0.0, 1.0);
615}

622@if(SMAA_BLENDING_WEIGHT_CALCULATION && !SMAA_DISABLE_DIAG_DETECTION)

624/**
625 * Allows to decode two binary values from a bilinear-filtered access.
626 */
627fn decode_diag_bilinear_access_2(in_e: vec2<f32>) -> vec2<f32> {
628    // Bilinear access for fetching 'e' have a 0.25 offset, and we are
629    // interested in the R and G edges:
630    //
631    // +---G---+-------+
632    // |   x o R   x   |
633    // +-------+-------+
634    //
635    // Then, if one of these edge is enabled:
636    //   Red:   (0.75 * X + 0.25 * 1) => 0.25 or 1.0
637    //   Green: (0.75 * 1 + 0.25 * X) => 0.75 or 1.0
638    //
639    // This function will unpack the values (mad + mul + round):
640    // wolframalpha.com: round(x * abs(5 * x - 5 * 0.75)) plot 0 to 1
641    var e = in_e;
642    e.r = e.r * abs(5.0 * e.r - 5.0 * 0.75);
643    return round(e);
644}
645
646@if(SMAA_BLENDING_WEIGHT_CALCULATION && !SMAA_DISABLE_DIAG_DETECTION)
647fn decode_diag_bilinear_access_4(e: vec4<f32>) -> vec4<f32> {
648    let e_rb = e.rb * abs(5.0 * e.rb - 5.0 * 0.75);
649    return round(vec4(e_rb.x, e.g, e_rb.y, e.a));
650}

652/**
653 * These functions allows to perform diagonal pattern searches.
654 */
655@if(SMAA_BLENDING_WEIGHT_CALCULATION && !SMAA_DISABLE_DIAG_DETECTION)
656fn search_diag_1(tex_coord: vec2<f32>, dir: vec2<f32>, e: ptr<function, vec2<f32>>) -> vec2<f32> {
657    var coord = vec4(tex_coord, -1.0, 1.0);
658    let t = vec3(smaa_info.rt_metrics.xy, 1.0);
659    while (coord.z < f32(SMAA_MAX_SEARCH_STEPS_DIAG - 1u) && coord.w > 0.9) {
660        coord = vec4(t * vec3(dir, 1.0) + coord.xyz, coord.w);
661        *e = textureSampleLevel(edges_texture, edges_sampler, coord.xy, 0.0).rg;
662        coord.w = dot(*e, vec2(0.5));
663    }
664    return coord.zw;
665}

667@if(SMAA_BLENDING_WEIGHT_CALCULATION && !SMAA_DISABLE_DIAG_DETECTION)
668fn search_diag_2(tex_coord: vec2<f32>, dir: vec2<f32>, e: ptr<function, vec2<f32>>) -> vec2<f32> {
669    var coord = vec4(tex_coord, -1.0, 1.0);
670    coord.x += 0.25 * smaa_info.rt_metrics.x; // See @SearchDiag2Optimization
671    let t = vec3(smaa_info.rt_metrics.xy, 1.0);
672    while (coord.z < f32(SMAA_MAX_SEARCH_STEPS_DIAG - 1u) && coord.w > 0.9) {
673        coord = vec4(t * vec3(dir, 1.0) + coord.xyz, coord.w);
674
675        // @SearchDiag2Optimization
676        // Fetch both edges at once using bilinear filtering:
677        *e = textureSampleLevel(edges_texture, edges_sampler, coord.xy, 0.0).rg;
678        *e = decode_diag_bilinear_access_2(*e);
679
680        // Non-optimized version:
681        // e.g = SMAASampleLevelZero(edgesTex, coord.xy).g;
682        // e.r = SMAASampleLevelZeroOffset(edgesTex, coord.xy, int2(1, 0)).r;
683
684        coord.w = dot(*e, vec2(0.5));
685    }
686    return coord.zw;
687}

689/**
690 * Similar to SMAAArea, this calculates the area corresponding to a certain
691 * diagonal distance and crossing edges 'e'.
692 */
693@if(SMAA_BLENDING_WEIGHT_CALCULATION && !SMAA_DISABLE_DIAG_DETECTION)
694fn area_diag(dist: vec2<f32>, e: vec2<f32>, offset: f32) -> vec2<f32> {
695    var tex_coord = vec2(SMAA_AREATEX_MAX_DISTANCE_DIAG) * e + dist;
696
697    // We do a scale and bias for mapping to texel space:
698    tex_coord = SMAA_AREATEX_PIXEL_SIZE * tex_coord + 0.5 * SMAA_AREATEX_PIXEL_SIZE;
699
700    // Diagonal areas are on the second half of the texture:
701    tex_coord.x += 0.5;
702
703    // Move to proper place, according to the subpixel offset:
704    tex_coord.y += SMAA_AREATEX_SUBTEX_SIZE * offset;
705
706    // Do it!
707    return textureSampleLevel(area_texture, edges_sampler, tex_coord, 0.0).rg;
708}

710/**
711 * This searches for diagonal patterns and returns the corresponding weights.
712 */
713@if(SMAA_BLENDING_WEIGHT_CALCULATION && !SMAA_DISABLE_DIAG_DETECTION)
714fn calculate_diag_weights(tex_coord: vec2<f32>, e: vec2<f32>, subsample_indices: vec4<f32>)
715        -> vec2<f32> {
716    var weights = vec2(0.0, 0.0);
717
718    // Search for the line ends:
719    var d = vec4(0.0);
720    var end = vec2(0.0);
721    if (e.r > 0.0) {
722        let d_xz = search_diag_1(tex_coord, vec2(-1.0, 1.0), &end);
723        d = vec4(d_xz.x, d.y, d_xz.y, d.w);
724        d.x += f32(end.y > 0.9);
725    } else {
726        d = vec4(0.0, d.y, 0.0, d.w);
727    }
728    let d_yw = search_diag_1(tex_coord, vec2(1.0, -1.0), &end);
729    d = vec4(d.x, d_yw.x, d.y, d_yw.y);
730
731    if (d.x + d.y > 2.0) {  // d.x + d.y + 1 > 3
732        // Fetch the crossing edges:
733        let coords = vec4(-d.x + 0.25, d.x, d.y, -d.y - 0.25) * smaa_info.rt_metrics.xyxy +
734            tex_coord.xyxy;
735        var c = vec4(
736            textureSampleLevel(edges_texture, edges_sampler, coords.xy, 0.0, vec2(-1, 0)).rg,
737            textureSampleLevel(edges_texture, edges_sampler, coords.zw, 0.0, vec2( 1, 0)).rg,
738        );
739        let c_yxwz = decode_diag_bilinear_access_4(c.xyzw);
740        c = c_yxwz.yxwz;
741
742        // Non-optimized version:
743        // float4 coords = mad(float4(-d.x, d.x, d.y, -d.y), SMAA_RT_METRICS.xyxy, texcoord.xyxy);
744        // float4 c;
745        // c.x = SMAASampleLevelZeroOffset(edgesTex, coords.xy, int2(-1,  0)).g;
746        // c.y = SMAASampleLevelZeroOffset(edgesTex, coords.xy, int2( 0,  0)).r;
747        // c.z = SMAASampleLevelZeroOffset(edgesTex, coords.zw, int2( 1,  0)).g;
748        // c.w = SMAASampleLevelZeroOffset(edgesTex, coords.zw, int2( 1, -1)).r;
749
750        // Merge crossing edges at each side into a single value:
751        var cc = vec2(2.0) * c.xz + c.yw;
752
753        // Remove the crossing edge if we didn't found the end of the line:
754        cc = select(cc, vec2(0.0, 0.0), vec2<bool>(step(vec2(0.9), d.zw)));
755
756        // Fetch the areas for this line:
757        weights += area_diag(d.xy, cc, subsample_indices.z);
758    }
759
760    // Search for the line ends:
761    let d_xz = search_diag_2(tex_coord, vec2(-1.0, -1.0), &end);
762    if (textureSampleLevel(edges_texture, edges_sampler, tex_coord, 0.0, vec2(1, 0)).r > 0.0) {
763        let d_yw = search_diag_2(tex_coord, vec2(1.0, 1.0), &end);
764        d = vec4(d_xz.x, d_yw.x, d_xz.y, d_yw.y);
765        d.y += f32(end.y > 0.9);
766    } else {
767        d = vec4(d_xz.x, 0.0, d_xz.y, 0.0);
768    }
769
770    if (d.x + d.y > 2.0) {  // d.x + d.y + 1 > 3
771        // Fetch the crossing edges:
772        let coords = vec4(-d.x, -d.x, d.y, d.y) * smaa_info.rt_metrics.xyxy + tex_coord.xyxy;
773        let c = vec4(
774            textureSampleLevel(edges_texture, edges_sampler, coords.xy, 0.0, vec2(-1,  0)).g,
775            textureSampleLevel(edges_texture, edges_sampler, coords.xy, 0.0, vec2( 0, -1)).r,
776            textureSampleLevel(edges_texture, edges_sampler, coords.zw, 0.0, vec2( 1,  0)).gr,
777        );
778        var cc = vec2(2.0) * c.xz + c.yw;
779
780        // Remove the crossing edge if we didn't found the end of the line:
781        cc = select(cc, vec2(0.0, 0.0), vec2<bool>(step(vec2(0.9), d.zw)));
782
783        // Fetch the areas for this line:
784        weights += area_diag(d.xy, cc, subsample_indices.w).gr;
785    }
786
787    return weights;
788}

794/**
795 * This allows to determine how much length should we add in the last step
796 * of the searches. It takes the bilinearly interpolated edge (see
797 * @PSEUDO_GATHER4), and adds 0, 1 or 2, depending on which edges and
798 * crossing edges are active.
799 */
800@if(SMAA_BLENDING_WEIGHT_CALCULATION)
801fn search_length(e: vec2<f32>, offset: f32) -> f32 {
802    // The texture is flipped vertically, with left and right cases taking half
803    // of the space horizontally:
804    var scale = SMAA_SEARCHTEX_SIZE * vec2(0.5, -1.0);
805    var bias = SMAA_SEARCHTEX_SIZE * vec2(offset, 1.0);
806
807    // Scale and bias to access texel centers:
808    scale += vec2(-1.0,  1.0);
809    bias  += vec2( 0.5, -0.5);
810
811    // Convert from pixel coordinates to texcoords:
812    // (We use SMAA_SEARCHTEX_PACKED_SIZE because the texture is cropped)
813    scale *= 1.0 / SMAA_SEARCHTEX_PACKED_SIZE;
814    bias *= 1.0 / SMAA_SEARCHTEX_PACKED_SIZE;
815
816    // Lookup the search texture:
817    return textureSampleLevel(search_texture, edges_sampler, scale * e + bias, 0.0).r;
818}

820/**
821 * Horizontal/vertical search functions for the 2nd pass.
822 */
823@if(SMAA_BLENDING_WEIGHT_CALCULATION)
824fn search_x_left(in_tex_coord: vec2<f32>, end: f32) -> f32 {
825    var tex_coord = in_tex_coord;
826
827    /**
828     * @PSEUDO_GATHER4
829     * This texcoord has been offset by (-0.25, -0.125) in the vertex shader to
830     * sample between edge, thus fetching four edges in a row.
831     * Sampling with different offsets in each direction allows to disambiguate
832     * which edges are active from the four fetched ones.
833     */
834    var e = vec2(0.0, 1.0);
835    while (tex_coord.x > end &&
836           e.g > 0.8281 &&  // Is there some edge not activated?
837           e.r == 0.0) {    // Or is there a crossing edge that breaks the line?
838        e = textureSampleLevel(edges_texture, edges_sampler, tex_coord, 0.0).rg;
839        tex_coord += -vec2(2.0, 0.0) * smaa_info.rt_metrics.xy;
840    }
841    let offset = -(255.0 / 127.0) * search_length(e, 0.0) + 3.25;
842    return smaa_info.rt_metrics.x * offset + tex_coord.x;
843}

845@if(SMAA_BLENDING_WEIGHT_CALCULATION)
846fn search_x_right(in_tex_coord: vec2<f32>, end: f32) -> f32 {
847    var tex_coord = in_tex_coord;
848
849    var e = vec2(0.0, 1.0);
850    while (tex_coord.x < end &&
851           e.g > 0.8281 &&  // Is there some edge not activated?
852           e.r == 0.0) {    // Or is there a crossing edge that breaks the line?
853        e = textureSampleLevel(edges_texture, edges_sampler, tex_coord, 0.0).rg;
854        tex_coord += vec2(2.0, 0.0) * smaa_info.rt_metrics.xy;
855    }
856    let offset = -(255.0 / 127.0) * search_length(e, 0.5) + 3.25;
857    return -smaa_info.rt_metrics.x * offset + tex_coord.x;
858}

860@if(SMAA_BLENDING_WEIGHT_CALCULATION)

864    var e = vec2(1.0, 0.0);
865    while (tex_coord.y > end &&
866           e.r > 0.8281 &&  // Is there some edge not activated?
867           e.g == 0.0) {    // Or is there a crossing edge that breaks the line?
868        e = textureSampleLevel(edges_texture, edges_sampler, tex_coord, 0.0).rg;
869        tex_coord += -vec2(0.0, 2.0) * smaa_info.rt_metrics.xy;

890/**
891 * Ok, we have the distance and both crossing edges. So, what are the areas
892 * at each side of current edge?
893 */
894@if(SMAA_BLENDING_WEIGHT_CALCULATION)
895fn area(dist: vec2<f32>, e1: f32, e2: f32, offset: f32) -> vec2<f32> {
896    // Rounding prevents precision errors of bilinear filtering:
897    var tex_coord = SMAA_AREATEX_MAX_DISTANCE * round(4.0 * vec2(e1, e2)) + dist;
898
899    // We do a scale and bias for mapping to texel space:
900    tex_coord = SMAA_AREATEX_PIXEL_SIZE * tex_coord + 0.5 * SMAA_AREATEX_PIXEL_SIZE;
901
902    // Move to proper place, according to the subpixel offset:
903    tex_coord.y += SMAA_AREATEX_SUBTEX_SIZE * offset;
904
905    // Do it!
906    return textureSampleLevel(area_texture, edges_sampler, tex_coord, 0.0).rg;
907}

910// Corner Detection Functions
911
912@if(SMAA_BLENDING_WEIGHT_CALCULATION)
913fn detect_horizontal_corner_pattern(weights: vec2<f32>, tex_coord: vec4<f32>, d: vec2<f32>)
914        -> vec2<f32> {
915@if(!SMAA_DISABLE_CORNER_DETECTION) {
916    let left_right = step(d.xy, d.yx);
917    var rounding = (1.0 - SMAA_CORNER_ROUNDING_NORM) * left_right;
918
919    rounding /= left_right.x + left_right.y; // Reduce blending for pixels in the center of a line.
920
921    var factor = vec2(1.0, 1.0);
922    factor.x -= rounding.x *
923        textureSampleLevel(edges_texture, edges_sampler, tex_coord.xy, 0.0, vec2(0,  1)).r;
924    factor.x -= rounding.y *
925        textureSampleLevel(edges_texture, edges_sampler, tex_coord.zw, 0.0, vec2(1,  1)).r;
926    factor.y -= rounding.x *
927        textureSampleLevel(edges_texture, edges_sampler, tex_coord.xy, 0.0, vec2(0, -2)).r;
928    factor.y -= rounding.y *
929        textureSampleLevel(edges_texture, edges_sampler, tex_coord.zw, 0.0, vec2(1, -2)).r;
930
931    return weights * saturate(factor);
932} @else {
933    return weights;
934}
935}

937@if(SMAA_BLENDING_WEIGHT_CALCULATION)
938fn detect_vertical_corner_pattern(weights: vec2<f32>, tex_coord: vec4<f32>, d: vec2<f32>)
939        -> vec2<f32> {
940@if(!SMAA_DISABLE_CORNER_DETECTION) {
941    let left_right = step(d.xy, d.yx);
942    var rounding = (1.0 - SMAA_CORNER_ROUNDING_NORM) * left_right;
943
944    rounding /= left_right.x + left_right.y;
945
946    var factor = vec2(1.0, 1.0);
947    factor.x -= rounding.x *
948        textureSampleLevel(edges_texture, edges_sampler, tex_coord.xy, 0.0, vec2( 1, 0)).g;
949    factor.x -= rounding.y *
950        textureSampleLevel(edges_texture, edges_sampler, tex_coord.zw, 0.0, vec2( 1, 1)).g;
951    factor.y -= rounding.x *
952        textureSampleLevel(edges_texture, edges_sampler, tex_coord.xy, 0.0, vec2(-2, 0)).g;
953    factor.y -= rounding.y *
954        textureSampleLevel(edges_texture, edges_sampler, tex_coord.zw, 0.0, vec2(-2, 1)).g;

956    return weights * saturate(factor);
957} @else {

959}

962//-----------------------------------------------------------------------------
963// Blending Weight Calculation Pixel Shader (Second Pass)
964
965@if(SMAA_BLENDING_WEIGHT_CALCULATION)
966@fragment
967fn blending_weight_calculation_fragment_main(in: BlendingWeightCalculationVaryings)
968        -> @location(0) vec4<f32> {
969    let subsample_indices = vec4(0.0);  // Just pass zero for SMAA 1x, see @SUBSAMPLE_INDICES.
970
971    var weights = vec4(0.0);
972
973    var e = textureSample(edges_texture, edges_sampler, in.tex_coord).rg;
974
975    if (e.g > 0.0) {    // Edge at north
976@if(!SMAA_DISABLE_DIAG_DETECTION) {
977        // Diagonals have both north and west edges, so searching for them in
978        // one of the boundaries is enough.
979        weights = vec4(calculate_diag_weights(in.tex_coord, e, subsample_indices), weights.ba);
980
981        // We give priority to diagonals, so if we find a diagonal we skip
982        // horizontal/vertical processing.
983        if (weights.r + weights.g != 0.0) {
984            return weights;
985        }
986}
987
988        var d: vec2<f32>;
989
990        // Find the distance to the left:
991        var coords: vec3<f32>;
992        coords.x = search_x_left(in.offset_0.xy, in.offset_2.x);
993        // in.offset_1.y = in.tex_coord.y - 0.25 * smaa_info.rt_metrics.y (@CROSSING_OFFSET)
994        coords.y = in.offset_1.y;
995        d.x = coords.x;
996
997        // Now fetch the left crossing edges, two at a time using bilinear
998        // filtering. Sampling at -0.25 (see @CROSSING_OFFSET) enables to
999        // discern what value each edge has:
1000        let e1 = textureSampleLevel(edges_texture, edges_sampler, coords.xy, 0.0).r;
1001
1002        // Find the distance to the right:
1003        coords.z = search_x_right(in.offset_0.zw, in.offset_2.y);
1004        d.y = coords.z;
1005
1006        // We want the distances to be in pixel units (doing this here allow to
1007        // better interleave arithmetic and memory accesses):
1008        d = abs(round(smaa_info.rt_metrics.zz * d - in.position.xx));
1009
1010        // SMAAArea below needs a sqrt, as the areas texture is compressed
1011        // quadratically:
1012        let sqrt_d = sqrt(d);
1013
1014        // Fetch the right crossing edges:
1015        let e2 = textureSampleLevel(
1016            edges_texture, edges_sampler, coords.zy, 0.0, vec2<i32>(1, 0)).r;
1017
1018        // Ok, we know how this pattern looks like, now it is time for getting
1019        // the actual area:
1020        weights = vec4(area(sqrt_d, e1, e2, subsample_indices.y), weights.ba);
1021
1022        // Fix corners:
1023        coords.y = in.tex_coord.y;
1024        weights = vec4(
1025            detect_horizontal_corner_pattern(weights.rg, coords.xyzy, d),
1026            weights.ba
1027        );
1028    }
1029
1030    if (e.r > 0.0) {    // Edge at west
1031        var d: vec2<f32>;
1032
1033        // Find the distance to the top:
1034        var coords: vec3<f32>;
1035        coords.y = search_y_up(in.offset_1.xy, in.offset_2.z);
1036        // in.offset_1.x = in.tex_coord.x - 0.25 * smaa_info.rt_metrics.x
1037        coords.x = in.offset_0.x;
1038        d.x = coords.y;
1039
1040        // Fetch the top crossing edges:
1041        let e1 = textureSampleLevel(edges_texture, edges_sampler, coords.xy, 0.0).g;
1042
1043        // Find the distance to the bottom:
1044        coords.z = search_y_down(in.offset_1.zw, in.offset_2.w);
1045        d.y = coords.z;
1046
1047        // We want the distances to be in pixel units:
1048        d = abs(round(smaa_info.rt_metrics.ww * d - in.position.yy));
1049
1050        // SMAAArea below needs a sqrt, as the areas texture is compressed
1051        // quadratically:
1052        let sqrt_d = sqrt(d);
1053
1054        // Fetch the bottom crossing edges:
1055        let e2 = textureSampleLevel(
1056            edges_texture, edges_sampler, coords.xz, 0.0, vec2<i32>(0, 1)).g;
1057
1058        // Get the area for this direction:
1059        weights = vec4(weights.rg, area(sqrt_d, e1, e2, subsample_indices.x));
1060
1061        // Fix corners:
1062        coords.x = in.tex_coord.x;
1063        weights = vec4(weights.rg, detect_vertical_corner_pattern(weights.ba, coords.xyxz, d));
1064    }
1065
1066    return weights;
1067}

1070//-----------------------------------------------------------------------------
1071// Neighborhood Blending Pixel Shader (Third Pass)
1072
1073@if(SMAA_NEIGHBORHOOD_BLENDING)
1074@fragment
1075fn neighborhood_blending_fragment_main(in: NeighborhoodBlendingVaryings) -> @location(0) vec4<f32> {
1076    // Fetch the blending weights for current pixel:
1077    let a = vec4(
1078        textureSample(blend_texture, blend_sampler, in.offset.xy).a,    // Right
1079        textureSample(blend_texture, blend_sampler, in.offset.zw).g,    // Top
1080        textureSample(blend_texture, blend_sampler, in.tex_coord).zx,   // Bottom / Left
1081    );
1082
1083    // Is there any blending weight with a value greater than 0.0?
1084    if (dot(a, vec4(1.0)) < 1.0e-5) {
1085        let color = textureSampleLevel(color_texture, blend_sampler, in.tex_coord, 0.0);
1086        // TODO: Reprojection
1087        return color;
1088    } else {
1089        let h = max(a.x, a.z) > max(a.y, a.w);  // max(horizontal) > max(vertical)
1090
1091        // Calculate the blending offsets:
1092        var blending_offset = vec4(0.0, a.y, 0.0, a.w);
1093        var blending_weight = a.yw;
1094        blending_offset = select(blending_offset, vec4(a.x, 0.0, a.z, 0.0), h);
1095        blending_weight = select(blending_weight, a.xz, h);
1096        blending_weight /= dot(blending_weight, vec2(1.0));
1097
1098        // Calculate the texture coordinates:
1099        let blending_coord =
1100            blending_offset * vec4(smaa_info.rt_metrics.xy, -smaa_info.rt_metrics.xy) +
1101            in.tex_coord.xyxy;
1102
1103        // We exploit bilinear filtering to mix current pixel with the chosen
1104        // neighbor:
1105        var color = blending_weight.x *
1106            textureSampleLevel(color_texture, blend_sampler, blending_coord.xy, 0.0);
1107        color += blending_weight.y *
1108            textureSampleLevel(color_texture, blend_sampler, blending_coord.zw, 0.0);
1109
1110        // TODO: Reprojection
1111
1112        return color;
1113    }
