86@compute
87@workgroup_size(256, 1, 1)
88fn downsample_first(
89    @builtin(workgroup_id) workgroup_id: vec3u,
90    @builtin(local_invocation_index) local_invocation_index: u32
91) {
92
93    let sub_xy = remap_for_wave_reduction(local_invocation_index % 64u);
94    let x = sub_xy.x + 8u * ((local_invocation_index >> 6u) % 2u);
95    let y = sub_xy.y + 8u * (local_invocation_index >> 7u);
96
97    spd_downsample_mips_0_1(x, y, workgroup_id.xy, local_invocation_index, constants.mips, workgroup_id.z);
98
99    spd_downsample_next_four(x, y, workgroup_id.xy, local_invocation_index, 2u, constants.mips, workgroup_id.z);
100}

118fn spd_downsample_mips_0_1(x: u32, y: u32, workgroup_id: vec2u, local_invocation_index: u32, mips: u32, slice: u32) {
119    var v: array<vec4f, 4>;
120
121    var tex = (workgroup_id * 64u) + vec2(x * 2u, y * 2u);
122    var pix = (workgroup_id * 32u) + vec2(x, y);
123    v[0] = spd_reduce_load_source_image(tex, slice);
124    spd_store(pix, v[0], 0u, slice);
125
126    tex = (workgroup_id * 64u) + vec2(x * 2u + 32u, y * 2u);
127    pix = (workgroup_id * 32u) + vec2(x + 16u, y);
128    v[1] = spd_reduce_load_source_image(tex, slice);
129    spd_store(pix, v[1], 0u, slice);
130
131    tex = (workgroup_id * 64u) + vec2(x * 2u, y * 2u + 32u);
132    pix = (workgroup_id * 32u) + vec2(x, y + 16u);
133    v[2] = spd_reduce_load_source_image(tex, slice);
134    spd_store(pix, v[2], 0u, slice);
135
136    tex = (workgroup_id * 64u) + vec2(x * 2u + 32u, y * 2u + 32u);
137    pix = (workgroup_id * 32u) + vec2(x + 16u, y + 16u);
138    v[3] = spd_reduce_load_source_image(tex, slice);
139    spd_store(pix, v[3], 0u, slice);
140
141    if mips <= 1u { return; }
142
143@if(SUBGROUP_SUPPORT) {
144    v[0] = spd_reduce_quad(v[0]);
145    v[1] = spd_reduce_quad(v[1]);
146    v[2] = spd_reduce_quad(v[2]);
147    v[3] = spd_reduce_quad(v[3]);
148
149    if local_invocation_index % 4u == 0u {
150        spd_store((workgroup_id * 16u) + vec2(x / 2u, y / 2u), v[0], 1u, slice);
151        spd_store_intermediate(x / 2u, y / 2u, v[0]);
152
153        spd_store((workgroup_id * 16u) + vec2(x / 2u + 8u, y / 2u), v[1], 1u, slice);
154        spd_store_intermediate(x / 2u + 8u, y / 2u, v[1]);
155
156        spd_store((workgroup_id * 16u) + vec2(x / 2u, y / 2u + 8u), v[2], 1u, slice);
157        spd_store_intermediate(x / 2u, y / 2u + 8u, v[2]);
158
159        spd_store((workgroup_id * 16u) + vec2(x / 2u + 8u, y / 2u + 8u), v[3], 1u, slice);
160        spd_store_intermediate(x / 2u + 8u, y / 2u + 8u, v[3]);
161    }
162}
163@else {
164    for (var i = 0u; i < 4u; i++) {
165        spd_store_intermediate(x, y, v[i]);
166        workgroupBarrier();
167        if local_invocation_index < 64u {
168            v[i] = spd_reduce_intermediate(
169                vec2(x * 2u + 0u, y * 2u + 0u),
170                vec2(x * 2u + 1u, y * 2u + 0u),
171                vec2(x * 2u + 0u, y * 2u + 1u),
172                vec2(x * 2u + 1u, y * 2u + 1u),
173            );
174            spd_store(vec2(workgroup_id * 16) + vec2(x + (i % 2u) * 8u, y + (i / 2u) * 8u), v[i], 1u, slice);
175        }
176        workgroupBarrier();
177    }
178
179    if local_invocation_index < 64u {
180        spd_store_intermediate(x + 0u, y + 0u, v[0]);
181        spd_store_intermediate(x + 8u, y + 0u, v[1]);
182        spd_store_intermediate(x + 0u, y + 8u, v[2]);
183        spd_store_intermediate(x + 8u, y + 8u, v[3]);
184    }
185}
186}

188fn spd_downsample_next_four(x: u32, y: u32, workgroup_id: vec2u, local_invocation_index: u32, base_mip: u32, mips: u32, slice: u32) {
189    if mips <= base_mip { return; }
190    workgroupBarrier();
191    spd_downsample_mip_2(x, y, workgroup_id, local_invocation_index, base_mip, slice);
192
193    if mips <= base_mip + 1u { return; }
194    workgroupBarrier();
195    spd_downsample_mip_3(x, y, workgroup_id, local_invocation_index, base_mip + 1u, slice);
196
197    if mips <= base_mip + 2u { return; }
198    workgroupBarrier();
199    spd_downsample_mip_4(x, y, workgroup_id, local_invocation_index, base_mip + 2u, slice);
200
201    if mips <= base_mip + 3u { return; }
202    workgroupBarrier();
203    spd_downsample_mip_5(x, y, workgroup_id, local_invocation_index, base_mip + 3u, slice);
204}

206fn spd_downsample_mip_2(x: u32, y: u32, workgroup_id: vec2u, local_invocation_index: u32, base_mip: u32, slice: u32) {
207@if(SUBGROUP_SUPPORT) {
208    var v = spd_load_intermediate(x, y);
209    v = spd_reduce_quad(v);
210    if local_invocation_index % 4u == 0u {
211        spd_store((workgroup_id * 8u) + vec2(x / 2u, y / 2u), v, base_mip, slice);
212        spd_store_intermediate(x + (y / 2u) % 2u, y, v);
213    }
214}
215@else {
216    if local_invocation_index < 64u {
217        let v = spd_reduce_intermediate(
218            vec2(x * 2u + 0u, y * 2u + 0u),
219            vec2(x * 2u + 1u, y * 2u + 0u),
220            vec2(x * 2u + 0u, y * 2u + 1u),
221            vec2(x * 2u + 1u, y * 2u + 1u),
222        );
223        spd_store((workgroup_id * 8u) + vec2(x, y), v, base_mip, slice);
224        spd_store_intermediate(x * 2u + y % 2u, y * 2u, v);
225    }
226}

229fn spd_downsample_mip_3(x: u32, y: u32, workgroup_id: vec2u, local_invocation_index: u32, base_mip: u32, slice: u32) {
230@if(SUBGROUP_SUPPORT)
231    if local_invocation_index < 64u {
232        var v = spd_load_intermediate(x * 2u + y % 2u, y * 2u);
233        v = spd_reduce_quad(v);
234        if local_invocation_index % 4u == 0u {
235            spd_store((workgroup_id * 4u) + vec2(x / 2u, y / 2u), v, base_mip, slice);
236            spd_store_intermediate(x * 2u + y / 2u, y * 2u, v);
237        }
238    }
239@else   // SUBGROUP_SUPPORT
240    if local_invocation_index < 16u {
241        let v = spd_reduce_intermediate(
242            vec2(x * 4u + 0u + 0u, y * 4u + 0u),
243            vec2(x * 4u + 2u + 0u, y * 4u + 0u),
244            vec2(x * 4u + 0u + 1u, y * 4u + 2u),
245            vec2(x * 4u + 2u + 1u, y * 4u + 2u),
246        );
247        spd_store((workgroup_id * 4u) + vec2(x, y), v, base_mip, slice);
248        spd_store_intermediate(x * 4u + y, y * 4u, v);
249    }

252fn spd_downsample_mip_4(x: u32, y: u32, workgroup_id: vec2u, local_invocation_index: u32, base_mip: u32, slice: u32) {
253@if(SUBGROUP_SUPPORT)
254    if local_invocation_index < 16u {
255        var v = spd_load_intermediate(x * 4u + y, y * 4u);
256        v = spd_reduce_quad(v);
257        if local_invocation_index % 4u == 0u {
258            spd_store((workgroup_id * 2u) + vec2(x / 2u, y / 2u), v, base_mip, slice);
259            spd_store_intermediate(x / 2u + y, 0u, v);
260        }
261    }
262@else   // SUBGROUP_SUPPORT
263    if local_invocation_index < 4u {
264        let v = spd_reduce_intermediate(
265            vec2(x * 8u + 0u + 0u + y * 2u, y * 8u + 0u),
266            vec2(x * 8u + 4u + 0u + y * 2u, y * 8u + 0u),
267            vec2(x * 8u + 0u + 1u + y * 2u, y * 8u + 4u),
268            vec2(x * 8u + 4u + 1u + y * 2u, y * 8u + 4u),
269        );
270        spd_store((workgroup_id * 2u) + vec2(x, y), v, base_mip, slice);
271        spd_store_intermediate(x + y * 2u, 0u, v);
272    }

276@if(SUBGROUP_SUPPORT)

291fn spd_downsample_mips_6_7(x: u32, y: u32, mips: u32, slice: u32) {
292    var tex = vec2(x * 4u + 0u, y * 4u + 0u);
293    var pix = vec2(x * 2u + 0u, y * 2u + 0u);
294    let v0 = spd_reduce_load_4(
295        vec2(x * 4u + 0u, y * 4u + 0u),
296        vec2(x * 4u + 1u, y * 4u + 0u),
297        vec2(x * 4u + 0u, y * 4u + 1u),
298        vec2(x * 4u + 1u, y * 4u + 1u),
299        slice
300    );
301    spd_store(pix, v0, 6u, slice);
302
303    tex = vec2(x * 4u + 2u, y * 4u + 0u);
304    pix = vec2(x * 2u + 1u, y * 2u + 0u);
305    let v1 = spd_reduce_load_4(
306        vec2(x * 4u + 2u, y * 4u + 0u),
307        vec2(x * 4u + 3u, y * 4u + 0u),
308        vec2(x * 4u + 2u, y * 4u + 1u),
309        vec2(x * 4u + 3u, y * 4u + 1u),
310        slice
311    );
312    spd_store(pix, v1, 6u, slice);
313
314    tex = vec2(x * 4u + 0u, y * 4u + 2u);
315    pix = vec2(x * 2u + 0u, y * 2u + 1u);
316    let v2 = spd_reduce_load_4(
317        vec2(x * 4u + 0u, y * 4u + 2u),
318        vec2(x * 4u + 1u, y * 4u + 2u),
319        vec2(x * 4u + 0u, y * 4u + 3u),
320        vec2(x * 4u + 1u, y * 4u + 3u),
321        slice
322    );
323    spd_store(pix, v2, 6u, slice);
324
325    tex = vec2(x * 4u + 2u, y * 4u + 2u);
326    pix = vec2(x * 2u + 1u, y * 2u + 1u);
327    let v3 = spd_reduce_load_4(
328        vec2(x * 4u + 2u, y * 4u + 2u),
329        vec2(x * 4u + 3u, y * 4u + 2u),
330        vec2(x * 4u + 2u, y * 4u + 3u),
331        vec2(x * 4u + 3u, y * 4u + 3u),
332        slice
333    );
334    spd_store(pix, v3, 6u, slice);
335
336    if mips < 7u { return; }
337
338    let v = spd_reduce_4(v0, v1, v2, v3);
339    spd_store(vec2(x, y), v, 7u, slice);
340    spd_store_intermediate(x, y, v);
341}

343fn remap_for_wave_reduction(a: u32) -> vec2u {
344    // This function maps linear thread IDs to 2D coordinates in a special pattern
345    // to ensure that neighboring threads process neighboring pixels
346    // For example, this transforms linear thread IDs 0,1,2,3 into a 2×2 square
347
348    // Extract bits to form the X and Y coordinates
349    let x = insertBits(extractBits(a, 2u, 3u), a, 0u, 1u);
350    let y = insertBits(extractBits(a, 3u, 3u), extractBits(a, 1u, 2u), 0u, 2u);
351
352    return vec2u(x, y);
353}

355@if(ARRAY_TEXTURE)
356fn spd_reduce_load_source_image(uv: vec2u, slice: u32) -> vec4f {
357    let texture_coord = (vec2f(uv) + 0.5) * constants.inverse_input_size;
358
359    @if(COMBINE_BIND_GROUP)
360    let result = textureSampleLevel(mip_0, sampler_linear_clamp, texture_coord, slice, 0.0);
361    @if(FIRST_PASS)
362    let result = textureSampleLevel(mip_0, sampler_linear_clamp, texture_coord, slice, 0.0);
363    @if(SECOND_PASS)
364    let result = textureSampleLevel(mip_6, sampler_linear_clamp, texture_coord, slice, 0.0);
365
366@if(SRGB_CONVERSION)
367    return vec4(
368        srgb_from_linear(result.r),
369        srgb_from_linear(result.g),
370        srgb_from_linear(result.b),
371        result.a
372    );
373@else   // SRGB_CONVERSION
374    return result;
375
376}

401@if(ARRAY_TEXTURE)
402fn spd_store(pix: vec2u, value: vec4f, mip: u32, slice: u32) {
403    if mip >= constants.mips { return; }
404    switch mip {
405        @if(COMBINE_BIND_GROUP)
406        case 0u: { textureStore(mip_1, pix, slice, value); }
407        @if(COMBINE_BIND_GROUP)
408        case 1u: { textureStore(mip_2, pix, slice, value); }
409        @if(COMBINE_BIND_GROUP)
410        case 2u: { textureStore(mip_3, pix, slice, value); }
411        @if(COMBINE_BIND_GROUP)
412        case 3u: { textureStore(mip_4, pix, slice, value); }
413        @if(COMBINE_BIND_GROUP)
414        case 4u: { textureStore(mip_5, pix, slice, value); }
415        @if(COMBINE_BIND_GROUP)

512fn spd_store_intermediate(x: u32, y: u32, value: vec4f) {
513    spd_intermediate_r[x][y] = value.x;
514    spd_intermediate_g[x][y] = value.y;
515    spd_intermediate_b[x][y] = value.z;
516    spd_intermediate_a[x][y] = value.w;
517}
518
519fn spd_load_intermediate(x: u32, y: u32) -> vec4f {
520    return vec4(spd_intermediate_r[x][y], spd_intermediate_g[x][y], spd_intermediate_b[x][y], spd_intermediate_a[x][y]);
521}
522
523fn spd_reduce_intermediate(i0: vec2u, i1: vec2u, i2: vec2u, i3: vec2u) -> vec4f {
524    let v0 = spd_load_intermediate(i0.x, i0.y);
525    let v1 = spd_load_intermediate(i1.x, i1.y);
526    let v2 = spd_load_intermediate(i2.x, i2.y);
527    let v3 = spd_load_intermediate(i3.x, i3.y);
528    return spd_reduce_4(v0, v1, v2, v3);
529}

570fn spd_reduce_4(v0: vec4f, v1: vec4f, v2: vec4f, v3: vec4f) -> vec4f {

583fn srgb_from_linear(value: f32) -> f32 {
584    let j = vec3(0.0031308 * 12.92, 12.92, 1.0 / 2.4);
585    let k = vec2(1.055, -0.055);
586    return clamp(j.x, value * j.y, pow(value, j.z) * k.x + k.y);
