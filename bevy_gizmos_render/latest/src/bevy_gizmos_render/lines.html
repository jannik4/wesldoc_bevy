29struct VertexOutput {
30    @builtin(position) clip_position: vec4<f32>,
31    @location(0) color: vec4<f32>,
32    @location(1) uv: f32,
33    @location(2) line_fraction: f32,
34};
35
36const EPSILON: f32 = 4.88e-04;
37
38@vertex
39fn vertex(vertex: VertexInput) -> VertexOutput {
40    var positions = array<vec2<f32>, 6>(
41        vec2(-0.5, 0.),
42        vec2(-0.5, 1.),
43        vec2(0.5, 1.),
44        vec2(-0.5, 0.),
45        vec2(0.5, 1.),
46        vec2(0.5, 0.)
47    );
48    let position = positions[vertex.index];
49
50    let world_from_local = affine3_to_square(line_gizmo.world_from_local);
51
52    // algorithm based on https://wwwtyro.net/2019/11/18/instanced-lines.html
53    var clip_a = view.clip_from_world * world_from_local * vec4(vertex.position_a, 1.);
54    var clip_b = view.clip_from_world * world_from_local * vec4(vertex.position_b, 1.);
55
56    // Manual near plane clipping to avoid errors when doing the perspective divide inside this shader.
57    clip_a = clip_near_plane(clip_a, clip_b);
58    clip_b = clip_near_plane(clip_b, clip_a);
59    let clip = mix(clip_a, clip_b, position.y);
60
61    let resolution = view.viewport.zw;
62    let screen_a = resolution * (0.5 * clip_a.xy / clip_a.w + 0.5);
63    let screen_b = resolution * (0.5 * clip_b.xy / clip_b.w + 0.5);
64
65    let y_basis = normalize(screen_b - screen_a);
66    let x_basis = vec2(-y_basis.y, y_basis.x);
67
68    var color = mix(vertex.color_a, vertex.color_b, position.y);
69
70    var line_width = line_gizmo.line_width;
71    var alpha = 1.;
72
73    var uv: f32;
74@if(PERSPECTIVE) {
75    line_width /= clip.w;
76
77    // get height of near clipping plane in world space
78    let pos0 = view.view_from_clip * vec4(0, -1, 0, 1); // Bottom of the screen
79    let pos1 = view.view_from_clip * vec4(0, 1, 0, 1); // Top of the screen
80    let near_clipping_plane_height = length(pos0.xyz - pos1.xyz);
81
82    // We can't use vertex.position_X because we may have changed the clip positions with clip_near_plane
83    let position_a = view.world_from_clip * clip_a;
84    let position_b = view.world_from_clip * clip_b;
85    let world_distance = length(position_a.xyz - position_b.xyz);
86
87    // Offset to compensate for moved clip positions. If removed dots on lines will slide when position a is ofscreen.
88    let clipped_offset = length(position_a.xyz - vertex.position_a);
89
90    uv = (clipped_offset + position.y * world_distance) * resolution.y / near_clipping_plane_height / line_gizmo.line_width;
91} @else {
92    // Get the distance of b to the camera along camera axes
93    let camera_b = view.view_from_clip * clip_b;
94
95    // This differentiates between orthographic and perspective cameras.
96    // For orthographic cameras no depth adaptment (depth_adaptment = 1) is needed.
97    var depth_adaptment: f32;
98    if (clip_b.w == 1.0) {
99        depth_adaptment = 1.0;
100    }
101    else {
102        depth_adaptment = -camera_b.z;
103    }
104    uv = position.y * depth_adaptment * length(screen_b - screen_a) / line_gizmo.line_width;
105}
106
107    // Line thinness fade from https://acegikmo.com/shapes/docs/#anti-aliasing
108    if line_width > 0.0 && line_width < 1. {
109        color.a *= line_width;
110        line_width = 1.;
111    }
112
113    let x_offset = line_width * position.x * x_basis;
114    let screen = mix(screen_a, screen_b, position.y) + x_offset;
115
116    var depth: f32;
117    if line_gizmo.depth_bias >= 0. {
118        depth = clip.z * (1. - line_gizmo.depth_bias);
119    } else {
120        // depth * (clip.w / depth)^-depth_bias. So that when -depth_bias is 1.0, this is equal to clip.w
121        // and when equal to 0.0, it is exactly equal to depth.
122        // the epsilon is here to prevent the depth from exceeding clip.w when -depth_bias = 1.0
123        // clip.w represents the near plane in homogeneous clip space in bevy, having a depth
124        // of this value means nothing can be in front of this
125        // The reason this uses an exponential function is that it makes it much easier for the
126        // user to chose a value that is convenient for them
127        depth = clip.z * exp2(-line_gizmo.depth_bias * log2(clip.w / clip.z - EPSILON));
128    }
129
130    var clip_position = vec4(clip.w * ((2. * screen) / resolution - 1.), depth, clip.w);
131
132    let line_fraction = 2.0 * line_gizmo.line_scale / (line_gizmo.gap_scale + line_gizmo.line_scale);
133    uv /= (line_gizmo.gap_scale + line_gizmo.line_scale) / 2.0;
134    return VertexOutput(clip_position, color, uv, line_fraction);
135}

137fn clip_near_plane(a: vec4<f32>, b: vec4<f32>) -> vec4<f32> {
138    // Move a if a is behind the near plane and b is in front.
139    if a.z > a.w && b.z <= b.w {
140        // Interpolate a towards b until it's at the near plane.
141        let distance_a = a.z - a.w;
142        let distance_b = b.z - b.w;
143        // Add an epsilon to the interpolator to ensure that the point is
144        // not just behind the clip plane due to floating-point imprecision,
145        // which can lead to lines showing up where they should not.
146        let t = distance_a / (distance_a - distance_b) + EPSILON;
147        return mix(a, b, t);
148    }
149    return a;
150}

152struct FragmentInput {
153    @builtin(position) position: vec4<f32>,
154    @location(0) color: vec4<f32>,
155    @location(1) uv: f32,
156    @location(2) line_fraction: f32,
157};
158
159struct FragmentOutput {
160    @location(0) color: vec4<f32>,
161};
162
163@fragment
164fn fragment_solid(in: FragmentInput) -> FragmentOutput {
165    return FragmentOutput(in.color);
166}
167@fragment
168fn fragment_dotted(in: FragmentInput) -> FragmentOutput {
169    var alpha: f32;
170    @if(PERSPECTIVE)
171    alpha = 1 - floor(in.uv % 2.0);
172    @else
173    alpha = 1 - floor((in.uv * in.position.w) % 2.0);
174    
175    return FragmentOutput(vec4(in.color.xyz, in.color.w * alpha));
176}
177
178@fragment
179fn fragment_dashed(in: FragmentInput) -> FragmentOutput {
180    @if(PERSPECTIVE)
181    let uv = in.uv;
182    @else
183    let uv = in.uv * in.position.w;
184    let alpha = 1.0 - floor(min((uv % 2.0) / in.line_fraction, 1.0));
185    
186    return FragmentOutput(vec4(in.color.xyz, in.color.w * alpha));
187}
