16// NOTE FOR CONVENTIONS: 
17// r:
18//   radius, or distance from planet center 
19//
20// altitude:
21//   distance from planet **surface**
22//
23// mu:
24//   cosine of the zenith angle of a ray with
25//   respect to the planet normal
26//
27// atmosphere space:
28//   abbreviated as "as" (contrast with vs, cs, ws), this space is similar
29//   to view space, but with the camera positioned horizontally on the planet
30//   surface, so the horizon is a horizontal line centered vertically in the
31//   frame. This enables the non-linear latitude parametrization the paper uses 
32//   to concentrate detail near the horizon 
33

35// CONSTANTS
36const FRAC_PI: f32 = 0.3183098862; // 1 / π
37const FRAC_2_PI: f32 = 0.15915494309;  // 1 / (2π)
38const FRAC_3_16_PI: f32 = 0.0596831036594607509; // 3 / (16π)
39const FRAC_4_PI: f32 = 0.07957747154594767; // 1 / (4π)
40const ROOT_2: f32 = 1.41421356; // √2
41const EPSILON: f32 = 1.0; // 1 meter
42const MIN_EXTINCTION: vec3<f32> = vec3(1e-12);

44// During raymarching, each segment is sampled at a single point. This constant determines
45// where in the segment that sample is taken (0.0 = start, 0.5 = middle, 1.0 = end).
46// We use 0.3 to sample closer to the start of each segment, which better approximates
47// the exponential falloff of atmospheric density.
48const MIDPOINT_RATIO: f32 = 0.3;

52fn unit_to_sub_uvs(val: vec2<f32>, resolution: vec2<f32>) -> vec2<f32> {
53    return (val + 0.5f / resolution) * (resolution / (resolution + 1.0f));
54}

56fn sub_uvs_to_unit(val: vec2<f32>, resolution: vec2<f32>) -> vec2<f32> {
57    return (val - 0.5f / resolution) * (resolution / (resolution - 1.0f));
58}

60fn multiscattering_lut_r_mu_to_uv(r: f32, mu: f32) -> vec2<f32> {
61    let u = 0.5 + 0.5 * mu;
62    let v = saturate((r - atmosphere.inner_radius) / (atmosphere.outer_radius - atmosphere.inner_radius)); //TODO
63    return unit_to_sub_uvs(vec2(u, v), vec2<f32>(settings.multiscattering_lut_size));
64}

66fn multiscattering_lut_uv_to_r_mu(uv: vec2<f32>) -> vec2<f32> {
67    let adj_uv = sub_uvs_to_unit(uv, vec2<f32>(settings.multiscattering_lut_size));
68    let r = mix(atmosphere.inner_radius, atmosphere.outer_radius, adj_uv.y);
69    let mu = adj_uv.x * 2 - 1;
70    return vec2(r, mu);
71}

73fn sky_view_lut_r_mu_azimuth_to_uv(r: f32, mu: f32, azimuth: f32) -> vec2<f32> {
74    let u = (azimuth * FRAC_2_PI) + 0.5;
75
76    let v_horizon = sqrt(r * r - atmosphere.inner_radius * atmosphere.inner_radius);
77    let cos_beta = v_horizon / r;
78    // Using fast_acos_4 for better precision at small angles
79    // to avoid artifacts at the horizon
80    let beta = fast_acos_4(cos_beta);
81    let horizon_zenith = PI - beta;
82    let view_zenith = fast_acos_4(mu);
83
84    // Apply non-linear transformation to compress more texels 
85    // near the horizon where high-frequency details matter most
86    // l is latitude in [-π/2, π/2] and v is texture coordinate in [0,1]
87    let l = view_zenith - horizon_zenith;
88    let abs_l = abs(l);
89
90    let v = 0.5 + 0.5 * sign(l) * sqrt(abs_l / HALF_PI);
91
92    return unit_to_sub_uvs(vec2(u, v), vec2<f32>(settings.sky_view_lut_size));
93}

95fn sky_view_lut_uv_to_zenith_azimuth(r: f32, uv: vec2<f32>) -> vec2<f32> {
96    let adj_uv = sub_uvs_to_unit(vec2(uv.x, 1.0 - uv.y), vec2<f32>(settings.sky_view_lut_size));
97    let azimuth = (adj_uv.x - 0.5) * PI_2;
98
99    // Horizon parameters
100    let v_horizon = sqrt(r * r - atmosphere.inner_radius * atmosphere.inner_radius);
101    let cos_beta = v_horizon / r;
102    let beta = fast_acos_4(cos_beta);
103    let horizon_zenith = PI - beta;
104
105    // Inverse of horizon-detail mapping to recover original latitude from texture coordinate
106    let t = abs(2.0 * (adj_uv.y - 0.5));
107    let l = sign(adj_uv.y - 0.5) * HALF_PI * t * t;
108
109    return vec2(horizon_zenith - l, azimuth);
110}

131fn ndc_to_camera_dist(ndc: vec3<f32>) -> f32 {
132    let view_pos = view.view_from_clip * vec4(ndc, 1.0);
133    let p_view = view_pos.xyz / view_pos.w;
134    let p_world = (view.world_from_view * vec4(p_view, 1.0)).xyz;
135    let p_atmo = (atmosphere.world_to_atmosphere * vec4(p_world, 1.0)).xyz;
136    let cam_atmo = (atmosphere.world_to_atmosphere * vec4(view.world_position, 1.0)).xyz;
137    return length(p_atmo - cam_atmo);
138}

140// RGB channels: total inscattered light along the camera ray to the current sample.
141// A channel: average transmittance across all wavelengths to the current sample.
142fn sample_aerial_view_lut(uv: vec2<f32>, t: f32) -> vec3<f32> {
143    let t_max = settings.aerial_view_lut_max_distance;
144    let num_slices = f32(settings.aerial_view_lut_size.z);
145    // Each texel stores the value of the scattering integral over the whole slice,
146    // which requires us to offset the w coordinate by half a slice. For
147    // example, if we wanted the value of the integral at the boundary between slices,
148    // we'd need to sample at the center of the previous slice, and vice-versa for
149    // sampling in the center of a slice.
150    let uvw = vec3(uv, saturate(t / t_max - 0.5 / num_slices));
151    let sample = textureSampleLevel(aerial_view_lut, atmosphere_lut_sampler, uvw, 0.0);
152    // Since sampling anywhere between w=0 and w=t_slice will clamp to the first slice,
153    // we need to do a linear step over the first slice towards zero at the camera's
154    // position to recover the correct integral value.
155    let t_slice = t_max / num_slices;
156    let fade = saturate(t / t_slice);
157    // Recover the values from log space
158    return exp(sample.rgb) * fade;
159}

163const ABSORPTION_DENSITY: f32 = 0.0;
164const SCATTERING_DENSITY: f32 = 1.0;

166// samples from the atmosphere density LUT.
167//
168// calling with `component = 0.0` will return the atmosphere's absorption density,
169// while calling with `component = 1.0` will return the atmosphere's scattering density.
170fn sample_density_lut(r: f32, component: f32) -> vec3<f32> {
171    // sampler clamps to [0, 1] anyways, no need to clamp the altitude
172    let normalized_altitude = (r - atmosphere.inner_radius) / (atmosphere.outer_radius - atmosphere.inner_radius);
173    let uv = vec2(1.0 - normalized_altitude, component);
174    return textureSampleLevel(medium_density_lut, medium_sampler, uv, 0.0).xyz;
175}

177// samples from the atmosphere scattering LUT. `neg_LdotV` is the dot product
178// of the light direction and the incoming view vector.
179// Nonlinear phase mapping to mitigate banding in low-resolution LUTs.
180const PHASE_MAPPING_N: f32 = 0.5;
181fn sample_scattering_lut(r: f32, neg_LdotV: f32) -> vec3<f32> {
182    let normalized_altitude = (r - atmosphere.inner_radius) / (atmosphere.outer_radius - atmosphere.inner_radius);
183    let phase_uv = 0.5 + 0.5 * sign(neg_LdotV) * (1.0 - pow(1.0 - abs(neg_LdotV), PHASE_MAPPING_N));
184    let uv = vec2(1.0 - normalized_altitude, phase_uv);
185    return textureSampleLevel(medium_scattering_lut, medium_sampler, uv, 0.0).xyz;
186}

188/// evaluates L_scat, equation 3 in the paper, which gives the total single-order scattering towards the view at a single point
189fn sample_local_inscattering(local_scattering: vec3<f32>, ray_dir: vec3<f32>, world_pos: vec3<f32>) -> vec3<f32> {
190    let local_r = length(world_pos);
191    let local_up = normalize(world_pos);
192    var inscattering = vec3(0.0);
193    for (var light_i: u32 = 0u; light_i < lights.n_directional_lights; light_i++) {
194        let light = &lights.directional_lights[light_i];
195
196        let mu_light = dot((*light).direction_to_light, local_up);
197
198        // -(L . V) == (L . -V). -V here is our ray direction, which points away from the view
199        // instead of towards it (as is the convention for V)
200        let neg_LdotV = dot((*light).direction_to_light, ray_dir);
201
202        let transmittance_to_light = sample_transmittance_lut(local_r, mu_light);
203        let shadow_factor = transmittance_to_light * f32(!ray_intersects_ground(local_r, mu_light));
204        let scattering_coeff = sample_scattering_lut(local_r, neg_LdotV);
205
206        // Transmittance from scattering event to light source
207        let scattering_factor = shadow_factor * scattering_coeff;
208
209        // Additive factor from the multiscattering LUT
210        let psi_ms = sample_multiscattering_lut(local_r, mu_light);
211        let multiscattering_factor = psi_ms * local_scattering;
212
213        inscattering += (*light).color.rgb * (scattering_factor + multiscattering_factor);
214    }
215    return inscattering;
216}

218fn sample_sun_radiance(ray_dir_ws: vec3<f32>) -> vec3<f32> {
219    let view_pos = get_view_position();
220    let r = length(view_pos);
221    let up = normalize(view_pos);
222    let mu_view = dot(ray_dir_ws, up);
223    let shadow_factor = f32(!ray_intersects_ground(r, mu_view));
224    var sun_radiance = vec3(0.0);
225    for (var light_i: u32 = 0u; light_i < lights.n_directional_lights; light_i++) {
226        let light = &lights.directional_lights[light_i];
227        let neg_LdotV = dot((*light).direction_to_light, ray_dir_ws);
228        let angle_to_sun = fast_acos(clamp(neg_LdotV, -1.0, 1.0));
229        let w = max(0.5 * fwidth(angle_to_sun), 1e-6);
230        let sun_angular_size = (*light).sun_disk_angular_size;
231        let sun_intensity = (*light).sun_disk_intensity;
232        if sun_angular_size > 0.0 && sun_intensity > 0.0 {
233            let factor = 1 - smoothstep(sun_angular_size * 0.5 - w, sun_angular_size * 0.5 + w, angle_to_sun);
234            let sun_solid_angle = (sun_angular_size * sun_angular_size) * 0.25 * PI;
235            sun_radiance += ((*light).color.rgb / sun_solid_angle) * sun_intensity * factor * shadow_factor;
236        }
237    }
238    return sun_radiance;
239}

241fn calculate_visible_sun_ratio(atmosphere: Atmosphere, r: f32, mu: f32, sun_angular_size: f32) -> f32 {
242    let inner_radius = atmosphere.inner_radius;
243    // Calculate the angle between horizon and sun center
244    // Invert the horizon angle calculation to fix shading direction
245    let horizon_cos = -sqrt(1.0 - (inner_radius * inner_radius) / (r * r));
246    let horizon_angle = fast_acos_4(horizon_cos);
247    let sun_zenith_angle = fast_acos_4(mu);
248    
249    // If sun is completely above horizon
250    if sun_zenith_angle + sun_angular_size * 0.5 <= horizon_angle {
251        return 1.0;
252    }
253    
254    // If sun is completely below horizon
255    if sun_zenith_angle - sun_angular_size * 0.5 >= horizon_angle {
256        return 0.0;
257    }
258    
259    // Calculate partial visibility using circular segment area formula
260    let d = (horizon_angle - sun_zenith_angle) / (sun_angular_size * 0.5);
261    let visible_ratio = 0.5 + d * 0.5;
262    return clamp(visible_ratio, 0.0, 1.0);
263}

267/// Clamp a position to the planet surface (with a small epsilon) to avoid underground artifacts.
268fn clamp_to_surface(atmosphere: Atmosphere, position: vec3<f32>) -> vec3<f32> {
269    let min_radius = atmosphere.inner_radius + EPSILON;
270    let r = length(position);
271    if r < min_radius {
272        let up = normalize(position);
273        return up * min_radius;
274    }
275    return position;
276}

326// Modified from skybox.wgsl. For this pass we don't need to apply a separate sky transform or consider camera viewport.
327// Returns a normalized ray direction in world space.
328fn uv_to_ray_direction(uv: vec2<f32>) -> vec3<f32> {
329    // Using world positions of the fragment and camera to calculate a ray direction
330    // breaks down at large translations. This code only needs to know the ray direction.
331    // The ray direction is along the direction from the camera to the fragment position.
332    // In view space, the camera is at the origin, so the view space ray direction is
333    // along the direction of the fragment position - (0,0,0) which is just the
334    // fragment position.
335    // Use the position on the near clipping plane to avoid -inf world position
336    // because the far plane of an infinite reverse projection is at infinity.
337    let view_position_homogeneous = view.view_from_clip * vec4(
338        uv_to_ndc(uv),
339        1.0,
340        1.0,
341    );
342
343    let view_ray_direction = view_position_homogeneous.xyz / view_position_homogeneous.w;
344    // Transforming the view space ray direction by the inverse view matrix, transforms the
345    // direction to world space. Note that the w element is set to 0.0, as this is a
346    // vector direction, not a position, That causes the matrix multiplication to ignore
347    // the translations from the view matrix.
348    let ray_direction = (view.world_from_view * vec4(view_ray_direction, 0.0)).xyz;
349
350    return normalize(ray_direction);
351}

353fn zenith_azimuth_to_ray_dir(zenith: f32, azimuth: f32) -> vec3<f32> {
354    let sin_zenith = sin(zenith);
355    let mu = cos(zenith);
356    let sin_azimuth = sin(azimuth);
357    let cos_azimuth = cos(azimuth);
358    return vec3(sin_azimuth * sin_zenith, mu, -cos_azimuth * sin_zenith);
359}

361struct RaymarchSegment {
362    start: f32,
363    end: f32,
364}
365
366fn get_raymarch_segment(r: f32, mu: f32) -> RaymarchSegment {
367    // Get both intersection points with atmosphere
368    let atmosphere_intersections = ray_sphere_intersect(r, mu, atmosphere.outer_radius);
369    let ground_intersections = ray_sphere_intersect(r, mu, atmosphere.inner_radius);
370
371    var segment: RaymarchSegment;
372
373    if r < atmosphere.inner_radius {
374        // Inside planet - start from bottom of atmosphere
375        segment.start = ground_intersections.y; // Use second intersection point with ground
376        segment.end = atmosphere_intersections.y;
377    } else if r < atmosphere.outer_radius {
378        // Inside atmosphere
379        segment.start = 0.0;
380        segment.end = select(atmosphere_intersections.y, ground_intersections.x, ray_intersects_ground(r, mu));
381    } else {
382        // Outside atmosphere
383        if atmosphere_intersections.x < 0.0 {
384            // No intersection with atmosphere
385            return segment;
386        }
387        // Start at atmosphere entry, end at exit or ground
388        segment.start = atmosphere_intersections.x;
389        segment.end = select(atmosphere_intersections.y, ground_intersections.x, ray_intersects_ground(r, mu));
390    }
391
392    return segment;
393}

395struct RaymarchResult {
396    inscattering: vec3<f32>,
397    transmittance: vec3<f32>,
398}
399
400fn raymarch_atmosphere(
401    pos: vec3<f32>,
402    ray_dir: vec3<f32>,
403    t_max: f32,
404    max_samples: u32,
405    uv: vec2<f32>,
406    ground: bool
407) -> RaymarchResult {
408    let r = length(pos);
409    let up = normalize(pos);
410    let mu = dot(ray_dir, up);
411
412    // Optimization: Reduce sample count at close proximity to the scene
413    let sample_count = mix(1.0, f32(max_samples), saturate(t_max * 0.01));
414
415    let segment = get_raymarch_segment(r, mu);
416    let t_start = segment.start;
417    var t_end = segment.end;
418
419    t_end = min(t_end, t_max);
420    let t_total = t_end - t_start;
421
422    var result: RaymarchResult;
423    result.inscattering = vec3(0.0);
424    result.transmittance = vec3(1.0);
425
426    // Skip if invalid segment
427    if t_total <= 0.0 {
428        return result;
429    }
430
431    var prev_t = t_start;
432    var optical_depth = vec3(0.0);
433    for (var s = 0.0; s < sample_count; s += 1.0) {
434        // Linear distribution from atmosphere entry to exit/ground
435        let t_i = t_start + t_total * (s + MIDPOINT_RATIO) / sample_count;
436        let dt_i = (t_i - prev_t);
437        prev_t = t_i;
438
439        let sample_pos = pos + ray_dir * t_i;
440        let local_r = length(sample_pos);
441        let local_up = normalize(sample_pos);
442
443        let absorption = sample_density_lut(local_r, ABSORPTION_DENSITY);
444        let scattering = sample_density_lut(local_r, SCATTERING_DENSITY);
445        let extinction = absorption + scattering;
446
447        let sample_optical_depth = extinction * dt_i;
448        optical_depth += sample_optical_depth;
449        let sample_transmittance = exp(-sample_optical_depth);
450
451        let inscattering = sample_local_inscattering(
452            scattering,
453            ray_dir,
454            sample_pos
455        );
456
457        let s_int = (inscattering - inscattering * sample_transmittance) / max(extinction, MIN_EXTINCTION);
458        result.inscattering += result.transmittance * s_int;
459
460        result.transmittance *= sample_transmittance;
461        if all(result.transmittance < vec3(0.001)) {
462            break;
463        }
464    }
465
466    // include reflected luminance from planet ground 
467    if ground && ray_intersects_ground(r, mu) {
468        for (var light_i: u32 = 0u; light_i < lights.n_directional_lights; light_i++) {
469            let light = &lights.directional_lights[light_i];
470            let light_dir = (*light).direction_to_light;
471            let light_color = (*light).color.rgb;
472            let transmittance_to_ground = exp(-optical_depth);
473            // position on the sphere and get the sphere normal (up)
474            let sphere_point = pos + ray_dir * t_end;
475            let sphere_normal = normalize(sphere_point);
476            let mu_light = dot(light_dir, sphere_normal);
477            let transmittance_to_light = sample_transmittance_lut(0.0, mu_light);
478            let light_luminance = transmittance_to_light * max(mu_light, 0.0) * light_color;
479            // Normalized Lambert BRDF
480            let ground_luminance = transmittance_to_ground * atmosphere.ground_albedo / PI;
481            result.inscattering += ground_luminance * light_luminance;
482        }
483    }
484
485    return result;
486}
