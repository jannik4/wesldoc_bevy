216@if(MESHLET_VISIBILITY_BUFFER_RASTER_PASS)
217fn get_meshlet_vertex_position(meshlet: ptr<function, Meshlet>, vertex_id: u32) -> vec3<f32> {
218    // Get bitstream start for the vertex
219    let unpacked = unpack4xU8((*meshlet).packed_b);
220    let bits_per_channel = unpacked.xyz;
221    let bits_per_vertex = bits_per_channel.x + bits_per_channel.y + bits_per_channel.z;
222    var start_bit = (*meshlet).start_vertex_position_bit + (vertex_id * bits_per_vertex);
223
224    // Read each vertex channel from the bitstream
225    var vertex_position_packed = vec3(0u);
226    for (var i = 0u; i < 3u; i++) {
227        let lower_word_index = start_bit / 32u;
228        let lower_word_bit_offset = start_bit & 31u;
229        var next_32_bits = meshlet_vertex_positions[lower_word_index] >> lower_word_bit_offset;
230        if lower_word_bit_offset + bits_per_channel[i] > 32u {
231            next_32_bits |= meshlet_vertex_positions[lower_word_index + 1u] << (32u - lower_word_bit_offset);
232        }
233        vertex_position_packed[i] = extractBits(next_32_bits, 0u, bits_per_channel[i]);
234        start_bit += bits_per_channel[i];
235    }
236
237    // Remap [0, range_max - range_min] vec3<u32> to [range_min, range_max] vec3<f32>
238    var vertex_position = vec3<f32>(vertex_position_packed) + vec3(
239        (*meshlet).min_vertex_position_channel_x,
240        (*meshlet).min_vertex_position_channel_y,
241        (*meshlet).min_vertex_position_channel_z,
242    );
243
244    // Reverse vertex quantization
245    let vertex_position_quantization_factor = unpacked.w;
246    vertex_position /= f32(1u << vertex_position_quantization_factor) * CENTIMETERS_PER_METER;
247
248    return vertex_position;
249}
