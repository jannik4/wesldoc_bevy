11@vertex
12fn vertex(@builtin(vertex_index) vertex_input: u32) -> @builtin(position) vec4<f32> {
13    let vertex_index = vertex_input % 3u;
14    let material_id = vertex_input / 3u;
15    let material_depth = f32(material_id) / 65535.0;
16    let uv = vec2<f32>(vec2(vertex_index >> 1u, vertex_index & 1u)) * 2.0;
17    return vec4(uv_to_ndc(uv), material_depth, 1.0);
18}

20@fragment
21fn fragment(@builtin(position) frag_coord: vec4<f32>) -> @location(0) vec4<f32> {
22    let vertex_output = resolve_vertex_output(frag_coord);
23    var rng = vertex_output.cluster_id;
24    let color = vec3(rand_f(&rng), rand_f(&rng), rand_f(&rng));
25    return vec4(color, 1.0);
26}

33    var out: prepass_io::FragmentOutput;

47    @if(DEFERRED_PREPASS)
