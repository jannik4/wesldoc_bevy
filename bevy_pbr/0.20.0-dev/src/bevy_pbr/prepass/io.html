66// The instance_index parameter must match vertex_in.instance_index. This is a work around for a wgpu dx12 bug.
67// See https://github.com/gfx-rs/naga/issues/2416
68fn decompress_vertex(vertex_in: Vertex, instance_index: u32) -> UncompressedVertex {
69    let mesh_metadata = package::render::mesh_functions::get_metadata(instance_index);
70    var uncompressed_vertex: UncompressedVertex;
71    uncompressed_vertex.instance_index = instance_index;
72@if(VERTEX_POSITIONS_COMPRESSED)
73    uncompressed_vertex.position = bevy_render::utils::decompress_vertex_position(vertex_in.compressed_position, mesh_metadata.aabb_center, mesh_metadata.aabb_half_extents);
74@else
75    uncompressed_vertex.position = vertex_in.position;
76@if(NORMAL_PREPASS_OR_DEFERRED_PREPASS && VERTEX_NORMALS && VERTEX_NORMALS_COMPRESSED)
77    uncompressed_vertex.normal = bevy_render::utils::decompress_vertex_normal(vertex_in.compressed_normal);
78@if(NORMAL_PREPASS_OR_DEFERRED_PREPASS && VERTEX_NORMALS && !VERTEX_NORMALS_COMPRESSED)
79    uncompressed_vertex.normal = vertex_in.normal;
80@if(NORMAL_PREPASS_OR_DEFERRED_PREPASS && VERTEX_TANGENTS && VERTEX_TANGENTS_COMPRESSED)
81    uncompressed_vertex.tangent = bevy_render::utils::decompress_vertex_tangent(vertex_in.compressed_tangent);
82@if(NORMAL_PREPASS_OR_DEFERRED_PREPASS && VERTEX_TANGENTS && !VERTEX_TANGENTS_COMPRESSED)
83    uncompressed_vertex.tangent = vertex_in.tangent;
84@if(VERTEX_UVS_A && VERTEX_UVS_A_COMPRESSED) {
85    let uv_min_and_extents_a = mesh_metadata.uv_channels_min_and_extents[0];
86    uncompressed_vertex.uv = bevy_render::utils::decompress_vertex_uv(vertex_in.compressed_uv, uv_min_and_extents_a);
87}
88@if(VERTEX_UVS_A && !VERTEX_UVS_A_COMPRESSED)
89    uncompressed_vertex.uv = vertex_in.uv;
90@if(VERTEX_UVS_B && VERTEX_UVS_B_COMPRESSED) {
91    let uv_min_and_extents_b = mesh_metadata.uv_channels_min_and_extents[1];
92    uncompressed_vertex.uv_b = bevy_render::utils::decompress_vertex_uv(vertex_in.compressed_uv_b, uv_min_and_extents_b);
93}
94@if(VERTEX_UVS_B && !VERTEX_UVS_B_COMPRESSED)
95    uncompressed_vertex.uv_b = vertex_in.uv_b;
96@if(VERTEX_COLORS)
97    uncompressed_vertex.color = vertex_in.color;
98@if(SKINNED) {
99    uncompressed_vertex.joint_indices = vertex_in.joint_indices;
100    uncompressed_vertex.joint_weights = vertex_in.joint_weights;
101}
102@if(MORPH_TARGETS)
103    uncompressed_vertex.index = vertex_in.index;
104    return uncompressed_vertex;
105}
