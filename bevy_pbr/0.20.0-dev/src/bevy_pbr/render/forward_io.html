57// The instance_index parameter must match vertex_in.instance_index. This is a work around for a wgpu dx12 bug.
58// See https://github.com/gfx-rs/naga/issues/2416
59fn decompress_vertex(vertex_in: Vertex, instance_index: u32) -> UncompressedVertex {
60    let mesh_metadata = package::render::mesh_functions::get_metadata(instance_index);
61    var uncompressed_vertex: UncompressedVertex;
62    uncompressed_vertex.instance_index = instance_index;
63@if(VERTEX_POSITIONS && VERTEX_POSITIONS_COMPRESSED)
64    uncompressed_vertex.position = bevy_render::utils::decompress_vertex_position(vertex_in.compressed_position, mesh_metadata.aabb_center, mesh_metadata.aabb_half_extents);
65@if(VERTEX_POSITIONS && !VERTEX_POSITIONS_COMPRESSED)
66    uncompressed_vertex.position = vertex_in.position;
67@if(VERTEX_NORMALS && VERTEX_NORMALS_COMPRESSED)
68    uncompressed_vertex.normal = bevy_render::utils::decompress_vertex_normal(vertex_in.compressed_normal);
69@if(VERTEX_NORMALS && !VERTEX_NORMALS_COMPRESSED)
70    uncompressed_vertex.normal = vertex_in.normal;
71@if(VERTEX_UVS_A && VERTEX_UVS_A_COMPRESSED) {
72    let uv_min_and_extents_a = mesh_metadata.uv_channels_min_and_extents[0];
73    uncompressed_vertex.uv = bevy_render::utils::decompress_vertex_uv(vertex_in.compressed_uv, uv_min_and_extents_a);
74}
75@if(VERTEX_UVS_A && !VERTEX_UVS_A_COMPRESSED)
76    uncompressed_vertex.uv = vertex_in.uv;
77@if(VERTEX_UVS_B && VERTEX_UVS_B_COMPRESSED) {
78    let uv_min_and_extents_b = mesh_metadata.uv_channels_min_and_extents[1];
79    uncompressed_vertex.uv_b = bevy_render::utils::decompress_vertex_uv(vertex_in.compressed_uv_b, uv_min_and_extents_b);
80}
81@if(VERTEX_UVS_B && !VERTEX_UVS_B_COMPRESSED)
82    uncompressed_vertex.uv_b = vertex_in.uv_b;
83@if(VERTEX_TANGENTS && VERTEX_TANGENTS_COMPRESSED)
84    uncompressed_vertex.tangent = bevy_render::utils::decompress_vertex_tangent(vertex_in.compressed_tangent);
85@if(VERTEX_TANGENTS && !VERTEX_TANGENTS_COMPRESSED)
86    uncompressed_vertex.tangent = vertex_in.tangent;
87@if(VERTEX_COLORS)
88    uncompressed_vertex.color = vertex_in.color;
89@if(SKINNED) {
90    uncompressed_vertex.joint_indices = vertex_in.joint_indices;
91    uncompressed_vertex.joint_weights = vertex_in.joint_weights;
92}
93@if(MORPH_TARGETS)
94    uncompressed_vertex.index = vertex_in.index;
95    return uncompressed_vertex;
96}
