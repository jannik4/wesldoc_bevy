4/// World space:
5/// +y is up
6
7/// View space:
8/// -z is forward, +x is right, +y is up
9/// Forward is from the camera position into the scene.
10/// (0.0, 0.0, -1.0) is linear distance of 1.0 in front of the camera's view relative to the camera's rotation
11/// (0.0, 1.0, 0.0) is linear distance of 1.0 above the camera's view relative to the camera's rotation
12
13/// NDC (normalized device coordinate):
14/// https://www.w3.org/TR/webgpu/#coordinate-systems
15/// (-1.0, -1.0) in NDC is located at the bottom-left corner of NDC
16/// (1.0, 1.0) in NDC is located at the top-right corner of NDC
17/// Z is depth where: 
18///    1.0 is near clipping plane
19///    Perspective projection: 0.0 is inf far away
20///    Orthographic projection: 0.0 is far clipping plane
21
22/// Clip space:
23/// This is NDC before the perspective divide, still in homogenous coordinate space.
24/// Dividing a clip space point by its w component yields a point in NDC space.
25
26/// UV space:
27/// 0.0, 0.0 is the top left
28/// 1.0, 1.0 is the bottom right
29

35/// Convert a view space position to world space
36fn position_view_to_world(view_pos: vec3<f32>) -> vec3<f32> {
37    let world_pos = view_bindings::view.world_from_view * vec4(view_pos, 1.0);
38    return world_pos.xyz;
39}

41/// Convert a clip space position to world space
42fn position_clip_to_world(clip_pos: vec4<f32>) -> vec3<f32> {
43    let world_pos = view_bindings::view.world_from_clip * clip_pos;
44    return world_pos.xyz;
45}

160// DEPTH -----------

163/// Retrieve the perspective camera near clipping plane
164fn perspective_camera_near() -> f32 {
165    return view_bindings::view.clip_from_view[3][2];
166}
167
168/// Convert ndc depth to linear view z. 
169/// Note: Depth values in front of the camera will be negative as -z is forward
170fn depth_ndc_to_view_z(ndc_depth: f32) -> f32 {
171@if(VIEW_PROJECTION_PERSPECTIVE)
172    return -perspective_camera_near() / ndc_depth;
173@elif(VIEW_PROJECTION_ORTHOGRAPHIC)
174    return -(view_bindings::view.clip_from_view[3][2] - ndc_depth) / view_bindings::view.clip_from_view[2][2];
175@else {
176    let view_pos = view_bindings::view.view_from_clip * vec4(0.0, 0.0, ndc_depth, 1.0);
177    return view_pos.z / view_pos.w;
178}
179}
180
181/// Convert linear view z to ndc depth. 
182/// Note: View z input should be negative for values in front of the camera as -z is forward
183fn view_z_to_depth_ndc(view_z: f32) -> f32 {
184@if(VIEW_PROJECTION_PERSPECTIVE)
185    return -perspective_camera_near() / view_z;
186@elif(VIEW_PROJECTION_ORTHOGRAPHIC)
187    return view_bindings::view.clip_from_view[3][2] + view_z * view_bindings::view.clip_from_view[2][2];
188@else {
189    let ndc_pos = view_bindings::view.clip_from_view * vec4(0.0, 0.0, view_z, 1.0);
190    return ndc_pos.z / ndc_pos.w;
191}
192}
193
194fn prev_view_z_to_depth_ndc(view_z: f32) -> f32 {
195@if(VIEW_PROJECTION_PERSPECTIVE)
196    return -perspective_camera_near() / view_z;
197@elif(VIEW_PROJECTION_ORTHOGRAPHIC)
198    return prepass_bindings::previous_view_uniforms.clip_from_view[3][2] +
199        view_z * prepass_bindings::previous_view_uniforms.clip_from_view[2][2];
200@else {
201    let ndc_pos = prepass_bindings::previous_view_uniforms.clip_from_view *
202        vec4(0.0, 0.0, view_z, 1.0);
203    return ndc_pos.z / ndc_pos.w;
204}

211/// Convert ndc space xy coordinate [-1.0 .. 1.0] to uv [0.0 .. 1.0]
212fn ndc_to_uv(ndc: vec2<f32>) -> vec2<f32> {
213    return ndc * vec2(0.5, -0.5) + vec2(0.5);
214}
215
216/// Convert uv [0.0 .. 1.0] coordinate to ndc space xy [-1.0 .. 1.0]
217fn uv_to_ndc(uv: vec2<f32>) -> vec2<f32> {
218    return uv * vec2(2.0, -2.0) + vec2(-1.0, 1.0);
219}
220
221/// returns the (0.0, 0.0) .. (1.0, 1.0) position within the viewport for the current render target
222/// [0 .. render target viewport size] eg. [(0.0, 0.0) .. (1280.0, 720.0)] to [(0.0, 0.0) .. (1.0, 1.0)]
223fn frag_coord_to_uv(frag_coord: vec2<f32>) -> vec2<f32> {
224    return (frag_coord - view_bindings::view.viewport.xy) / view_bindings::view.viewport.zw;
225}
226
227/// Convert frag coord to ndc
228fn frag_coord_to_ndc(frag_coord: vec4<f32>) -> vec3<f32> {
229    return vec3(uv_to_ndc(frag_coord_to_uv(frag_coord.xy)), frag_coord.z);
230}
231
232/// Convert ndc space xy coordinate [-1.0 .. 1.0] to [0 .. render target
233/// viewport size]
234fn ndc_to_frag_coord(ndc: vec2<f32>) -> vec2<f32> {
235    return ndc_to_uv(ndc) * view_bindings::view.viewport.zw;
