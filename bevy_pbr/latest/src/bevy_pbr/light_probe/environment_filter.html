20// Sample an environment map with a specific LOD
21fn sample_environment(dir: vec3f, level: f32) -> vec4f {
22    let cube_uv = dir_to_cube_uv(dir);
23    return textureSampleLevel(input_texture, input_sampler, cube_uv.uv, cube_uv.face, level);
24}

26// Blue noise randomization
27@if(HAS_BLUE_NOISE)
28fn sample_noise(pixel_coords: vec2u) -> vec4f {
29    let noise_size = vec2u(1) << constants.noise_size_bits;
30    let noise_size_mask = noise_size - vec2u(1u);
31    let noise_coords = pixel_coords & noise_size_mask;
32    let uv = vec2f(noise_coords) / vec2f(noise_size);
33    return textureSampleLevel(blue_noise_texture, input_sampler, uv, 0u, 0.0);
34}
35@else
36// pseudo-random numbers using RNG
37fn sample_noise(pixel_coords: vec2u) -> vec4f {
38    var rng_state: u32 = (pixel_coords.x * 3966231743u) ^ (pixel_coords.y * 3928936651u);
39    let rnd = rand_vec2f(&rng_state);
40    return vec4f(rnd, 0.0, 0.0);
41}

43// Calculate LOD for environment map lookup using filtered importance sampling
44fn calculate_environment_map_lod(pdf: f32, width: f32, samples: f32) -> f32 {
45    // Solid angle of current sample
46    let omega_s = 1.0 / (samples * pdf);
47    
48    // Solid angle of a texel in the environment map
49    let omega_p = 4.0 * PI / (6.0 * width * width);
50    
51    // Filtered importance sampling: compute the correct LOD
52    return 0.5 * log2(omega_s / omega_p);
53}

55@compute
56@workgroup_size(8, 8, 1)
57fn generate_radiance_map(@builtin(global_invocation_id) global_id: vec3u) {
58    let size = textureDimensions(output_texture).xy;
59    let invSize = 1.0 / vec2f(size);
60    
61    let coords = vec2u(global_id.xy);
62    let face = global_id.z;
63    
64    if (any(coords >= size)) {
65        return;
66    }
67    
68    // Convert texture coordinates to direction vector
69    let uv = (vec2f(coords) + 0.5) * invSize;
70    let normal = sample_cube_dir(uv, face);
71    
72    // For radiance map, view direction = normal for perfect reflection
73    let view = normal;
74    
75    // Convert perceptual roughness to physical microfacet roughness
76    let perceptual_roughness = constants.roughness;
77    let roughness = lighting::perceptualRoughnessToRoughness(perceptual_roughness);
78    
79    // Get blue noise offset for stratification
80    let vector_noise = sample_noise(coords);
81    
82    var radiance = vec3f(0.0);
83    var total_weight = 0.0;
84    
85    // Skip sampling for mirror reflection (roughness = 0)
86    if (roughness < 0.01) {
87        radiance = sample_environment(normal, 0.0).rgb;
88        textureStore(output_texture, coords, face, vec4f(radiance, 1.0));
89        return;
90    }
91    
92    // For higher roughness values, use importance sampling
93    let sample_count = constants.sample_count;
94    
95    for (var i = 0u; i < sample_count; i++) {
96        // Get sample coordinates from Hammersley sequence with blue noise offset
97        var xi = hammersley_2d(i, sample_count);
98        xi = fract(xi + vector_noise.rg); // Apply Cranley-Patterson rotation
99        
100        // Sample the GGX distribution with the spherical-cap VNDF method
101        let light_dir = lighting::sample_visible_ggx(xi, roughness, normal, view);
102        
103        // Calculate weight (N·L)
104        let NdotL = dot(normal, light_dir);
105        
106        if (NdotL > 0.0) {
107            // Reconstruct the microfacet half-vector from view and light and compute PDF terms
108            let half_vector = normalize(view + light_dir);
109            let NdotH = dot(normal, half_vector);
110            let NdotV = dot(normal, view);
111            
112            // Get the geometric shadowing term
113            let G = lighting::G_Smith(NdotV, NdotL, roughness);
114            
115            // PDF that matches the bounded-VNDF sampling
116            let pdf = lighting::ggx_vndf_pdf(view, NdotH, roughness);
117            
118            // Calculate LOD using filtered importance sampling
119            // This is crucial to avoid fireflies and improve quality
120            let width = f32(size.x);
121            let lod = calculate_environment_map_lod(pdf, width, f32(sample_count));
122            
123            // Get source mip level - ensure we don't go negative
124            let source_mip = max(0.0, lod);
125            
126            // Sample environment map with the light direction
127            var sample_color = sample_environment(light_dir, source_mip).rgb;
128            
129            // Accumulate weighted sample, including geometric term
130            radiance += sample_color * NdotL * G;
131            total_weight += NdotL * G;
132        }
133    }
134    
135    // Normalize by total weight
136    if (total_weight > 0.0) {
137        radiance = radiance / total_weight;
138    }
139    
140    // Write result to output texture
141    textureStore(output_texture, coords, face, vec4f(radiance, 1.0));
142}

144@compute
145@workgroup_size(8, 8, 1)
146fn generate_irradiance_map(@builtin(global_invocation_id) global_id: vec3u) {
147    let size = textureDimensions(output_texture).xy;
148    let invSize = 1.0 / vec2f(size);
149    
150    let coords = vec2u(global_id.xy);
151    let face = global_id.z;
152    
153    if (any(coords >= size)) {
154        return;
155    }
156    
157    // Convert texture coordinates to direction vector
158    let uv = (vec2f(coords) + 0.5) * invSize;
159    let normal = sample_cube_dir(uv, face);
160    
161    var irradiance = vec3f(0.0);
162    
163    // Use uniform sampling on a hemisphere
164    for (var i = 0u; i < constants.sample_count; i++) {
165        // Build a deterministic RNG seed for this pixel / sample
166        // 4 randomly chosen 32-bit primes
167        var rng: u32 = (coords.x * 2131358057u) ^ (coords.y * 3416869721u) ^ (face * 1199786941u) ^ (i * 566200673u);
168
169        // Sample a direction from the upper hemisphere around the normal
170        var sample_dir = sample_cosine_hemisphere(normal, &rng);
171
172        // Sample environment with level 0 (no mip)
173        var sample_color = sample_environment(sample_dir, 0.0).rgb;
174        
175        // Accumulate the contribution
176        irradiance += sample_color;
177    }
178
179    // Normalize by number of samples (cosine-weighted sampling already accounts for PDF)
180    irradiance = irradiance / f32(constants.sample_count);
181    
182    // Write result to output texture
183    textureStore(output_texture, coords, face, vec4f(irradiance, 1.0));
184}
