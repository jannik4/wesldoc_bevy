6// Fog formulas adapted from:
7// https://learn.microsoft.com/en-us/windows/win32/direct3d9/fog-formulas
8// https://catlikecoding.com/unity/tutorials/rendering/part-14/
9// https://iquilezles.org/articles/fog/ (Atmospheric Fog and Scattering)
10
11fn scattering_adjusted_fog_color(
12    fog_params: Fog,
13    scattering: vec3<f32>,
14) -> vec4<f32> {
15    if (fog_params.directional_light_color.a > 0.0) {
16        return vec4<f32>(
17            fog_params.base_color.rgb
18                + scattering * fog_params.directional_light_color.rgb * fog_params.directional_light_color.a,
19            fog_params.base_color.a,
20        );
21    } else {
22        return fog_params.base_color;
23    }
24}
25
26fn linear_fog(
27    fog_params: Fog,
28    input_color: vec4<f32>,
29    distance: f32,
30    scattering: vec3<f32>,
31) -> vec4<f32> {
32    var fog_color = scattering_adjusted_fog_color(fog_params, scattering);
33    let start = fog_params.be.x;
34    let end = fog_params.be.y;
35    fog_color.a *= 1.0 - clamp((end - distance) / (end - start), 0.0, 1.0);
36    return vec4<f32>(mix(input_color.rgb, fog_color.rgb, fog_color.a), input_color.a);
37}
38
39fn exponential_fog(
40    fog_params: Fog,
41    input_color: vec4<f32>,
42    distance: f32,
43    scattering: vec3<f32>,
44) -> vec4<f32> {
45    var fog_color = scattering_adjusted_fog_color(fog_params, scattering);
46    let density = fog_params.be.x;
47    fog_color.a *= 1.0 - 1.0 / exp(distance * density);
48    return vec4<f32>(mix(input_color.rgb, fog_color.rgb, fog_color.a), input_color.a);
49}
50
51fn exponential_squared_fog(
52    fog_params: Fog,
53    input_color: vec4<f32>,
54    distance: f32,
55    scattering: vec3<f32>,
56) -> vec4<f32> {
57    var fog_color = scattering_adjusted_fog_color(fog_params, scattering);
58    let distance_times_density = distance * fog_params.be.x;
59    fog_color.a *= 1.0 - 1.0 / exp(distance_times_density * distance_times_density);
60    return vec4<f32>(mix(input_color.rgb, fog_color.rgb, fog_color.a), input_color.a);
61}
62
63fn atmospheric_fog(
64    fog_params: Fog,
65    input_color: vec4<f32>,
66    distance: f32,
67    scattering: vec3<f32>,
68) -> vec4<f32> {
69    var fog_color = scattering_adjusted_fog_color(fog_params, scattering);
70    let extinction_factor = 1.0 - 1.0 / exp(distance * fog_params.be);
71    let inscattering_factor = 1.0 - 1.0 / exp(distance * fog_params.bi);
72    return vec4<f32>(
73        input_color.rgb * (1.0 - extinction_factor * fog_color.a)
74            + fog_color.rgb * inscattering_factor * fog_color.a,
75        input_color.a
76    );
77}
