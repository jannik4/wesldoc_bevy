67    ambient_color: vec4<f32>,
68    // x/y/z dimensions and n_clusters in w
69    cluster_dimensions: vec4<u32>,
70    // xy are vec2<f32>(cluster_dimensions.xy) / vec2<f32>(view.width, view.height)
71    //
72    // For perspective projections:
73    // z is (cluster_dimensions.z - 1) / log(far / near)
74    // w is (cluster_dimensions.z - 1) * log(near) / log(far / near)
75    //
76    // For orthographic projections:
77    // NOTE: near and far are +ve but -z is infront of the camera
78    // z is -near
79    // w is cluster_dimensions.z / (-far - -near)
80    cluster_factors: vec4<f32>,

82    spot_light_shadowmap_offset: i32,
