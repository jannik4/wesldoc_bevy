30fn layer_count(descriptor_index: u32) -> u32 {
31@if(SKINS_USE_UNIFORM_BUFFERS)

33@if(SKINS_USE_UNIFORM_BUFFERS)

39@if(MORPH_TARGETS && SKINS_USE_UNIFORM_BUFFERS)
40fn component_texture_coord(vertex_index: u32, component_offset: u32) -> vec2<u32> {
41    let width = u32(textureDimensions(morph_targets).x);
42    let component_index = total_component_count * vertex_index + component_offset;
43    return vec2<u32>(component_index % width, component_index / width);
44}

46@if(MORPH_TARGETS)
47fn weight_at(weight_index: u32, descriptor_index: u32) -> f32 {
48@if(SKINS_USE_UNIFORM_BUFFERS)
49    let i = weight_index;
50@if(SKINS_USE_UNIFORM_BUFFERS)
51    return morph_weights.weights[i / 4u][i % 4u];
52@if(!SKINS_USE_UNIFORM_BUFFERS)
53    let weights_offset = morph_descriptors[descriptor_index].current_weights_offset;
54@if(!SKINS_USE_UNIFORM_BUFFERS)
55    return morph_weights[weights_offset + weight_index];
56}

70@if(MORPH_TARGETS && SKINS_USE_UNIFORM_BUFFERS)
71fn morph_pixel(vertex: u32, component: u32, weight: u32) -> f32 {
72    let coord = component_texture_coord(vertex, component);
73    // Due to https://gpuweb.github.io/gpuweb/wgsl/#texel-formats
74    // While the texture stores a f32, the textureLoad returns a vec4<>, where
75    // only the first component is set.
76    return textureLoad(morph_targets, vec3(coord, weight), 0).r;
77}

79@if(MORPH_TARGETS && SKINS_USE_UNIFORM_BUFFERS)
80fn morph(vertex_index: u32, component_offset: u32, weight_index: u32) -> vec3<f32> {
81    return vec3<f32>(
82        morph_pixel(vertex_index, component_offset, weight_index),
83        morph_pixel(vertex_index, component_offset + 1u, weight_index),
84        morph_pixel(vertex_index, component_offset + 2u, weight_index),
85    );
86}
