33// prepare a basic PbrInput from the vertex stage output, mesh binding and view binding
34fn pbr_input_from_vertex_output(
35    in: VertexOutput,
36    is_front: bool,
37    double_sided: bool,
38) -> pbr_types::PbrInput {
39    var pbr_input: pbr_types::PbrInput = pbr_types::pbr_input_new();
40
41@if(MESHLET_MESH_MATERIAL_PASS)
42    pbr_input.flags = in.mesh_flags;
43@else
44    pbr_input.flags = mesh[in.instance_index].flags;
45
46    pbr_input.is_orthographic = view.clip_from_view[3].w == 1.0;
47    pbr_input.V = pbr_functions::calculate_view(in.world_position, pbr_input.is_orthographic);
48    pbr_input.frag_coord = in.position;
49    pbr_input.world_position = in.world_position;
50
51@if(VERTEX_COLORS)
52    pbr_input.material.base_color = in.color;
53
54    let is_logically_front = pbr_functions::winding_corrected_front_facing(pbr_input.flags, is_front);
55    pbr_input.world_normal = pbr_functions::prepare_world_normal(
56        in.world_normal,
57        double_sided,
58        is_logically_front,
59    );
60
61@if(LOAD_PREPASS_NORMALS)
62    pbr_input.N = prepass_utils::prepass_normal(in.position, 0u);
63@else
64    pbr_input.N = normalize(pbr_input.world_normal);
65
66    return pbr_input;
67}

501            let coords = uv;
502            @if(MESHLET_MESH_MATERIAL_PASS)
503            diffuse_transmission *= textureSampleGrad(texture, sampler_, coords, bias.ddx_uv, bias.ddy_uv).a;
504            @else
505            diffuse_transmission *= textureSampleBias(texture, sampler_, coords, bias.mip_bias).a;
506        }
507        pbr_input.material.diffuse_transmission = diffuse_transmission;
508
509        var diffuse_occlusion: vec3<f32> = vec3(1.0);
510        var specular_occlusion: f32 = 1.0;
511@if(VERTEX_UVS)
512        if ((flags & pbr_types::STANDARD_MATERIAL_FLAGS_OCCLUSION_TEXTURE_BIT) != 0u) {
513            @if(BINDLESS)
514            let texture = bindless_textures_2d[material_indices[slot].occlusion_texture];
515            @else
516            let texture = pbr_bindings::occlusion_texture;
517            @if(BINDLESS)
518            let sampler_ = bindless_samplers_filtering[material_indices[slot].occlusion_sampler];
519            @else
520            let sampler_ = pbr_bindings::occlusion_sampler;
521            @if(STANDARD_MATERIAL_OCCLUSION_UV_B)
522            let coords = uv_b;
523            @else
524            let coords = uv;
525            @if(MESHLET_MESH_MATERIAL_PASS)
526            diffuse_occlusion *= textureSampleGrad(texture, sampler_, coords, bias.ddx_uv, bias.ddy_uv).r;
527            @else
528            diffuse_occlusion *= textureSampleBias(texture, sampler_, coords, bias.mip_bias).r;
529        }
530@if(SCREEN_SPACE_AMBIENT_OCCLUSION) {
531        let ssao_tex_pos = vec2<i32>(in.position.xy - view.viewport.xy);
532        let ssao = textureLoad(screen_space_ambient_occlusion_texture, ssao_tex_pos, 0i).r;
533        let ssao_multibounce = ssao_multibounce(ssao, pbr_input.material.base_color.rgb);
534        diffuse_occlusion = min(diffuse_occlusion, ssao_multibounce);
535        // Use SSAO to estimate the specular occlusion.
536        // Lagarde and Rousiers 2014, "Moving Frostbite to Physically Based Rendering"
537        let roughness = lighting::perceptualRoughnessToRoughness(pbr_input.material.perceptual_roughness);
538        specular_occlusion = saturate(pow(NdotV + ssao, exp2(-16.0 * roughness - 1.0)) - 1.0 + ssao);
539}
540        pbr_input.diffuse_occlusion = diffuse_occlusion;
541        pbr_input.specular_occlusion = specular_occlusion;
542
543        // N (normal vector)
544@if(!LOAD_PREPASS_NORMALS) {
545
546        pbr_input.N = normalize(pbr_input.world_normal);
547        pbr_input.clearcoat_N = pbr_input.N;
548
549@if(VERTEX_UVS && VERTEX_TANGENTS)
550        let TBN = pbr_functions::calculate_tbn_mikktspace(pbr_input.world_normal, in.world_tangent);
551
552@if(VERTEX_UVS && VERTEX_TANGENTS && STANDARD_MATERIAL_NORMAL_MAP) {
553
554        @if(BINDLESS)
555        let texture = bindless_textures_2d[material_indices[slot].normal_map_texture];
556        @else
557        let texture = pbr_bindings::normal_map_texture;
558        @if(BINDLESS)
559        let sampler_ = bindless_samplers_filtering[material_indices[slot].normal_map_sampler];
560        @else
561        let sampler_ = pbr_bindings::normal_map_sampler;
562        @if(STANDARD_MATERIAL_NORMAL_MAP_UV_B)
563        let coords = uv_b;
564        @else
565        let coords = uv;
566        @if(MESHLET_MESH_MATERIAL_PASS)
567        let Nt = textureSampleGrad(texture, sampler_, coords, bias.ddx_uv, bias.ddy_uv).rgb;
568        @else
569        let Nt = textureSampleBias(texture, sampler_, coords, bias.mip_bias).rgb;
570
571        pbr_input.N = pbr_functions::apply_normal_mapping(
572            flags,
573            TBN,
574            double_sided,
575            is_logically_front,
576            Nt,
577        );
578
579}
580
581@if(VERTEX_UVS && VERTEX_TANGENTS && STANDARD_MATERIAL_CLEARCOAT) {
582
583        // Note: `KHR_materials_clearcoat` specifies that, if there's no
584        // clearcoat normal map, we must set the normal to the mesh's normal,
585        // and not to the main layer's bumped normal.
586
587@if(STANDARD_MATERIAL_CLEARCOAT_NORMAL_MAP) {
588
589        @if(BINDLESS)
590        let clearcoat_texture = bindless_textures_2d[material_indices[slot].clearcoat_normal_texture];
591        @else
592        let clearcoat_texture = pbr_bindings::clearcoat_normal_texture;
593        @if(BINDLESS)
594        let clearcoat_sampler = bindless_samplers_filtering[material_indices[slot].clearcoat_normal_sampler];
595        @else
596        let clearcoat_sampler = pbr_bindings::clearcoat_normal_sampler;
597        @if(STANDARD_MATERIAL_CLEARCOAT_NORMAL_UV_B)
598        let clearcoat_coords = uv_b;
599        @else
600        let clearcoat_coords = uv;
601        @if(MESHLET_MESH_MATERIAL_PASS)
602        let clearcoat_Nt = textureSampleGrad(clearcoat_texture, clearcoat_sampler, clearcoat_coords, bias.ddx_uv, bias.ddy_uv).rgb;
603        @else
604        let clearcoat_Nt = textureSampleBias(clearcoat_texture, clearcoat_sampler, clearcoat_coords, bias.mip_bias).rgb;
605
606        pbr_input.clearcoat_N = pbr_functions::apply_normal_mapping(
607            flags,
608            TBN,
609            double_sided,
610            is_logically_front,
611            clearcoat_Nt,
612        );
613
614}
615
616}
617
618        // Take anisotropy into account.
619        //
620        // This code comes from the `KHR_materials_anisotropy` spec:
621        // <https://github.com/KhronosGroup/glTF/blob/main/extensions/2.0/Khronos/KHR_materials_anisotropy/README.md#individual-lights>
622@if(PBR_ANISOTROPY_TEXTURE_SUPPORTED && VERTEX_TANGENTS && STANDARD_MATERIAL_ANISOTROPY) {
623
624@if(BINDLESS)
625        var anisotropy_strength =
626                pbr_bindings::material_array[material_indices[slot].material].anisotropy_strength;
627@else
628        var anisotropy_strength = pbr_bindings::material.anisotropy_strength;
629@if(BINDLESS)
630        var anisotropy_direction =
631                pbr_bindings::material_array[material_indices[slot].material].anisotropy_rotation;
632@else
633        var anisotropy_direction = pbr_bindings::material.anisotropy_rotation;
634
635        // Adjust based on the anisotropy map if there is one.
636        if ((flags & pbr_types::STANDARD_MATERIAL_FLAGS_ANISOTROPY_TEXTURE_BIT) != 0u) {
637            @if(BINDLESS)
638            let texture = bindless_textures_2d[material_indices[slot].anisotropy_texture];
639            @else
640            let texture = pbr_bindings::anisotropy_texture;
641            @if(BINDLESS)
642            let sampler_ = bindless_samplers_filtering[material_indices[slot].anisotropy_sampler];
643            @else
644            let sampler_ = pbr_bindings::anisotropy_sampler;
645            @if(STANDARD_MATERIAL_ANISOTROPY_UV_B)
646            let coords = uv_b;
647            @else
648            let coords = uv;
649            @if(MESHLET_MESH_MATERIAL_PASS)
650            let anisotropy_texel = textureSampleGrad(texture, sampler_, coords, bias.ddx_uv, bias.ddy_uv).rgb;
651            @else
652            let anisotropy_texel = textureSampleBias(texture, sampler_, coords, bias.mip_bias).rgb;
653
654            let anisotropy_direction_from_texture = normalize(anisotropy_texel.rg * 2.0 - 1.0);
655            // Rotate by the anisotropy direction.
656            anisotropy_direction =
657                mat2x2(anisotropy_direction.xy, anisotropy_direction.yx * vec2(-1.0, 1.0)) *
658                anisotropy_direction_from_texture;
659            anisotropy_strength *= anisotropy_texel.b;
660        }
661
662        pbr_input.anisotropy_strength = anisotropy_strength;
663
664        let anisotropy_T = normalize(TBN * vec3(anisotropy_direction, 0.0));
665        let anisotropy_B = normalize(cross(pbr_input.world_normal, anisotropy_T));
666        pbr_input.anisotropy_T = anisotropy_T;
667        pbr_input.anisotropy_B = anisotropy_B;
668
669}
670
671}
672
673// TODO: Meshlet support
674@if(LIGHTMAP) {
675
676@if(BINDLESS)
677        let lightmap_exposure =
678                pbr_bindings::material_array[material_indices[slot].material].lightmap_exposure;
679@else
680        let lightmap_exposure = pbr_bindings::material.lightmap_exposure;
681
682        pbr_input.lightmap_light = lightmap(in.uv_b, lightmap_exposure, in.instance_index);
683}
684    }
685
686    return pbr_input;
