12// From the Filament design doc
13// https://google.github.io/filament/Filament.md.html#table_symbols
14// Symbol Definition
15// v    View unit vector
16// l    Incident light unit vector
17// n    Surface normal unit vector
18// h    Half unit vector between l and v
19// f    BRDF
20// f_d    Diffuse component of a BRDF
21// f_r    Specular component of a BRDF
22// α    Roughness, remapped from using input perceptualRoughness
23// σ    Diffuse reflectance
24// Ω    Spherical domain
25// f0    Reflectance at normal incidence
26// f90    Reflectance at grazing angle
27// χ+(a)    Heaviside function (1 if a>0 and 0 otherwise)
28// nior    Index of refraction (IOR) of an interface
29// ⟨n⋅l⟩    Dot product clamped to [0..1]
30// ⟨a⟩    Saturated value (clamped to [0..1])
31
32// The Bidirectional Reflectance Distribution Function (BRDF) describes the surface response of a standard material
33// and consists of two components, the diffuse component (f_d) and the specular component (f_r):
34// f(v,l) = f_d(v,l) + f_r(v,l)
35//
36// The form of the microfacet model is the same for diffuse and specular
37// f_r(v,l) = f_d(v,l) = 1 / { |n⋅v||n⋅l| } ∫_Ω D(m,α) G(v,l,m) f_m(v,l,m) (v⋅m) (l⋅m) dm
38//
39// In which:
40// D, also called the Normal Distribution Function (NDF) models the distribution of the microfacets
41// G models the visibility (or occlusion or shadow-masking) of the microfacets
42// f_m is the microfacet BRDF and differs between specular and diffuse components
43//
44// The above integration needs to be approximated.
45

123// distanceAttenuation is simply the square falloff of light intensity
124// combined with a smooth attenuation at the edge of the light radius
125//
126// light radius is a non-physical construct for efficiency purposes,
127// because otherwise every light affects every fragment in the scene
128fn getDistanceAttenuation(distanceSquare: f32, inverseRangeSquared: f32) -> f32 {
129    return getRangeFalloff(distanceSquare, inverseRangeSquared) * 1.0 / max(distanceSquare, 0.0001);
130}
131
132// Falloff without the distance attenuation, for lights that have it baked-in (eg LTC lights)
133fn getRangeFalloff(distanceSquare: f32, inverseRangeSquared: f32) -> f32 {
134    let factor = distanceSquare * inverseRangeSquared;
135    let smoothFactor = saturate(1.0 - factor * factor);
136    return smoothFactor * smoothFactor;
137}

139// Normal distribution function (specular D)
140// Based on https://google.github.io/filament/Filament.md.html#citation-walter07
141
142// D_GGX(h,α) = α^2 / { π ((n⋅h)^2 (α2−1) + 1)^2 }
143
144// Simple implementation, has precision problems when using fp16 instead of fp32
145// see https://google.github.io/filament/Filament.md.html#listing_speculardfp16
146fn D_GGX(roughness: f32, NdotH: f32) -> f32 {
147    let oneMinusNdotHSquared = 1.0 - NdotH * NdotH;
148    let a = NdotH * roughness;
149    let k = roughness / (oneMinusNdotHSquared + a * a);
150    let d = k * k * (1.0 / PI);
151    return d;
152}

154// An approximation of the anisotropic GGX distribution function.
155//
156//                                     1
157//     D(𝐡) = ───────────────────────────────────────────────────
158//            παₜα_b((𝐡 ⋅ 𝐭)² / αₜ²) + (𝐡 ⋅ 𝐛)² / α_b² + (𝐡 ⋅ 𝐧)²)²
159//
160// * `T` = 𝐭 = the tangent direction = the direction of increased roughness.
161//
162// * `B` = 𝐛 = the bitangent direction = the direction of decreased roughness.
163//
164// * `at` = αₜ = the alpha-roughness in the tangent direction.
165//
166// * `ab` = α_b = the alpha-roughness in the bitangent direction.
167//
168// This is from the `KHR_materials_anisotropy` spec:
169// <https://github.com/KhronosGroup/glTF/blob/main/extensions/2.0/Khronos/KHR_materials_anisotropy/README.md#individual-lights>
170fn D_GGX_anisotropic(at: f32, ab: f32, NdotH: f32, TdotH: f32, BdotH: f32) -> f32 {
171    let a2 = at * ab;
172    let f = vec3(ab * TdotH, at * BdotH, a2 * NdotH);
173    let w2 = a2 / dot(f, f);
174    let d = a2 * w2 * w2 * (1.0 / PI);
175    return d;
176}

178// Visibility function (Specular G)
179// V(v,l,a) = G(v,l,α) / { 4 (n⋅v) (n⋅l) }
180// such that f_r becomes
181// f_r(v,l) = D(h,α) V(v,l,α) F(v,h,f0)
182// where
183// V(v,l,α) = 0.5 / { n⋅l sqrt((n⋅v)^2 (1−α2) + α2) + n⋅v sqrt((n⋅l)^2 (1−α2) + α2) }
184// Note the two sqrt's, that may be slow on mobile, see https://google.github.io/filament/Filament.md.html#listing_approximatedspecularv
185fn V_SmithGGXCorrelated(roughness: f32, NdotV: f32, NdotL: f32) -> f32 {
186    let a2 = roughness * roughness;
187    let lambdaV = NdotL * sqrt((NdotV - a2 * NdotV) * NdotV + a2);
188    let lambdaL = NdotV * sqrt((NdotL - a2 * NdotL) * NdotL + a2);
189    let v = 0.5 / (lambdaV + lambdaL);
190    return v;
191}
192
193// The visibility function, anisotropic variant.
194fn V_GGX_anisotropic(
195    at: f32,
196    ab: f32,
197    NdotL: f32,
198    NdotV: f32,
199    BdotV: f32,
200    TdotV: f32,
201    TdotL: f32,
202    BdotL: f32,
203) -> f32 {
204    let GGX_V = NdotL * length(vec3(at * TdotV, ab * BdotV, NdotV));
205    let GGX_L = NdotV * length(vec3(at * TdotL, ab * BdotL, NdotL));
206    let v = 0.5 / (GGX_V + GGX_L);
207    return saturate(v);
208}

210// Probability-density function that matches the bounded VNDF sampler
211// https://gpuopen.com/download/Bounded_VNDF_Sampling_for_Smith-GGX_Reflections.pdf (Listing 2)
212fn ggx_vndf_pdf(i: vec3<f32>, NdotH: f32, roughness: f32) -> f32 {
213    let ndf = D_GGX(roughness, NdotH);
214
215    // Common terms
216    let ai = roughness * i.xy;
217    let len2 = dot(ai, ai);
218    let t = sqrt(len2 + i.z * i.z);
219    if i.z >= 0.0 {
220        let a = roughness;
221        let s = 1.0 + length(i.xy);
222        let a2 = a * a;
223        let s2 = s * s;
224        let k = (1.0 - a2) * s2 / (s2 + a2 * i.z * i.z);
225        return ndf / (2.0 * (k * i.z + t));
226    }
227
228    // Backfacing case
229    return ndf * (t - i.z) / (2.0 * len2);
230}

232// https://gpuopen.com/download/Bounded_VNDF_Sampling_for_Smith-GGX_Reflections.pdf (Listing 1)
233fn sample_visible_ggx(
234    xi: vec2<f32>,
235    roughness: f32,
236    normal: vec3<f32>,
237    view: vec3<f32>,
238) -> vec3<f32> {
239    let n = normal;
240    let alpha = roughness;
241
242    // Decompose view into components parallel/perpendicular to the normal
243    let wi_n = dot(view, n);
244    let wi_z = -n * wi_n;
245    let wi_xy = view + wi_z;
246
247    // Warp view vector to the unit-roughness configuration
248    let wi_std = -normalize(alpha * wi_xy + wi_z);
249
250    // Compute wi_std.z once for reuse
251    let wi_std_z = dot(wi_std, n);
252
253    // Bounded VNDF sampling
254    // Compute the bound parameter k (Eq. 5) and the scaled z–limit b (Eq. 6)
255    let s = 1.0 + length(wi_xy);
256    let a = clamp(alpha, 0.0, 1.0);
257    let a2 = a * a;
258    let s2 = s * s;
259    let k = (1.0 - a2) * s2 / (s2 + a2 * wi_n * wi_n);
260    let b = select(wi_std_z, k * wi_std_z, wi_n > 0.0);
261
262    // Sample a spherical cap in (-b, 1]
263    let z = 1.0 - xi.y * (1.0 + b);
264    let sin_theta = sqrt(max(0.0, 1.0 - z * z));
265    let phi = 2.0 * PI * xi.x - PI;
266    let x = sin_theta * cos(phi);
267    let y = sin_theta * sin(phi);
268    let c_std = vec3f(x, y, z);
269
270    // Rotate the sample so that the normal aligns with +Z
271    let TBN = orthonormalize(n);
272    let c = TBN * c_std;
273
274    // Half-vector in the standard frame
275    let wm_std = c + wi_std;
276    let wm_std_z = n * dot(n, wm_std);
277    let wm_std_xy = wm_std_z - wm_std;
278
279    // Unwarp back to original roughness and compute microfacet normal
280    let H = normalize(alpha * wm_std_xy + wm_std_z);
281
282    // Reflect view to obtain the outgoing (light) direction
283    return reflect(-view, H);
284}

286// Smith geometric shadowing function
287fn G_Smith(NdotV: f32, NdotL: f32, roughness: f32) -> f32 {
288    let k = roughness / 2.0;
289    let GGXL = NdotL / (NdotL * (1.0 - k) + k);
290    let GGXV = NdotV / (NdotV * (1.0 - k) + k);
291    return GGXL * GGXV;
292}

294// A simpler, but nonphysical, alternative to Smith-GGX. We use this for
295// clearcoat, per the Filament spec.
296//
297// https://google.github.io/filament/Filament.md.html#materialsystem/clearcoatmodel
298fn V_Kelemen(LdotH: f32) -> f32 {
299    let l_dot_h = max(LdotH, 0.0001);
300    return 0.25 / (l_dot_h * l_dot_h);
301}

303// Fresnel function
304// see https://google.github.io/filament/Filament.md.html#citation-schlick94
305// F_Schlick(v,h,f_0,f_90) = f_0 + (f_90 − f_0) (1 − v⋅h)^5
306fn F_Schlick_vec(f0: vec3<f32>, f90: f32, VdotH: f32) -> vec3<f32> {
307    // not using mix to keep the vec3 and float versions identical
308    return f0 + (f90 - f0) * pow(1.0 - VdotH, 5.0);
309}
310
311fn F_Schlick(f0: f32, f90: f32, VdotH: f32) -> f32 {
312    // not using mix to keep the vec3 and float versions identical
313    return f0 + (f90 - f0) * pow(1.0 - VdotH, 5.0);
314}
315
316fn fresnel(f0: vec3<f32>, LdotH: f32) -> vec3<f32> {
317    // f_90 suitable for ambient occlusion
318    // see https://google.github.io/filament/Filament.md.html#lighting/occlusion
319    let f90 = saturate(dot(f0, vec3<f32>(50.0 * 0.33)));
320    return F_Schlick_vec(f0, f90, LdotH);
321}

323// Given distribution, visibility, and Fresnel term, calculates the final
324// specular light.
325//
326// Multiscattering approximation:
327// <https://google.github.io/filament/Filament.md.html#listing_energycompensationimpl>
328fn specular_multiscatter(
329    D: f32,
330    V: f32,
331    F: vec3<f32>,
332    F0: vec3<f32>,
333    F_ab: vec2<f32>,
334    specular_intensity: f32,
335) -> vec3<f32> {
336    var Fr = (specular_intensity * D * V) * F;
337    // F_ab.x + F_ab.y is dfg.y in Filament
338    // See section 9.5 and listing 29 in the Filament spec
339    Fr *= 1.0 + F0 * (1.0 / (F_ab.x + F_ab.y) - 1.0);
340    return Fr;
341}

346// N, V, and L must all be normalized.
347fn derive_lighting_input(N: vec3<f32>, V: vec3<f32>, L: vec3<f32>) -> DerivedLightingInput {
348    var input: DerivedLightingInput;
349    var H: vec3<f32> = normalize(L + V);
350    input.H = H;
351    input.NdotL = saturate(dot(N, L));
352    input.NdotH = saturate(dot(N, H));
353    input.LdotH = saturate(dot(L, H));
354    return input;
355}

357// Returns L in the `xyz` components and the modified roughness in the `w` component.
358fn compute_specular_layer_values_for_point_light(
359    input: ptr<function, LightingInput>,
360    layer: u32,
361    V: vec3<f32>,
362    light_to_frag: vec3<f32>,
363    light_radius: f32,
364    distance: f32,
365) -> vec4<f32> {
366    // Unpack.
367    let R = (*input).layers[layer].R;
368    let a = (*input).layers[layer].roughness;
369
370    // Representative Point Area Lights.
371    // see http://blog.selfshadow.com/publications/s2013-shading-course/karis/s2013_pbs_epic_notes_v2.pdf p14-16
372    var LtFdotR = dot(light_to_frag, R);
373
374    // HACK: the following line is an amendment to fix a discontinuity when a surface
375    // intersects the light sphere. See https://github.com/bevyengine/bevy/issues/13318
376    //
377    // This sentence in the reference is crux of the problem: "We approximate finding the point with the
378    // smallest angle to the reflection ray by finding the point with the smallest distance to the ray."
379    // This approximation turns out to be completely wrong for points inside or near the sphere.
380    // Clamping this dot product to be positive ensures `centerToRay` lies on ray and not behind it.
381    // Any non-zero epsilon works here, it just has to be positive to avoid a singularity at zero.
382    // However, this is still far from physically accurate. Deriving an exact solution would help,
383    // but really we should adopt a superior solution to area lighting, such as:
384    // Polygonal-Light Shading with Linearly Transformed Cosines by Eric Heitz et al.
385    LtFdotR = max(0.0001, LtFdotR);
386
387    let centerToRay = LtFdotR * R - light_to_frag;
388    let closestPoint = light_to_frag + centerToRay * saturate(
389        light_radius * inverseSqrt(dot(centerToRay, centerToRay)));
390    let LspecLengthInverse = inverseSqrt(dot(closestPoint, closestPoint));
391
392    // Karis 2013, page 14. The constant 2 (or 3) is hand tuned to fit reference.
393    // https://cdn2.unrealengine.com/Resources/files/2013SiggraphPresentationsNotes-26915738.pdf
394    let a_prime = saturate(a + light_radius / (2.0 * distance));
395
396    let L: vec3<f32> = closestPoint * LspecLengthInverse; // normalize() equivalent?
397    return vec4(L, a_prime);
398}

400// Cook-Torrance approximation of the microfacet model integration using Fresnel law F to model f_m
401// f_r(v,l) = { D(h,α) G(v,l,α) F(v,h,f0) } / { 4 (n⋅v) (n⋅l) }
402fn specular(
403    input: ptr<function, LightingInput>,
404    derived_input: ptr<function, DerivedLightingInput>,
405    roughness: f32,
406    specular_intensity: f32,
407) -> vec3<f32> {
408    // Unpack.
409    let NdotV = (*input).layers[LAYER_BASE].NdotV;
410    let F0 = mix((*input).F0_dielectric, (*input).F0_metallic, (*input).metallic);
411    let NdotL = (*derived_input).NdotL;
412    let NdotH = (*derived_input).NdotH;
413    let LdotH = (*derived_input).LdotH;
414
415    // Calculate distribution.
416    let D = D_GGX(roughness, NdotH);
417    // Calculate visibility.
418    let V = V_SmithGGXCorrelated(roughness, NdotV, NdotL);
419    // Calculate the Fresnel term.
420    let F = fresnel(F0, LdotH);
421
422    // Calculate the specular light.
423    let Fr = specular_multiscatter(D, V, F, F0, (*input).F_ab, specular_intensity);
424    return Fr;
425}

427// Calculates the specular light for the clearcoat layer. Returns Fc, the
428// Fresnel term, in the first channel, and Frc, the specular clearcoat light, in
429// the second channel.
430//
431// <https://google.github.io/filament/Filament.md.html#listing_clearcoatbrdf>
432fn specular_clearcoat(
433    input: ptr<function, LightingInput>,
434    derived_input: ptr<function, DerivedLightingInput>,
435    clearcoat_strength: f32,
436    roughness: f32,
437    specular_intensity: f32,
438) -> vec2<f32> {
439    // Unpack.
440    let NdotH = (*derived_input).NdotH;
441    let LdotH = (*derived_input).LdotH;
442
443    // Calculate distribution.
444    let Dc = D_GGX(roughness, NdotH);
445    // Calculate visibility.
446    let Vc = V_Kelemen(LdotH);
447    // Calculate the Fresnel term.
448    let Fc = F_Schlick(0.04, 1.0, LdotH) * clearcoat_strength;
449    // Calculate the specular light.
450    let Frc = (specular_intensity * Dc * Vc) * Fc;
451    return vec2(Fc, Frc);
452}

454@if(STANDARD_MATERIAL_ANISOTROPY)
455fn specular_anisotropy(
456    input: ptr<function, LightingInput>,
457    derived_input: ptr<function, DerivedLightingInput>,
458    L: vec3<f32>,
459    roughness: f32,
460    specular_intensity: f32,
461) -> vec3<f32> {
462    // Unpack.
463    let NdotV = (*input).layers[LAYER_BASE].NdotV;
464    let V = (*input).V;
465    let F0 = mix((*input).F0_dielectric, (*input).F0_metallic, (*input).metallic);
466    let anisotropy = (*input).anisotropy;
467    let Ta = (*input).Ta;
468    let Ba = (*input).Ba;
469    let H = (*derived_input).H;
470    let NdotL = (*derived_input).NdotL;
471    let NdotH = (*derived_input).NdotH;
472    let LdotH = (*derived_input).LdotH;
473
474    let TdotL = dot(Ta, L);
475    let BdotL = dot(Ba, L);
476    let TdotH = dot(Ta, H);
477    let BdotH = dot(Ba, H);
478    let TdotV = dot(Ta, V);
479    let BdotV = dot(Ba, V);
480
481    let ab = roughness * roughness;
482    let at = mix(ab, 1.0, anisotropy * anisotropy);
483
484    let Da = D_GGX_anisotropic(at, ab, NdotH, TdotH, BdotH);
485    let Va = V_GGX_anisotropic(at, ab, NdotL, NdotV, BdotV, TdotV, TdotL, BdotL);
486    let Fa = fresnel(F0, LdotH);
487
488    // Calculate the specular light.
489    let Fr = specular_multiscatter(Da, Va, Fa, F0, (*input).F_ab, specular_intensity);
490    return Fr;
491}

493// Diffuse BRDF
494// https://google.github.io/filament/Filament.md.html#materialsystem/diffusebrdf
495// fd(v,l) = σ/π * 1 / { |n⋅v||n⋅l| } ∫Ω D(m,α) G(v,l,m) (v⋅m) (l⋅m) dm
496//
497// simplest approximation
498// float Fd_Lambert() {
499//     return 1.0 / PI;
500// }
501//
502// vec3 Fd = diffuseColor * Fd_Lambert();
503//
504// Disney approximation
505// See https://google.github.io/filament/Filament.md.html#citation-burley12
506// minimal quality difference
507fn Fd_Burley(
508    input: ptr<function, LightingInput>,
509    derived_input: ptr<function, DerivedLightingInput>,
510) -> f32 {
511    // Unpack.
512    let roughness = (*input).layers[LAYER_BASE].roughness;
513    let NdotV = (*input).layers[LAYER_BASE].NdotV;
514    let NdotL = (*derived_input).NdotL;
515    let LdotH = (*derived_input).LdotH;
516
517    let f90 = 0.5 + 2.0 * roughness * LdotH * LdotH;
518    let lightScatter = F_Schlick(1.0, f90, NdotL);
519    let viewScatter = F_Schlick(1.0, f90, NdotV);
520    return lightScatter * viewScatter * (1.0 / PI);
521}

523// Scale/bias approximation
524fn F_AB(perceptual_roughness: f32, NdotV: f32) -> vec2<f32> {
525@if(DFG_LUT)
526    return textureSampleLevel(view_bindings::dfg_lut, view_bindings::dfg_lut_sampler, vec2<f32>(NdotV, perceptual_roughness), 0.0).rg;
527@else {
528    // Polynomial approximation, see https://www.unrealengine.com/en-US/blog/physically-based-shading-on-mobile
529    let c0 = vec4<f32>(-1.0, -0.0275, -0.572, 0.022);
530    let c1 = vec4<f32>(1.0, 0.0425, 1.04, -0.04);
531    let r = perceptual_roughness * c0 + c1;
532    let a004 = min(r.x * r.x, exp2(-9.28 * NdotV)) * r.x + r.y;
533    // Keep F_ab positive to avoid divide-by-zero in downstream BRDF terms.
534    let f_ab_epsilon = 0.00005;
535    return max(vec2<f32>(-1.04, 1.04) * a004 + r.zw, vec2<f32>(f_ab_epsilon));
536}
537}
538
539fn EnvBRDFApprox(F0: vec3<f32>, F_ab: vec2<f32>) -> vec3<f32> {
540    return F0 * F_ab.x + F_ab.y;
541}

543fn perceptualRoughnessToRoughness(perceptualRoughness: f32) -> f32 {
544    // clamp perceptual roughness to prevent precision problems
545    // According to Filament design 0.089 is recommended for mobile
546    // Filament uses 0.045 for non-mobile
547    let clampedPerceptualRoughness = clamp(perceptualRoughness, 0.089, 1.0);
548    return clampedPerceptualRoughness * clampedPerceptualRoughness;
549}

551// this must align with CubemapLayout in decal/clustered.rs
552const CUBEMAP_TYPE_CROSS_VERTICAL: u32 = 0;
553const CUBEMAP_TYPE_CROSS_HORIZONTAL: u32 = 1;
554const CUBEMAP_TYPE_SEQUENCE_VERTICAL: u32 = 2;
555const CUBEMAP_TYPE_SEQUENCE_HORIZONTAL: u32 = 3;
556
557const X_PLUS: u32 = 0;
558const X_MINUS: u32 = 1;
559const Y_PLUS: u32 = 2;
560const Y_MINUS: u32 = 3;
561const Z_MINUS: u32 = 4;
562const Z_PLUS: u32 = 5;

564fn cubemap_uv(direction: vec3<f32>, cubemap_type: u32) -> vec2<f32> {
565    let abs_direction = abs(direction);
566    let max_axis = max(abs_direction.x, max(abs_direction.y, abs_direction.z));
567
568    let face_index = select(
569        select(X_PLUS, X_MINUS, direction.x < 0.0),
570        select(
571            select(Y_PLUS, Y_MINUS, direction.y < 0.0),
572            select(Z_PLUS, Z_MINUS, direction.z < 0.0),
573            max_axis != abs_direction.y
574        ),
575        max_axis != abs_direction.x
576    );
577
578    var face_uv: vec2<f32>;
579    var divisor: f32;
580    var corner_uv: vec2<u32> = vec2(0, 0);
581    var face_size: vec2<f32>;
582
583    switch face_index {
584        case X_PLUS:  { face_uv = vec2<f32>(direction.z, -direction.y); divisor = direction.x; }
585        case X_MINUS: { face_uv = vec2<f32>(-direction.z, -direction.y); divisor = -direction.x; }
586        case Y_PLUS:  { face_uv = vec2<f32>(direction.x,  -direction.z); divisor = direction.y; }
587        case Y_MINUS: { face_uv = vec2<f32>(direction.x, direction.z); divisor = -direction.y; }
588        case Z_PLUS:  { face_uv = vec2<f32>(direction.x, direction.y); divisor = direction.z; }
589        case Z_MINUS: { face_uv = vec2<f32>(direction.x, -direction.y); divisor = -direction.z; }
590        default: {}
591    }
592    face_uv = (face_uv / divisor) * 0.5 + 0.5;
593
594    switch cubemap_type {
595        case CUBEMAP_TYPE_CROSS_VERTICAL: {
596            face_size = vec2(1.0/3.0, 1.0/4.0);
597            corner_uv = vec2<u32>((0x111102u >> (4 * face_index)) & 0xFu, (0x132011u >> (4 * face_index)) & 0xFu);
598        }
599        case CUBEMAP_TYPE_CROSS_HORIZONTAL: {
600            face_size = vec2(1.0/4.0, 1.0/3.0);
601            corner_uv = vec2<u32>((0x131102u >> (4 * face_index)) & 0xFu, (0x112011u >> (4 * face_index)) & 0xFu);
602        }
603        case CUBEMAP_TYPE_SEQUENCE_HORIZONTAL: {
604            face_size = vec2(1.0/6.0, 1.0);
605            corner_uv.x = face_index;
606        }
607        case CUBEMAP_TYPE_SEQUENCE_VERTICAL: {
608            face_size = vec2(1.0, 1.0/6.0);
609            corner_uv.y = face_index;
610        }
611        default: {}
612    }
613
614    return (vec2<f32>(corner_uv) + face_uv) * face_size;
615}

617// This is a modification to Karis 2013 for area lights to fix an issue where the specular
618// reflection on smooth materials looks too rough and dim. We lerp between the base roughness
619// and Karis2013 roughness with a lerp factor tuned by looking at reference renders. The goal
620// is to preserve sharp specular highlights on smooth materials, without blowing out specular
621// highlights on rough materials.
622//
623// The ideal solution is to switch to Linearly Transformed Cosines, which is more accurate in
624// all cases, and a popular choice for realtime.
625fn specular_fix_remap(a: f32) -> f32 {
626    let inv_a_sq = (1.0 - a) * (1.0 - a);
627    return 1.0 - inv_a_sq * inv_a_sq;
628}

630fn point_light(
631    light_id: u32,
632    input: ptr<function, LightingInput>,
633    enable_diffuse: bool,
634    enable_texture: bool,
635) -> vec3<f32> {
636    // Unpack.
637    let diffuse_color = (*input).diffuse_color;
638    let P = (*input).P;
639    let N = (*input).layers[LAYER_BASE].N;
640    let V = (*input).V;
641
642    let light = &view_bindings::clustered_lights.data[light_id];
643    let light_to_frag = (*light).position_radius.xyz - P;
644    let L = normalize(light_to_frag);
645    let distance_square = dot(light_to_frag, light_to_frag);
646    let distance = sqrt(distance_square);
647    let rangeAttenuation = getDistanceAttenuation(distance_square, (*light).color_inverse_square_range.w);
648
649    // Base layer
650
651    let a = (*input).layers[LAYER_BASE].roughness;
652    let specular_L_a_prime = compute_specular_layer_values_for_point_light(
653        input,
654        LAYER_BASE,
655        V,
656        light_to_frag,
657        (*light).position_radius.w,
658        distance,
659    );
660    let L_spec = specular_L_a_prime.xyz;
661    let a_prime = specular_L_a_prime.w;
662    var specular_derived_input = derive_lighting_input(N, V, L_spec);
663
664    let normalizationFactor = a / a_prime;
665    let specular_intensity = normalizationFactor * normalizationFactor;
666
667    let brdf_roughness = mix(a, a_prime, specular_fix_remap(a));
668
669@if(STANDARD_MATERIAL_ANISOTROPY)
670    var specular_light = specular_anisotropy(input, &specular_derived_input, L, brdf_roughness, specular_intensity);
671@else   // STANDARD_MATERIAL_ANISOTROPY
672    var specular_light = specular(input, &specular_derived_input, brdf_roughness, specular_intensity);
673
674    // Sphere area light visibility (solid-angle attenuation)
675    let light_radius = (*light).position_radius.w;
676    if light_radius > 0.0 {
677        let solid_angle = light_radius * light_radius / (distance * distance);
678        specular_light *= saturate(specular_derived_input.NdotL / max(specular_derived_input.NdotL + solid_angle, 1e-4));
679    }
680
681    // Clearcoat
682
683    // Unpack.
684    @if(STANDARD_MATERIAL_CLEARCOAT)
685    let clearcoat_N = (*input).layers[LAYER_CLEARCOAT].N;
686    @if(STANDARD_MATERIAL_CLEARCOAT)
687    let clearcoat_strength = (*input).clearcoat_strength;
688    @if(STANDARD_MATERIAL_CLEARCOAT)
689    let clearcoat_a = (*input).layers[LAYER_CLEARCOAT].roughness;
690
691    // Perform specular input calculations again for the clearcoat layer. We
692    // can't reuse the above because the clearcoat normal might be different
693    // from the main layer normal.
694    @if(STANDARD_MATERIAL_CLEARCOAT)
695    let clearcoat_specular_L_a_prime = compute_specular_layer_values_for_point_light(
696        input,
697        LAYER_CLEARCOAT,
698        V,
699        light_to_frag,
700        (*light).position_radius.w,
701        distance,
702    );
703    @if(STANDARD_MATERIAL_CLEARCOAT)
704    let L_clearcoat_spec = clearcoat_specular_L_a_prime.xyz;
705    @if(STANDARD_MATERIAL_CLEARCOAT)
706    let clearcoat_a_prime = clearcoat_specular_L_a_prime.w;
707    @if(STANDARD_MATERIAL_CLEARCOAT)
708    var clearcoat_specular_derived_input =
709        derive_lighting_input(clearcoat_N, V, L_clearcoat_spec);
710
711    // Calculate the specular light.
712    @if(STANDARD_MATERIAL_CLEARCOAT)
713    let clearcoat_normalizationFactor = clearcoat_a / clearcoat_a_prime;
714
715    @if(STANDARD_MATERIAL_CLEARCOAT)
716    let clearcoat_specular_intensity = clearcoat_normalizationFactor * clearcoat_normalizationFactor;
717
718    @if(STANDARD_MATERIAL_CLEARCOAT)
719    let clearcoat_brdf_roughness = mix(clearcoat_a, clearcoat_a_prime, specular_fix_remap(clearcoat_a));
720
721    @if(STANDARD_MATERIAL_CLEARCOAT)
722    let Fc_Frc = specular_clearcoat(
723        input,
724        &clearcoat_specular_derived_input,
725        clearcoat_strength,
726        clearcoat_brdf_roughness,
727        clearcoat_specular_intensity
728    );
729    @if(STANDARD_MATERIAL_CLEARCOAT)
730    let inv_Fc = 1.0 - Fc_Frc.r;    // Inverse Fresnel term.
731    @if(STANDARD_MATERIAL_CLEARCOAT)
732    var Frc = Fc_Frc.g;             // Clearcoat light.
733
734    // Sphere area light visibility (solid-angle attenuation) for clearcoat
735    @if(STANDARD_MATERIAL_CLEARCOAT)
736    if light_radius > 0.0 {
737        let solid_angle = light_radius * light_radius / (distance * distance);
738        Frc *= saturate(clearcoat_specular_derived_input.NdotL / max(clearcoat_specular_derived_input.NdotL + solid_angle, 1e-4));
739    }
740
741    // Diffuse.
742    // Comes after specular since its N⋅L is used in the lighting equation.
743    var derived_input = derive_lighting_input(N, V, L);
744    var diffuse = vec3(0.0);
745    if (enable_diffuse) {
746        diffuse = diffuse_color * Fd_Burley(input, &derived_input);
747    }
748
749    // See https://google.github.io/filament/Filament.md.html#mjx-eqn-pointLightLuminanceEquation
750    // Lout = f(v,l) Φ / { 4 π d^2 }⟨n⋅l⟩
751    // where
752    // f(v,l) = (f_d(v,l) + f_r(v,l)) * light_color
753    // Φ is luminous power in lumens
754    // our rangeAttenuation = 1 / d^2 multiplied with an attenuation factor for smoothing at the edge of the non-physical maximum light radius
755
756    // For a point light, luminous intensity, I, in lumens per steradian is given by:
757    // I = Φ / 4 π
758    // The derivation of this can be seen here: https://google.github.io/filament/Filament.md.html#mjx-eqn-pointLightLuminousPower
759
760    // NOTE: (*light).color.rgb is premultiplied with (*light).intensity / 4 π (which would be the luminous intensity) on the CPU
761
762    var color_times_NdotL: vec3<f32>;
763@if(STANDARD_MATERIAL_CLEARCOAT)
764    // Account for the Fresnel term from the clearcoat darkening the main layer.
765    //
766    // <https://google.github.io/filament/Filament.md.html#materialsystem/clearcoatmodel/integrationinthesurfaceresponse>
767    color_times_NdotL = (diffuse * derived_input.NdotL + specular_light * specular_derived_input.NdotL * inv_Fc) * inv_Fc + Frc * clearcoat_specular_derived_input.NdotL;
768@else   // STANDARD_MATERIAL_CLEARCOAT
769    color_times_NdotL = diffuse * derived_input.NdotL + specular_light * specular_derived_input.NdotL;
770
771    var texture_sample = 1f;
772
773@if(LIGHT_TEXTURES)
774    if enable_texture && (*light).decal_index != 0xFFFFFFFFu {
775        let relative_position = (view_bindings::clustered_decals.decals[(*light).decal_index].local_from_world * vec4(P, 1.0)).xyz;
776        let cubemap_type = view_bindings::clustered_decals.decals[(*light).decal_index].tag;
777        let decal_uv = cubemap_uv(relative_position, cubemap_type);
778        let image_index = view_bindings::clustered_decals.decals[(*light).decal_index].base_color_texture_index;
779
780        texture_sample = textureSampleLevel(
781            view_bindings::clustered_decal_textures[image_index],
782            view_bindings::clustered_decal_sampler,
783            decal_uv,
784            0.0
785        ).r;
786    }
787
788    return color_times_NdotL * (*light).color_inverse_square_range.rgb *
789        rangeAttenuation * texture_sample;
790}

792fn spot_light(
793    light_id: u32,
794    input: ptr<function, LightingInput>,
795    enable_diffuse: bool
796) -> vec3<f32> {
797    // reuse the point light calculations
798    let point_light = point_light(light_id, input, enable_diffuse, false);
799
800    let light = &view_bindings::clustered_lights.data[light_id];
801
802    // reconstruct spot dir from x/z and y-direction flag
803    var spot_dir = vec3<f32>((*light).light_custom_data.x, 0.0, (*light).light_custom_data.y);
804    spot_dir.y = sqrt(max(0.0, 1.0 - spot_dir.x * spot_dir.x - spot_dir.z * spot_dir.z));
805    if ((*light).flags & POINT_LIGHT_FLAGS_SPOT_LIGHT_Y_NEGATIVE) != 0u {
806        spot_dir.y = -spot_dir.y;
807    }
808    let light_to_frag = (*light).position_radius.xyz - (*input).P.xyz;
809
810    // calculate attenuation based on filament formula https://google.github.io/filament/Filament.md.html#listing_glslpunctuallight
811    // spot_scale and spot_offset have been precomputed
812    // note we normalize here to get "l" from the filament listing. spot_dir is already normalized
813    let cd = dot(-spot_dir, normalize(light_to_frag));
814    let attenuation = saturate(cd * (*light).light_custom_data.z + (*light).light_custom_data.w);
815    let spot_attenuation = attenuation * attenuation;
816
817    var texture_sample = 1f;
818
819@if(LIGHT_TEXTURES)
820    if (*light).decal_index != 0xFFFFFFFFu {
821        let local_position = (view_bindings::clustered_decals.decals[(*light).decal_index].local_from_world *
822            vec4((*input).P, 1.0)).xyz;
823        if local_position.z < 0.0 {
824            let decal_uv = (local_position.xy / (local_position.z * (*light).spot_light_tan_angle)) * vec2(-0.5, 0.5) + 0.5;
825            let image_index = view_bindings::clustered_decals.decals[(*light).decal_index].base_color_texture_index;
826
827            texture_sample = textureSampleLevel(
828                view_bindings::clustered_decal_textures[image_index],
829                view_bindings::clustered_decal_sampler,
830                decal_uv,
831                0.0
832            ).r;
833        }
834    }
835
836    return point_light * spot_attenuation * texture_sample;
837}

840    light_id: u32,
841    input: ptr<function, LightingInput>,
842    enable_diffuse: bool
843) -> vec3<f32> {
844    // Unpack.
845    let diffuse_color = (*input).diffuse_color;

852
853    let L = (*light).direction_to_light.xyz;
854    var derived_input = derive_lighting_input(N, V, L);
855
856    var diffuse = vec3(0.0);
857    if (enable_diffuse) {
858        diffuse = diffuse_color * Fd_Burley(input, &derived_input);
859    }
860
861@if(STANDARD_MATERIAL_ANISOTROPY)
862    let specular_light = specular_anisotropy(input, &derived_input, L, roughness, 1.0);
863@else   // STANDARD_MATERIAL_ANISOTROPY
864    let specular_light = specular(input, &derived_input, roughness, 1.0);
865
866    @if(STANDARD_MATERIAL_CLEARCOAT)
867    let clearcoat_N = (*input).layers[LAYER_CLEARCOAT].N;
868    @if(STANDARD_MATERIAL_CLEARCOAT)
869    let clearcoat_strength = (*input).clearcoat_strength;
870    @if(STANDARD_MATERIAL_CLEARCOAT)
871    let clearcoat_roughness = (*input).layers[LAYER_CLEARCOAT].roughness;
872
873    // Perform specular input calculations again for the clearcoat layer. We
874    // can't reuse the above because the clearcoat normal might be different
875    // from the main layer normal.
876    @if(STANDARD_MATERIAL_CLEARCOAT)
877    var derived_clearcoat_input = derive_lighting_input(clearcoat_N, V, L);
878
879    @if(STANDARD_MATERIAL_CLEARCOAT)
880    let Fc_Frc =
881        specular_clearcoat(input, &derived_clearcoat_input, clearcoat_strength, clearcoat_roughness, 1.0);
882    @if(STANDARD_MATERIAL_CLEARCOAT)
883    let inv_Fc = 1.0 - Fc_Frc.r;
884    @if(STANDARD_MATERIAL_CLEARCOAT)
885    let Frc = Fc_Frc.g;
886
887    var color: vec3<f32>;
888@if(STANDARD_MATERIAL_CLEARCOAT)

890    //
891    // <https://google.github.io/filament/Filament.md.html#materialsystem/clearcoatmodel/integrationinthesurfaceresponse>
892    color = (diffuse + specular_light * inv_Fc) * inv_Fc * derived_input.NdotL +
893        Frc * derived_clearcoat_input.NdotL;
894@else   // STANDARD_MATERIAL_CLEARCOAT
895    color = (diffuse + specular_light) * derived_input.NdotL;
896
897    var texture_sample = 1f;
898
899@if(LIGHT_TEXTURES)
900    if (*light).decal_index != 0xFFFFFFFFu {
901        let local_position = (view_bindings::clustered_decals.decals[(*light).decal_index].local_from_world *
902            vec4((*input).P, 1.0)).xyz;
903        let decal_uv = local_position.xy * vec2(-0.5, 0.5) + 0.5;
904
905        // if tiled or within tile
906        if (view_bindings::clustered_decals.decals[(*light).decal_index].tag != 0u)
907                || all(clamp(decal_uv, vec2(0.0), vec2(1.0)) == decal_uv)
908        {
909            let image_index = view_bindings::clustered_decals.decals[(*light).decal_index].base_color_texture_index;
910
911            texture_sample = textureSampleLevel(
912                view_bindings::clustered_decal_textures[image_index],
913                view_bindings::clustered_decal_sampler,
914                decal_uv - floor(decal_uv),
915                0.0
916            ).r;
917        } else {
918            texture_sample = 0f;
919        }
920    }
921
922color *= (*light).color.rgb * texture_sample;
923
924@if(ATMOSPHERE) {
925    let P = (*input).P;
926    let atmosphere = view_bindings::atmosphere;
927    let P_as = (
928        view_bindings::atmosphere.world_to_atmosphere * vec4(P, 1.0)
929    ).xyz;

931    let r = length(P_clamped);
932    let local_up = normalize(P_clamped);
933    let mu_light = dot(L, local_up);
934
935    // Sample atmosphere
936    let transmittance = sample_transmittance_lut(r, mu_light);
937    let sun_visibility = calculate_visible_sun_ratio(atmosphere, r, mu_light, (*light).sun_disk_angular_size);
938
939    // Apply atmospheric effects
940    color *= transmittance * sun_visibility;
941}
942
943    return color;
944}

946// Linearly Transformed Cosines (LTC) area light evaluation.
947// Based on: Heitz et al. 2016, "Real-Time Polygonal-Light Shading with Linearly Transformed Cosines"
948
949// Integrate one edge of the spherical polygon formed by the LTC-transformed quad.
950// Implements Eq. 11 using a polynomial approximation based on https://advances.realtimerendering.com/s2016/s2016_ltc_rnd.pdf
951// Using the equation as-is produces visible artifacts.
952fn ltc_integrate_edge(v1: vec3<f32>, v2: vec3<f32>) -> f32 {
953    let x = dot(v1, v2);
954    let y = abs(x);
955    let a = 0.8543985 + (0.4965155 + 0.0145206 * y) * y;
956    let b = 3.4175940 + (4.1616724 + y) * y;
957    let v = a / b;
958    let theta_sintheta = select(0.5 * inverseSqrt(max(1.0 - x * x, 1e-7)) - v, v, x > 0.0);
959    return cross(v1, v2).z * theta_sintheta;
960}
961
962fn ltc_integrate_quad(
963    N: vec3<f32>,
964    V: vec3<f32>,
965    P: vec3<f32>,
966    Minv: mat3x3<f32>,
967    points: array<vec3<f32>, 4>
968) -> f32 {
969    let T1 = normalize(V - N * dot(V, N));
970    let T2 = -cross(N, T1);
971
972    let Minv_local = Minv * transpose(mat3x3<f32>(T1, T2, N));
973
974    // Transform quad vertices into LTC-distorted space
975    var L: array<vec3<f32>, 4>;
976    L[0] = Minv_local * (points[0] - P);
977    L[1] = Minv_local * (points[1] - P);
978    L[2] = Minv_local * (points[2] - P);
979    L[3] = Minv_local * (points[3] - P);
980
981    // Clip against z >= 0 hemisphere (at most 5 verts output for a quad)
982
983    // TODO: clipping could be made cheaper with a spherical proxy, see https://advances.realtimerendering.com/s2016/s2016_ltc_rnd.pdf slides 87-102
984    var clipped: array<vec3<f32>, 5>;
985    var n_clipped: i32 = 0;
986
987    for (var i = 0i; i < 4i; i++) {
988        let a = L[i];
989        let b = L[(i + 1) % 4];
990        if (a.z >= 0.0) {
991            clipped[n_clipped] = a;
992            n_clipped++;
993        }
994        if ((a.z >= 0.0) != (b.z >= 0.0)) {
995            let t = a.z / (a.z - b.z);
996            clipped[n_clipped] = mix(a, b, t);
997            n_clipped++;
998        }
999    }
1000
1001    if (n_clipped == 0) {
1002        return 0.0;
1003    }
1004
1005    for (var i = 0i; i < n_clipped; i++) {
1006        clipped[i] = normalize(clipped[i]);
1007    }
1008
1009    // Sum edge integrals over clipped polygon
1010    var sum = 0.0;
1011    for (var i = 0i; i < n_clipped; i++) {
1012        sum += ltc_integrate_edge(clipped[i], clipped[(i + 1) % n_clipped]);
1013    }
1014
1015    return sum;
1016}

1018@if(AREA_LIGHT_LUTS)
1019// Decodes a rect light packed into a `ClusteredLight` (see `GpuClusteredLight`

1038@if(AREA_LIGHT_LUTS)
1039fn rect_light(
1040    light: ptr<function, RectLight>,
1041    input: ptr<function, LightingInput>,
1042    enable_diffuse: bool,
1043) -> vec3<f32> {
1044    // Unpack
1045    let diffuse_color = (*input).diffuse_color;
1046    let P = (*input).P;
1047    let N = (*input).layers[LAYER_BASE].N;
1048    let V = (*input).V;
1049    let NdotV = (*input).layers[LAYER_BASE].NdotV;
1050    let perceptual_roughness = (*input).layers[LAYER_BASE].perceptual_roughness;
1051
1052    let light_to_frag = (*light).position - P;
1053    let distance_square = dot(light_to_frag, light_to_frag);
1054    let inverse_range_squared = 1.0 / max((*light).range * (*light).range, 0.0001);
1055    let range_falloff = getRangeFalloff(distance_square, inverse_range_squared);
1056
1057    let light_normal = cross((*light).up, (*light).right);
1058    let hw = (*light).right * (*light).width  * 0.5;
1059    let hh = (*light).up * (*light).height * 0.5;
1060    var corners: array<vec3<f32>, 4>;
1061    corners[0] = (*light).position + hw - hh;
1062    corners[1] = (*light).position - hw - hh;
1063    corners[2] = (*light).position - hw + hh;
1064    corners[3] = (*light).position + hw + hh;
1065
1066    // Backface test
1067    if dot(light_normal, P - corners[0]) <= 0.0 {
1068        return vec3<f32>(0.0);
1069    }
1070
1071    let LUT_SCALE = 63.0 / 64.0;
1072    let LUT_BIAS  =  0.5 / 64.0;
1073    let uv = vec2<f32>(perceptual_roughness, sqrt(1.0 - NdotV)) * LUT_SCALE + LUT_BIAS;
1074    let t1 = textureSampleLevel(view_bindings::area_light_luts, view_bindings::area_light_luts_sampler, uv, 0, 0.0);
1075    let t2 = textureSampleLevel(view_bindings::area_light_luts, view_bindings::area_light_luts_sampler, uv, 1, 0.0);
1076
1077    // Reconstruct the GGX inverse-LTC matrix
1078    let Minv = mat3x3<f32>(
1079        vec3<f32>(t1.x, 0.0, t1.y),
1080        vec3<f32>(0.0,  1.0, 0.0),
1081        vec3<f32>(t1.z, 0.0, t1.w),
1082    );
1083    let spec = ltc_integrate_quad(N, V, P, Minv, corners);
1084
1085    // Use Lambertian diffuse, Burley would require a second LUT
1086    let identity = mat3x3<f32>(
1087        vec3<f32>(1.0, 0.0, 0.0),
1088        vec3<f32>(0.0, 1.0, 0.0),
1089        vec3<f32>(0.0, 0.0, 1.0),
1090    );
1091    let diff = select(0.0, ltc_integrate_quad(N, V, P, identity, corners), enable_diffuse);
1092
1093    // t2.x encodes the bsdf magnitude and t2.y the fresnel direction
1094    let F0 = mix((*input).F0_dielectric, (*input).F0_metallic, (*input).metallic);
1095    let spec_weight = F0 * t2.x + (1.0 - F0) * t2.y;
1096
1097@if(STANDARD_MATERIAL_CLEARCOAT) {
1098    let clearcoat_N = (*input).layers[LAYER_CLEARCOAT].N;
1099    let clearcoat_strength = (*input).clearcoat_strength;
1100    let clearcoat_perceptual_roughness = (*input).layers[LAYER_CLEARCOAT].perceptual_roughness;
1101    let clearcoat_NdotV = (*input).layers[LAYER_CLEARCOAT].NdotV;
1102
1103    // Sample LUTs for clearcoat layer
1104    let cc_uv = vec2<f32>(clearcoat_perceptual_roughness, sqrt(1.0 - clearcoat_NdotV)) * LUT_SCALE + LUT_BIAS;
1105    let tc1 = textureSampleLevel(view_bindings::area_light_luts, view_bindings::area_light_luts_sampler, cc_uv, 0, 0.0);
1106    let tc2 = textureSampleLevel(view_bindings::area_light_luts, view_bindings::area_light_luts_sampler, cc_uv, 1, 0.0);
1107    let Minv_cc = mat3x3<f32>(
1108        vec3<f32>(tc1.x, 0.0, tc1.y),
1109        vec3<f32>(0.0,   1.0, 0.0),
1110        vec3<f32>(tc1.z, 0.0, tc1.w),
1111    );
1112    let spec_cc = ltc_integrate_quad(clearcoat_N, V, P, Minv_cc, corners);
1113
1114    // Clearcoat has F0=0.04
1115    let spec_weight_cc = 0.04 * tc2.x + (1.0 - 0.04) * tc2.y;
1116    let Fc = clearcoat_strength * spec_weight_cc;
1117    let inv_Fc = 1.0 - Fc;
1118
1119    return ((spec_weight * spec * inv_Fc + diffuse_color * diff) * inv_Fc
1120        + spec_weight_cc * spec_cc * clearcoat_strength) * (*light).color.rgb * range_falloff;
1121} @else {
1122    return (spec_weight * spec + diffuse_color * diff) * (*light).color.rgb * range_falloff;
1123}
1124}
