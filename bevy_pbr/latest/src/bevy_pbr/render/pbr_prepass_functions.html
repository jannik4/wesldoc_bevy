17// Cutoff used for the premultiplied alpha modes BLEND, ADD, and ALPHA_TO_COVERAGE.
18const PREMULTIPLIED_ALPHA_CUTOFF = 0.05;
19
20// Reusable alpha discard logic suitable for use by custom materials in the prepass.
21// We can use a simplified version of alpha_discard() here since we only need to handle the alpha_cutoff
22fn prepass_alpha_discard(material_flags: u32, alpha_cutoff: f32, output_color: vec4<f32>) {
23    let alpha_mode = material_flags & pbr_types::STANDARD_MATERIAL_FLAGS_ALPHA_MODE_RESERVED_BITS;
24    if alpha_mode == pbr_types::STANDARD_MATERIAL_FLAGS_ALPHA_MODE_MASK {
25        if output_color.a < alpha_cutoff {
26            discard;
27        }
28    } else if (alpha_mode == pbr_types::STANDARD_MATERIAL_FLAGS_ALPHA_MODE_BLEND ||
29            alpha_mode == pbr_types::STANDARD_MATERIAL_FLAGS_ALPHA_MODE_ADD ||
30            alpha_mode == pbr_types::STANDARD_MATERIAL_FLAGS_ALPHA_MODE_ALPHA_TO_COVERAGE) {
31        if output_color.a < PREMULTIPLIED_ALPHA_CUTOFF {
32            discard;
33        }
34    } else if alpha_mode == pbr_types::STANDARD_MATERIAL_FLAGS_ALPHA_MODE_PREMULTIPLIED {
35        if all(output_color < vec4(PREMULTIPLIED_ALPHA_CUTOFF)) {
36            discard;
37        }
38    }
39}

93fn calculate_motion_vector(world_position: vec4<f32>, previous_world_position: vec4<f32>) -> vec2<f32> {
94    let clip_position_t = view.unjittered_clip_from_world * world_position;
95    let clip_position = clip_position_t.xy / clip_position_t.w;
96    let previous_clip_position_t = previous_view_uniforms.unjittered_clip_from_world * previous_world_position;
97    let previous_clip_position = previous_clip_position_t.xy / previous_clip_position_t.w;
98    // These motion vectors are used as offsets to UV positions and are stored
99    // in the range -1,1 to allow offsetting from the one corner to the
100    // diagonally-opposite corner in UV coordinates, in either direction.
101    // A difference between diagonally-opposite corners of clip space is in the
102    // range -2,2, so this needs to be scaled by 0.5. And the V direction goes
103    // down where clip space y goes up, so y needs to be flipped.
104    return (clip_position - previous_clip_position) * vec2(0.5, -0.5);
105}
