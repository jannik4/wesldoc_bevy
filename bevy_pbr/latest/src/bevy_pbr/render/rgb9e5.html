20fn floor_log2_(x: f32) -> i32 {
21    let f = bitcast<u32>(x);
22    let biasedexponent = (f & 0x7F800000u) >> 23u;
23    return i32(biasedexponent) - 127;
24}

26// https://www.khronos.org/registry/OpenGL/extensions/EXT/EXT_texture_shared_exponent.txt
27fn vec3_to_rgb9e5_(rgb_in: vec3<f32>) -> u32 {
28    let rgb = clamp(rgb_in, vec3(0.0), vec3(MAX_RGB9E5_));
29
30    let maxrgb = max(rgb.r, max(rgb.g, rgb.b));
31    var exp_shared = max(-RGB9E5_EXP_BIAS - 1, floor_log2_(maxrgb)) + 1 + RGB9E5_EXP_BIAS;
32    var denom = exp2(f32(exp_shared - RGB9E5_EXP_BIAS - RGB9E5_MANTISSA_BITS));
33
34    let maxm = i32(floor(maxrgb / denom + 0.5));
35    if (maxm == RGB9E5_MANTISSA_VALUES) {
36        denom *= 2.0;
37        exp_shared += 1;
38    }
39
40    let n = vec3<u32>(floor(rgb / denom + 0.5));
41    
42    return (u32(exp_shared) << 27u) | (n.b << 18u) | (n.g << 9u) | (n.r << 0u);
43}

45// Builtin extractBits() is not working on WEBGL or DX12
46// DX12: HLSL: Unimplemented("write_expr_math ExtractBits")
47fn extract_bits(value: u32, offset: u32, bits: u32) -> u32 {
48    let mask = (1u << bits) - 1u;
49    return (value >> offset) & mask;
50}

52fn rgb9e5_to_vec3_(v: u32) -> vec3<f32> {
53    let exponent = i32(extract_bits(v, 27u, RGB9E5_EXPONENT_BITS)) - RGB9E5_EXP_BIAS - RGB9E5_MANTISSA_BITS;
54    let scale = exp2(f32(exponent));
55
56    return vec3(
57        f32(extract_bits(v, 0u, RGB9E5_MANTISSA_BITSU)),
58        f32(extract_bits(v, 9u, RGB9E5_MANTISSA_BITSU)),
59        f32(extract_bits(v, 18u, RGB9E5_MANTISSA_BITSU))
60    ) * scale;
61}
