64@if(SKINNED)
65fn inverse_transpose_3x3m(in: mat3x3<f32>) -> mat3x3<f32> {
66    let x = cross(in[1], in[2]);
67    let y = cross(in[2], in[0]);
68    let z = cross(in[0], in[1]);
69    let det = dot(in[2], z);
70    return mat3x3<f32>(
71        x / det,
72        y / det,
73        z / det
74    );
75}
