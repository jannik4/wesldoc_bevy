39fn load_noise(pixel_coordinates: vec2<i32>) -> vec2<f32> {
40    var index = textureLoad(hilbert_index_lut, pixel_coordinates % 64, 0).r;
41
42@if(TEMPORAL_JITTER)
43    index += 288u * (globals.frame_count % 64u);
44
45    // R2 sequence - http://extremelearning.com.au/unreasonable-effectiveness-of-quasirandom-sequences
46    return fract(0.5 + f32(index) * vec2<f32>(0.75487766624669276005, 0.5698402909980532659114));
47}

49// Calculate differences in depth between neighbor pixels (later used by the spatial denoiser pass to preserve object edges)
50fn calculate_neighboring_depth_differences(pixel_coordinates: vec2<i32>) -> f32 {
51    // Sample the pixel's depth and 4 depths around it
52    let uv = vec2<f32>(pixel_coordinates) / view.viewport.zw;
53    let depths_upper_left = textureGather(0, preprocessed_depth, point_clamp_sampler, uv);
54    let depths_bottom_right = textureGather(0, preprocessed_depth, point_clamp_sampler, uv, vec2<i32>(1i, 1i));
55    let depth_center = depths_upper_left.y;
56    let depth_left = depths_upper_left.x;
57    let depth_top = depths_upper_left.z;
58    let depth_bottom = depths_bottom_right.x;
59    let depth_right = depths_bottom_right.z;
60
61    // Calculate the depth differences (large differences represent object edges)
62    var edge_info = vec4<f32>(depth_left, depth_right, depth_top, depth_bottom) - depth_center;
63    let slope_left_right = (edge_info.y - edge_info.x) * 0.5;
64    let slope_top_bottom = (edge_info.w - edge_info.z) * 0.5;
65    let edge_info_slope_adjusted = edge_info + vec4<f32>(slope_left_right, -slope_left_right, slope_top_bottom, -slope_top_bottom);
66    edge_info = min(abs(edge_info), abs(edge_info_slope_adjusted));
67    let bias = 0.25; // Using the bias and then saturating nudges the values a bit
68    let scale = depth_center * 0.011; // Weight the edges by their distance from the camera
69    edge_info = saturate((1.0 + bias) - edge_info / scale); // Apply the bias and scale, and invert edge_info so that small values become large, and vice versa
70
71    // Pack the edge info into the texture
72    let edge_info_packed = vec4<u32>(pack4x8unorm(edge_info), 0u, 0u, 0u);
73    textureStore(depth_differences, pixel_coordinates, edge_info_packed);
74
75    return depth_center;
76}

78fn load_normal_view_space(uv_normals: vec2<f32>) -> vec3<f32> {
79    var world_normal = textureSampleLevel(normals, point_clamp_sampler, uv_normals, 0.0).xyz;
80    world_normal = (world_normal * 2.0) - 1.0;
81    let view_from_world = mat3x3<f32>(
82        view.view_from_world[0].xyz,
83        view.view_from_world[1].xyz,
84        view.view_from_world[2].xyz,
85    );
86    return view_from_world * world_normal;
87}

89fn reconstruct_view_space_position(depth: f32, uv: vec2<f32>) -> vec3<f32> {
90    let clip_xy = vec2<f32>(uv.x * 2.0 - 1.0, 1.0 - 2.0 * uv.y);
91    let t = view.view_from_clip * vec4<f32>(clip_xy, depth, 1.0);
92    let view_xyz = t.xyz / t.w;
93    return view_xyz;
94}

96fn load_and_reconstruct_view_space_position(uv: vec2<f32>, sample_mip_level: f32) -> vec3<f32> {
97    let depth = textureSampleLevel(preprocessed_depth, linear_clamp_sampler, uv, sample_mip_level).r;
98    return reconstruct_view_space_position(depth, uv);
99}

101fn updateSectors(
102    min_horizon: f32,
103    max_horizon: f32,
104    samples_per_slice: f32,
105    bitmask: u32,
106) -> u32 {
107    let start_horizon = u32(min_horizon * samples_per_slice);
108    let angle_horizon = u32(ceil((max_horizon - min_horizon) * samples_per_slice));
109
110    return insertBits(bitmask, 0xFFFFFFFFu, start_horizon, angle_horizon);
111}

113fn processSample(
114    delta_position: vec3<f32>,
115    view_vec: vec3<f32>,
116    sampling_direction: f32,
117    n: vec2<f32>,
118    samples_per_slice: f32,
119    bitmask: ptr<function, u32>,
120) {
121    let delta_position_back_face = delta_position - view_vec * settings.thickness;
122
123    var front_back_horizon = vec2(
124        fast_acos(dot(normalize(delta_position), view_vec)),
125        fast_acos(dot(normalize(delta_position_back_face), view_vec)),
126    );
127
128    front_back_horizon = saturate(fma(vec2(sampling_direction), -front_back_horizon * (1.0 / PI), n));
129    front_back_horizon = select(front_back_horizon.xy, front_back_horizon.yx, sampling_direction >= 0.0);
130
131    *bitmask = updateSectors(front_back_horizon.x, front_back_horizon.y, samples_per_slice, *bitmask);
132}

134@compute

136fn ssao(@builtin(global_invocation_id) global_id: vec3<u32>) {
137    let slice_count = f32(constants::SLICE_COUNT);
138    let samples_per_slice_side = f32(constants::SAMPLES_PER_SLICE_SIDE);
139
140    let pixel_coordinates = vec2<i32>(global_id.xy);
141    let uv = (vec2<f32>(pixel_coordinates) + 0.5) / view.viewport.zw;
142
143    var pixel_depth = calculate_neighboring_depth_differences(pixel_coordinates);
144    pixel_depth += 0.00001; // Avoid depth precision issues
145
146    let pixel_position = reconstruct_view_space_position(pixel_depth, uv);
147    let normals_size = vec2<f32>(textureDimensions(normals));
148    let uv_normals = (vec2<f32>(pixel_coordinates) + 0.5 + view.viewport.xy) / normals_size;
149    let pixel_normal = load_normal_view_space(uv_normals);
150    let view_vec = normalize(-pixel_position);
151
152    let noise = load_noise(pixel_coordinates);
153    let safe_radius = max(settings.radius, 0.0001);
154    let sample_scale = (-0.5 * safe_radius * view.clip_from_view[0][0]) / pixel_position.z;
155
156    var visibility = 0.0;
157    var occluded_sample_count = 0u;
158    for (var slice_t = 0.0; slice_t < slice_count; slice_t += 1.0) {
159        let slice = slice_t + noise.x;
160        let phi = (PI / slice_count) * slice;
161        let omega = vec2<f32>(cos(phi), sin(phi));
162
163        let direction = vec3<f32>(omega.xy, 0.0);
164        let orthographic_direction = direction - (dot(direction, view_vec) * view_vec);
165        let axis = cross(direction, view_vec);
166        let projected_normal = pixel_normal - axis * dot(pixel_normal, axis);
167        let projected_normal_length = length(projected_normal);
168
169        let sign_norm = sign(dot(orthographic_direction, projected_normal));
170        let cos_norm = saturate(dot(projected_normal, view_vec) / projected_normal_length);
171        let n = vec2((HALF_PI - sign_norm * fast_acos(cos_norm)) * (1.0 / PI));
172
173        var bitmask = 0u;
174
175        let sample_mul = vec2<f32>(omega.x, -omega.y) * sample_scale;
176        for (var sample_t = 0.0; sample_t < samples_per_slice_side; sample_t += 1.0) {
177            var sample_noise = (slice_t + sample_t * samples_per_slice_side) * 0.6180339887498948482;
178            sample_noise = fract(noise.y + sample_noise);
179
180            var s = (sample_t + sample_noise) / samples_per_slice_side;
181            s *= s; // https://github.com/GameTechDev/XeGTAO#sample-distribution
182            let sample = s * sample_mul;
183
184            // * view.viewport.zw gets us from [0, 1] to [0, viewport_size], which is needed for this to get the correct mip levels
185            let sample_mip_level = clamp(log2(length(sample * view.viewport.zw)) - 3.3, 0.0, 5.0); // https://github.com/GameTechDev/XeGTAO#memory-bandwidth-bottleneck
186            let sample_position_1 = load_and_reconstruct_view_space_position(uv + sample, sample_mip_level);
187            let sample_position_2 = load_and_reconstruct_view_space_position(uv - sample, sample_mip_level);
188
189            let sample_difference_1 = sample_position_1 - pixel_position;
190            let sample_difference_2 = sample_position_2 - pixel_position;
191
192            processSample(sample_difference_1, view_vec, -1.0, n, samples_per_slice_side * 2.0, &bitmask);
193            processSample(sample_difference_2, view_vec, 1.0, n, samples_per_slice_side * 2.0, &bitmask);
194        }
195
196        occluded_sample_count += countOneBits(bitmask);
197    }
198
199    visibility = 1.0 - f32(occluded_sample_count) / (slice_count * 2.0 * samples_per_slice_side);
200
201    visibility = clamp(visibility, 0.03, 1.0);
202
203    textureStore(ambient_occlusion, pixel_coordinates, vec4<f32>(visibility, 0.0, 0.0, 0.0));
204}
