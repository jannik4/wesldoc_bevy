16// The IOR impacts how much a single microfacet refracts incoming light. In turn this changes how
17// "wide" the distribution of output ray directions is and how rough the BRDF should be.
18// This function scales the roughness toward 0.0 as IOR tends towards 1.0, while leaving the default
19// roughness for IOR = 1.5.
20// Adapted from Khronos gltf-Sample-Renderer (https://github.com/KhronosGroup/glTF-Sample-Renderer/blob/bec106e53da4a6a398aa3205f0f96563519a657e/source/Renderer/shaders/functions.glsl#L84-L89)
21fn ior_corrected_roughness(roughness: f32, ior: f32) -> f32 {
22    return roughness * clamp(ior * 2.0 - 2.0, 0.0, 1.0);
23}

25fn specular_transmissive_light(world_position: vec4<f32>, frag_coord: vec3<f32>, view_z: f32, N: vec3<f32>, V: vec3<f32>, F0: vec3<f32>, ior: f32, thickness: f32, perceptual_roughness: f32, specular_transmissive_color: vec3<f32>, transmitted_environment_light_specular: vec3<f32>) -> vec3<f32> {
26    // Calculate the ratio between refraction indexes. Assume air/vacuum for the space outside the mesh
27    let eta = 1.0 / ior;
28
29    // Calculate incidence vector (opposite to view vector) and its dot product with the mesh normal
30    let I = -V;
31    let NdotI = dot(N, I);
32
33    // Calculate refracted direction using Snell's law
34    let k = 1.0 - eta * eta * (1.0 - NdotI * NdotI);
35    let T = eta * I - (eta * NdotI + sqrt(k)) * N;
36
37    // Calculate the exit position of the refracted ray, by propagating refracted direction through thickness
38    let exit_position = world_position.xyz + T * thickness;
39
40    // Transform exit_position into clip space
41    let clip_exit_position = view_bindings::view.clip_from_world * vec4<f32>(exit_position, 1.0);
42
43    // Scale / offset position so that coordinate is in right space for sampling transmissive background texture
44    let subview_position = (clip_exit_position.xy / clip_exit_position.w) * vec2<f32>(0.5, -0.5) + 0.5;
45    let viewport_origin = view_bindings::view.viewport.xy;
46    let viewport_size = view_bindings::view.viewport.zw;
47    let full_size = vec2<f32>(textureDimensions(view_bindings::view_transmission_texture));
48    let offset_position = (subview_position * viewport_size + viewport_origin) / full_size;
49
50    // Fetch background color
51    var background_color: vec4<f32>;
52    let transmission_roughness = ior_corrected_roughness(perceptual_roughness, ior);
53    if transmission_roughness == 0.0 {
54        // If transmission roughness is zero, we can use a faster approach without the blur.
55        background_color = fetch_transmissive_background_non_rough(offset_position, frag_coord);
56    } else {
57        background_color = fetch_transmissive_background(offset_position, frag_coord, view_z, transmission_roughness);
58    }
59
60    // Compensate for exposure, since the background color is coming from an already exposure-adjusted texture
61    background_color = vec4(background_color.rgb / view_bindings::view.exposure, background_color.a);
62
63    // Dot product of the refracted direction with the exit normal (Note: We assume the exit normal is the entry normal but inverted)
64    let MinusNdotT = dot(-N, T);
65
66    // Calculate 1.0 - fresnel factor (how much light is _NOT_ reflected, i.e. how much is transmitted)
67    let F = vec3(1.0) - lighting::fresnel(F0, MinusNdotT);
68
69    // Calculate final color by applying fresnel multiplied specular transmissive color to a mix of background color and transmitted specular environment light
70    return F * specular_transmissive_color * mix(transmitted_environment_light_specular, background_color.rgb, background_color.a);
71}

73fn fetch_transmissive_background_non_rough(offset_position: vec2<f32>, frag_coord: vec3<f32>) -> vec4<f32> {
74    var background_color = textureSampleLevel(
75        view_bindings::view_transmission_texture,
76        view_bindings::view_transmission_sampler,
77        offset_position,
78        0.0
79    );
80
81    @if(DEPTH_PREPASS && !WEBGL2)
82    // Use depth prepass data to reject values that are in front of the current fragment
83    if prepass_utils::prepass_depth(vec4<f32>(offset_position * view_bindings::view.viewport.zw, 0.0, 0.0), 0u) > frag_coord.z {
84        background_color.a = 0.0;
85    }
86
87    @if(TONEMAP_IN_SHADER)
88    background_color = approximate_inverse_tone_mapping(background_color, view_bindings::view.color_grading);
89
90    return background_color;
91}

93fn fetch_transmissive_background(offset_position: vec2<f32>, frag_coord: vec3<f32>, view_z: f32, transmission_roughness: f32) -> vec4<f32> {
94    // Calculate view aspect ratio, used to scale offset so that it's proportionate
95    let aspect = view_bindings::view.viewport.z / view_bindings::view.viewport.w;
96
97    // Calculate how “blurry” the transmission should be.
98    // Blur is more or less eyeballed to look approximately “right”, since the “correct”
99    // approach would involve projecting many scattered rays and figuring out their individual
100    // exit positions. IRL, light rays can be scattered when entering/exiting a material (due to
101    // roughness) or inside the material (due to subsurface scattering). Here, we only consider
102    // the first scenario.
103    //
104    // Blur intensity is:
105    // - proportional to the square of `perceptual_roughness`
106    // - reduced for IOR < 1.5 (through `transmission_roughness`)
107    // - proportional to the inverse of view z
108    let blur_intensity = (transmission_roughness * transmission_roughness) / view_z;
109
110    @if(SCREEN_SPACE_SPECULAR_TRANSMISSION_BLUR_TAPS)
111    let num_taps = constants::SCREEN_SPACE_SPECULAR_TRANSMISSION_BLUR_TAPS; // Controlled by the `ScreenSpaceTransmission::quality` property
112    @else
113    let num_taps = 8; // Fallback to 8 taps, if not specified
114    let num_spirals = i32(ceil(f32(num_taps) / 8.0));
115    @if(TEMPORAL_JITTER)
116    let random_angle = interleaved_gradient_noise(frag_coord.xy, view_bindings::globals.frame_count);
117    @else
118    let random_angle = interleaved_gradient_noise(frag_coord.xy, 0u);
119    // Pixel checkerboard pattern (helps make the interleaved gradient noise pattern less visible)
120    @if(TEMPORAL_JITTER)
121    let pixel_checkboard = (
122        // 0 or 1 on even/odd pixels, alternates every frame
123        (i32(frag_coord.x) + i32(frag_coord.y) + i32(view_bindings::globals.frame_count)) % 2
124    );
125    @else
126    let pixel_checkboard = (
127        // 0 or 1 on even/odd pixels
128        (i32(frag_coord.x) + i32(frag_coord.y)) % 2
129    );
130
131    var result = vec4<f32>(0.0);
132    for (var i: i32 = 0; i < num_taps; i = i + 1) {
133        let current_spiral = (i >> 3u);
134        let angle = (random_angle + f32(current_spiral) / f32(num_spirals)) * 2.0 * PI;
135        let m = vec2(sin(angle), cos(angle));
136        let rotation_matrix = mat2x2(
137            m.y, -m.x,
138            m.x, m.y
139        );
140
141        // Get spiral offset
142        var spiral_offset: vec2<f32>;
143        switch i & 7 {
144            // https://www.iryoku.com/next-generation-post-processing-in-call-of-duty-advanced-warfare (slides 120-135)
145            // TODO: Figure out a more reasonable way of doing this, as WGSL
146            // seems to only allow constant indexes into constant arrays at the moment.
147            // The downstream shader compiler should be able to optimize this into a single
148            // constant when unrolling the for loop, but it's still not ideal.
149            case 0: { spiral_offset = utils::SPIRAL_OFFSET_0_; } // Note: We go even first and then odd, so that the lowest
150            case 1: { spiral_offset = utils::SPIRAL_OFFSET_2_; } // quality possible (which does 4 taps) still does a full spiral
151            case 2: { spiral_offset = utils::SPIRAL_OFFSET_4_; } // instead of just the first half of it
152            case 3: { spiral_offset = utils::SPIRAL_OFFSET_6_; }
153            case 4: { spiral_offset = utils::SPIRAL_OFFSET_1_; }
154            case 5: { spiral_offset = utils::SPIRAL_OFFSET_3_; }
155            case 6: { spiral_offset = utils::SPIRAL_OFFSET_5_; }
156            case 7: { spiral_offset = utils::SPIRAL_OFFSET_7_; }
157            default: {}
158        }
159
160        // Make each consecutive spiral slightly smaller than the previous one
161        spiral_offset *= 1.0 - (0.5 * f32(current_spiral + 1) / f32(num_spirals));
162
163        // Rotate and correct for aspect ratio
164        let rotated_spiral_offset = (rotation_matrix * spiral_offset) * vec2(1.0, aspect);
165
166        // Calculate final offset position, with blur and spiral offset
167        let modified_offset_position = offset_position + rotated_spiral_offset * blur_intensity * (1.0 - f32(pixel_checkboard) * 0.1);
168
169        // Sample the view transmission texture at the offset position + noise offset, to get the background color
170        var sample = textureSampleLevel(
171            view_bindings::view_transmission_texture,
172            view_bindings::view_transmission_sampler,
173            modified_offset_position,
174            0.0
175        );
176
177        @if(DEPTH_PREPASS && !WEBGL2)
178        // Use depth prepass data to reject values that are in front of the current fragment
179        if prepass_utils::prepass_depth(vec4<f32>(modified_offset_position * view_bindings::view.viewport.zw, 0.0, 0.0), 0u) > frag_coord.z {
180            sample = vec4<f32>(0.0);
181        }
182
183        // As blur intensity grows higher, gradually limit *very bright* color RGB values towards a
184        // maximum length of 1.0 to prevent stray “firefly” pixel artifacts. This can potentially make
185        // very strong emissive meshes appear much dimmer, but the artifacts are noticeable enough to
186        // warrant this treatment.
187        let normalized_rgb = normalize(sample.rgb);
188        result += vec4(min(sample.rgb, normalized_rgb / saturate(blur_intensity / 2.0)), sample.a);
189    }
190
191    result /= f32(num_taps);
192
193    @if(TONEMAP_IN_SHADER)
194    result = approximate_inverse_tone_mapping(result, view_bindings::view.color_grading);
195
196    return result;
197}
