1// The chromatic aberration postprocessing effect.
2//
3// This makes edges of objects turn into multicolored streaks.
4
5// See `package::effect_stack::ChromaticAberration` for more
6// information on these fields.
7struct ChromaticAberrationSettings {
8    intensity: f32,
9    max_samples: u32,
10    unused_a: u32,
11    unused_b: u32,
12}
13
14// The source framebuffer texture.
15@group(0) @binding(0) var source_texture: texture_2d<f32>;
16// The sampler used to sample the source framebuffer texture.
17@group(0) @binding(1) var common_sampler: sampler;
18// The 1D lookup table for chromatic aberration.
19@group(0) @binding(2) var chromatic_aberration_lut_texture: texture_2d<f32>;
20// The settings supplied by the developer.
21@group(0) @binding(3) var<uniform> chromatic_aberration_settings: ChromaticAberrationSettings;
22
23fn chromatic_aberration(start_pos: vec2<f32>) -> vec3<f32> {
24    // Radial chromatic aberration implemented using the *Inside* technique:
25    //
26    // <https://github.com/playdeadgames/publications/blob/master/INSIDE/rendering_inside_gdc2016.pdf>
27
28    let end_pos = mix(start_pos, vec2(0.5), chromatic_aberration_settings.intensity);
29
30    // Determine the number of samples. We aim for one sample per texel, unless
31    // that's higher than the developer-specified maximum number of samples, in
32    // which case we choose the maximum number of samples.
33    let texel_length = length((end_pos - start_pos) *
34        vec2<f32>(textureDimensions(source_texture)));
35    let sample_count = min(u32(ceil(texel_length)), chromatic_aberration_settings.max_samples);
36
37    var color: vec3<f32>;
38    if (sample_count > 1u) {
39        // The LUT texture is in clamp-to-edge mode, so we start at 0.5 texels
40        // from the sides so that we have a nice gradient over the entire LUT
41        // range.
42        let lut_u_offset = 0.5 / f32(textureDimensions(chromatic_aberration_lut_texture).x);
43
44        var sample_sum = vec3(0.0);
45        var modulate_sum = vec3(0.0);
46
47        // Start accumulating samples.
48        for (var sample_index = 0u; sample_index < sample_count; sample_index += 1u) {
49            let t = (f32(sample_index) + 0.5) / f32(sample_count);
50
51            // Sample the framebuffer.
52            let sample_uv = mix(start_pos, end_pos, t);
53            let sample = textureSampleLevel(
54                source_texture,
55                common_sampler,
56                sample_uv,
57                0.0,
58            ).rgb;
59
60            // Sample the LUT.
61            let lut_u = mix(lut_u_offset, 1.0 - lut_u_offset, t);
62            let modulate = textureSampleLevel(
63                chromatic_aberration_lut_texture,
64                common_sampler,
65                vec2(lut_u, 0.5),
66                0.0,
67            ).rgb;
68
69            // Modulate the sample by the LUT value.
70            sample_sum += sample * modulate;
71            modulate_sum += modulate;
72        }
73
74        color = sample_sum / modulate_sum;
75    } else {
76        // If there's only one sample, don't do anything. If we don't do this,
77        // then this shader will apply whatever tint is in the center of the LUT
78        // texture to such pixels, which is wrong.
79        color = textureSampleLevel(
80            source_texture,
81            common_sampler,
82            start_pos,
83            0.0,
84        ).rgb;
85    }
86
87    return color;
88}
