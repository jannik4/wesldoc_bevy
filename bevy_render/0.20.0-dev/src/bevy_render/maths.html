42// Returns the inverse of a 3x3 matrix.
43fn inverse_mat3x3(matrix: mat3x3<f32>) -> mat3x3<f32> {
44    let tmp0 = cross(matrix[1], matrix[2]);
45    let tmp1 = cross(matrix[2], matrix[0]);
46    let tmp2 = cross(matrix[0], matrix[1]);
47    let inv_det = 1.0 / dot(matrix[2], tmp2);
48    return transpose(mat3x3<f32>(tmp0 * inv_det, tmp1 * inv_det, tmp2 * inv_det));
49}

51// Returns the inverse of an affine matrix.
52//
53// https://en.wikipedia.org/wiki/Affine_transformation#Groups
54fn inverse_affine3(affine: mat4x3<f32>) -> mat4x3<f32> {
55    let matrix3 = affine3_to_mat3x3(affine);
56    let inv_matrix3 = inverse_mat3x3(matrix3);
57    return mat4x3<f32>(inv_matrix3[0], inv_matrix3[1], inv_matrix3[2], -(inv_matrix3 * affine[3]));
58}

65// Copy the sign bit from B onto A.
66// copysign allows proper handling of negative zero to match the rust implementation of orthonormalize
67fn copysign(a: f32, b: f32) -> f32 {
68    return bitcast<f32>((bitcast<u32>(a) & 0x7FFFFFFF) | (bitcast<u32>(b) & 0x80000000));
69}

71// Constructs a right-handed orthonormal basis from a given unit Z vector.
72//
73// NOTE: requires unit-length (normalized) input to function properly.
74//
75// https://jcgt.org/published/0006/01/01/paper.pdf
76// this method of constructing a basis from a vec3 is also used by `glam::Vec3::any_orthonormal_pair`
77// the construction of the orthonormal basis up and right vectors here needs to precisely match the rust
78// implementation in bevy_light/spot_light.rs:spot_light_world_from_view
79fn orthonormalize(z_basis: vec3<f32>) -> mat3x3<f32> {
80    let sign = copysign(1.0, z_basis.z);
81    let a = -1.0 / (sign + z_basis.z);
82    let b = z_basis.x * z_basis.y * a;
83    let x_basis = vec3(1.0 + sign * z_basis.x * z_basis.x * a, sign * b, -sign * z_basis.x);
84    let y_basis = vec3(b, sign + z_basis.y * z_basis.y * a, -z_basis.y);
85    return mat3x3(x_basis, y_basis, z_basis);
86}

94// Returns true if any part of a sphere is on the positive side of a plane.
95//
96// `sphere_center.w` should be 1.0.
97//
98// This is used for frustum culling.
99fn sphere_intersects_plane_half_space(
100    plane: vec4<f32>,
101    sphere_center: vec4<f32>,
102    sphere_radius: f32
103) -> bool {
104    return dot(plane, sphere_center) + sphere_radius > 0.0;

107// Returns the distances along the ray to its intersections with a sphere
108// centered at the origin.
109//
110// r: distance from the sphere center to the ray origin
111// mu: cosine of the zenith angle
112// sphere_radius: radius of the sphere
113//
114// Returns vec2(t0, t1). If there is no intersection, returns vec2(-1.0).
115fn ray_sphere_intersect(r: f32, mu: f32, sphere_radius: f32) -> vec2<f32> {
116    let discriminant = r * r * (mu * mu - 1.0) + sphere_radius * sphere_radius;
117    
118    // No intersection
119    if discriminant < 0.0 {
120        return vec2(-1.0);
121    }
122    
123    let q = -r * mu;
124    let sqrt_discriminant = sqrt(discriminant);
125    
126    // Return both intersection distances
127    return vec2(
128        q - sqrt_discriminant,
129        q + sqrt_discriminant
130    );
131}

138// https://en.wikipedia.org/wiki/Vector_projection#Vector_projection_2
139fn project_onto(lhs: vec3<f32>, rhs: vec3<f32>) -> vec3<f32> {
140    let other_len_sq_rcp = 1.0 / dot(rhs, rhs);
141    return rhs * dot(lhs, rhs) * other_len_sq_rcp;
142}

144// Below are fast approximations of common irrational and trig functions. These
145// are likely most useful when raymarching, for example, where complete numeric
146// accuracy can be sacrificed for greater sample count.
147
148// Slightly less accurate than fast_acos_4, but much simpler.
149fn fast_acos(in_x: f32) -> f32 {
150    let x = abs(in_x);
151    var res = -0.156583 * x + HALF_PI;
152    res *= sqrt(1.0 - x);
153    return select(PI - res, res, in_x >= 0.0);
154}

156// 4th order polynomial approximation
157// 4 VGRP, 16 ALU Full Rate
158// 7 * 10^-5 radians precision
159// Reference : Handbook of Mathematical Functions (chapter : Elementary Transcendental Functions), M. Abramowitz and I.A. Stegun, Ed.
160fn fast_acos_4(x: f32) -> f32 {
161    let x1 = abs(x);
162    let x2 = x1 * x1;
163    let x3 = x2 * x1;
164    var s: f32;
165
166    s = -0.2121144 * x1 + 1.5707288;
167    s = 0.0742610 * x2 + s;
168    s = -0.0187293 * x3 + s;
169    s = sqrt(1.0 - x1) * s;
170
171	// acos function mirroring
172    return select(PI - s, s, x >= 0.0);
173}

175fn fast_atan2(y: f32, x: f32) -> f32 {
176    var t0 = max(abs(x), abs(y));
177    var t1 = min(abs(x), abs(y));
178    var t3 = t1 / t0;
179    var t4 = t3 * t3;
180
181    t0 = 0.0872929;
182    t0 = t0 * t4 - 0.301895;
183    t0 = t0 * t4 + 1.0;
184    t3 = t0 * t3;
185
186    t3 = select(t3, (0.5 * PI) - t3, abs(y) > abs(x));
187    t3 = select(t3, PI - t3, x < 0);
188    t3 = select(-t3, t3, y > 0);
189
190    return t3;
