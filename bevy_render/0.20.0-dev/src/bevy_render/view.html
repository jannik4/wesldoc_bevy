14struct View {
15    clip_from_world: mat4x4<f32>,
16    unjittered_clip_from_world: mat4x4<f32>,
17    world_from_clip: mat4x4<f32>,
18    world_from_view: mat4x4<f32>,
19    view_from_world: mat4x4<f32>,
20    // Typically a column-major right-handed projection matrix, one of either:
21    //
22    // Perspective (infinite reverse z)
23    // ```
24    // f = 1 / tan(fov_y_radians / 2)
25    //
26    // ⎡ f / aspect  0   0     0 ⎤
27    // ⎢          0  f   0     0 ⎥
28    // ⎢          0  0   0  near ⎥
29    // ⎣          0  0  -1     0 ⎦
30    // ```
31    //
32    // Orthographic
33    // ```
34    // w = right - left
35    // h = top - bottom
36    // d = far - near
37    // cw = -right - left
38    // ch = -top - bottom
39    //
40    // ⎡ 2 / w      0      0   cw / w ⎤
41    // ⎢     0  2 / h      0   ch / h ⎥
42    // ⎢     0      0  1 / d  far / d ⎥
43    // ⎣     0      0      0        1 ⎦
44    // ```
45    //
46    // `clip_from_view[3][3] == 1.0` is the standard way to check if a projection is orthographic
47    //
48    // Wgsl matrices are column major, so for example getting the near plane of a perspective projection is `clip_from_view[3][2]`
49    //
50    // Custom projections are also possible however.
51    clip_from_view: mat4x4<f32>,
52    view_from_clip: mat4x4<f32>,
53    world_position: vec3<f32>,
54    exposure: f32,
55    // viewport(x_origin, y_origin, width, height)
56    viewport: vec4<f32>,
57    main_pass_viewport: vec4<f32>,
58    // 6 world-space half spaces (normal: vec3, distance: f32) ordered left, right, top, bottom, near, far.
59    // The normal vectors point towards the interior of the frustum.
60    // A half space contains `p` if `normal.dot(p) + distance > 0.`
61    frustum: array<vec4<f32>, 6>,
62    // The world-space position of the camera used to resolve visibility ranges.
63    //
64    // This is the position of the camera itself, unless this view isn't
65    // associated with a camera, in which case it's the position of the primary
66    // camera.
67    lod_view_world_position: vec3<f32>,
68    color_grading: ColorGrading,
69    mip_bias: f32,
70    frame_count: u32,
71};

73/// World space:
74/// +y is up
75
76/// View space:
77/// -z is forward, +x is right, +y is up
78/// Forward is from the camera position into the scene.
79/// (0.0, 0.0, -1.0) is linear distance of 1.0 in front of the camera's view relative to the camera's rotation
80/// (0.0, 1.0, 0.0) is linear distance of 1.0 above the camera's view relative to the camera's rotation
81
82/// NDC (normalized device coordinate):
83/// https://www.w3.org/TR/webgpu/#coordinate-systems
84/// (-1.0, -1.0) in NDC is located at the bottom-left corner of NDC
85/// (1.0, 1.0) in NDC is located at the top-right corner of NDC
86/// Z is depth where:
87///    1.0 is near clipping plane
88///    Perspective projection: 0.0 is inf far away
89///    Orthographic projection: 0.0 is far clipping plane
90
91/// Clip space:
92/// This is NDC before the perspective divide, still in homogenous coordinate space.
93/// Dividing a clip space point by its w component yields a point in NDC space.
94
95/// UV space:
96/// 0.0, 0.0 is the top left
97/// 1.0, 1.0 is the bottom right
98

216/// Retrieve the perspective camera near clipping plane
217fn perspective_camera_near(clip_from_view: mat4x4<f32>) -> f32 {
218    return clip_from_view[3][2];
219}
220
221/// Convert ndc depth to linear view z.
222/// Note: Depth values in front of the camera will be negative as -z is forward
223fn depth_ndc_to_view_z(ndc_depth: f32, clip_from_view: mat4x4<f32>, view_from_clip: mat4x4<f32>) -> f32 {
224@if(VIEW_PROJECTION_PERSPECTIVE)
225    return -perspective_camera_near(clip_from_view) / ndc_depth;
226@elif(VIEW_PROJECTION_ORTHOGRAPHIC)
227    return -(clip_from_view[3][2] - ndc_depth) / clip_from_view[2][2];
228@else {
229    let view_pos = view_from_clip * vec4(0.0, 0.0, ndc_depth, 1.0);
230    return view_pos.z / view_pos.w;
231}
232}
233
234/// Convert linear view z to ndc depth.
235/// Note: View z input should be negative for values in front of the camera as -z is forward
236fn view_z_to_depth_ndc(view_z: f32, clip_from_view: mat4x4<f32>) -> f32 {
237@if(VIEW_PROJECTION_PERSPECTIVE)
238    return -perspective_camera_near(clip_from_view) / view_z;
239@elif(VIEW_PROJECTION_ORTHOGRAPHIC)
240    return clip_from_view[3][2] + view_z * clip_from_view[2][2];
241@else {
242    let ndc_pos = clip_from_view * vec4(0.0, 0.0, view_z, 1.0);
243    return ndc_pos.z / ndc_pos.w;
244}

251/// Convert ndc space xy coordinate [-1.0 .. 1.0] to uv [0.0 .. 1.0]
252fn ndc_to_uv(ndc: vec2<f32>) -> vec2<f32> {
253    return ndc * vec2(0.5, -0.5) + vec2(0.5);
254}
255
256/// Convert uv [0.0 .. 1.0] coordinate to ndc space xy [-1.0 .. 1.0]
257fn uv_to_ndc(uv: vec2<f32>) -> vec2<f32> {
258    return uv * vec2(2.0, -2.0) + vec2(-1.0, 1.0);
259}

261/// returns the (0.0, 0.0) .. (1.0, 1.0) position within the viewport for the current render target
262/// [0 .. render target viewport size] eg. [(0.0, 0.0) .. (1280.0, 720.0)] to [(0.0, 0.0) .. (1.0, 1.0)]
263fn frag_coord_to_uv(frag_coord: vec2<f32>, viewport: vec4<f32>) -> vec2<f32> {
264    return (frag_coord - viewport.xy) / viewport.zw;
265}
266
267/// Convert frag coord to ndc
268fn frag_coord_to_ndc(frag_coord: vec4<f32>, viewport: vec4<f32>) -> vec3<f32> {
269    return vec3(uv_to_ndc(frag_coord_to_uv(frag_coord.xy, viewport)), frag_coord.z);
270}
271
272/// Convert ndc space xy coordinate [-1.0 .. 1.0] to [0 .. render target
273/// viewport size]
274fn ndc_to_frag_coord(ndc: vec2<f32>, viewport: vec4<f32>) -> vec2<f32> {
275    return ndc_to_uv(ndc) * viewport.zw;
