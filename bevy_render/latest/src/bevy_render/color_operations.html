5// https://en.wikipedia.org/wiki/SRGB
6fn gamma(value: f32) -> f32 {
7    if value <= 0.0 {
8        return value;
9    }
10    if value <= 0.04045 {
11        return value / 12.92; // linear falloff in dark values
12    } else {
13        return pow((value + 0.055) / 1.055, 2.4); // gamma curve in other area
14    }
15}

17// https://en.wikipedia.org/wiki/SRGB
18fn inverse_gamma(value: f32) -> f32 {
19    if value <= 0.0 {
20        return value;
21    }
22
23    if value <= 0.0031308 {
24        return value * 12.92; // linear falloff in dark values
25    } else {
26        return 1.055 * pow(value, 1.0 / 2.4) - 0.055; // gamma curve in other area
27    }
28}

55fn oklab_to_linear_rgb(c: vec3<f32>) -> vec3<f32> {
56    let l_ = c.x + 0.39633778 * c.y + 0.21580376 * c.z;
57    let m_ = c.x - 0.105561346 * c.y - 0.06385417 * c.z;
58    let s_ = c.x - 0.08948418 * c.y - 1.2914855 * c.z;
59    let l = l_ * l_ * l_;
60    let m = m_ * m_ * m_;
61    let s = s_ * s_ * s_;
62    return vec3(
63        4.0767417 * l - 3.3077116 * m + 0.23096994 * s,
64        -1.268438 * l + 2.6097574 * m - 0.34131938 * s,
65        -0.0041960863 * l - 0.7034186 * m + 1.7076147 * s,
66    );
67}

69fn okhsl_to_oklab(okhsl: vec3<f32>) -> vec3<f32> {
70    let h = okhsl.x;
71    let s = okhsl.y;
72    let l = okhsl.z;
73
74    if (l >= 1.0) {
75        return vec3<f32>(1.0, 0.0, 0.0);
76    }
77    if (l <= 0.0) {
78        return vec3<f32>(0.0, 0.0, 0.0);
79    }
80
81    let a_ = cos(2.0 * PI * h);
82    let b_ = sin(2.0 * PI * h);
83    let L = okhsl_toe_inv(l);
84
85    let cs = okhsl_get_Cs(L, a_, b_);
86    let C_0 = cs.x;
87    let C_mid = cs.y;
88    let C_max = cs.z;
89
90    let mid = 0.8;
91    let mid_inv = 1.25;
92
93    var C: f32 = 0.0;
94    var t: f32 = 0.0;
95    var k_0: f32 = 0.0;
96    var k_1: f32 = 0.0;
97    var k_2: f32 = 0.0;
98
99    if (s < mid) {
100        t = mid_inv * s;
101        k_1 = mid * C_0;
102        k_2 = 1.0 - k_1 / C_mid;
103        C = t * k_1 / (1.0 - k_2 * t);
104    } else {
105        t = (s - mid) / (1.0 - mid);
106        k_0 = C_mid;
107        k_1 = (1.0 - mid) * C_mid * C_mid * mid_inv * mid_inv / C_0;
108        k_2 = 1.0 - k_1 / (C_max - C_mid);
109        C = k_0 + t * k_1 / (1.0 - k_2 * t);
110    }
111
112    return vec3<f32>(L, C * a_, C * b_);
113}

132fn hsl_to_linear_rgb(hsl: vec3<f32>) -> vec3<f32> {
133    let h = hsl.x;
134    let s = hsl.y;
135    let l = hsl.z;
136    let c = (1.0 - abs(2.0 * l - 1.0)) * s;
137    let hp = h * 6.0;
138    let x = c * (1.0 - abs(hp % 2.0 - 1.0));
139    var r: f32 = 0.0;
140    var g: f32 = 0.0;
141    var b: f32 = 0.0;
142    if 0.0 <= hp && hp < 1.0 {
143        r = c; g = x; b = 0.0;
144    } else if 1.0 <= hp && hp < 2.0 {
145        r = x; g = c; b = 0.0;
146    } else if 2.0 <= hp && hp < 3.0 {
147        r = 0.0; g = c; b = x;
148    } else if 3.0 <= hp && hp < 4.0 {
149        r = 0.0; g = x; b = c;
150    } else if 4.0 <= hp && hp < 5.0 {
151        r = x; g = 0.0; b = c;
152    } else if 5.0 <= hp && hp < 6.0 {
153        r = c; g = 0.0; b = x;
154    }
155    let m = l - 0.5 * c;
156    return srgb_to_linear_rgb(vec3(r + m, g + m, b + m));
157}

159fn hsv_to_linear_rgb(hsva: vec3<f32>) -> vec3<f32> {
160    let h = hsva.x * 6.0;
161    let s = hsva.y;
162    let v = hsva.z;
163    let c = v * s;
164    let x = c * (1.0 - abs(h % 2.0 - 1.0));
165    let m = v - c;
166    var r: f32 = 0.0;
167    var g: f32 = 0.0;
168    var b: f32 = 0.0;
169    if 0.0 <= h && h < 1.0 {
170        r = c; g = x; b = 0.0;
171    } else if 1.0 <= h && h < 2.0 {
172        r = x; g = c; b = 0.0;
173    } else if 2.0 <= h && h < 3.0 {
174        r = 0.0; g = c; b = x;
175    } else if 3.0 <= h && h < 4.0 {
176        r = 0.0; g = x; b = c;
177    } else if 4.0 <= h && h < 5.0 {
178        r = x; g = 0.0; b = c;
179    } else if 5.0 <= h && h < 6.0 {
180        r = c; g = 0.0; b = x;
181    }
182    return srgb_to_linear_rgb(vec3(r + m, g + m, b + m));
183}

190fn mix_oklch(a: vec3<f32>, b: vec3<f32>, t: f32) -> vec3<f32> {
191    // If the chroma is close to zero for one of the endpoints, don't interpolate 
192    // the hue and instead use the hue of the other endpoint. This allows gradients that smoothly 
193    // transition from black or white to a target color without passing through unrelated hues.
194    var h = a.z;
195    var g = b.z;
196    if a.y < HUE_GUARD {
197        h = g;
198    } else if b.y < HUE_GUARD {
199        g = h;
200    }
201
202    let hue_diff = g - h;
203    if abs(hue_diff) > 0.5 {
204        if hue_diff > 0.0 {
205            h += (hue_diff - 1.) * t;
206        } else {
207            h += (hue_diff + 1.) * t;
208        }
209    } else {
210        h += hue_diff * t;
211    }
212    return vec3(
213        mix(a.x, b.x, t),
214        mix(a.y, b.y, t),
215        fract(h),
216    );
217}

371// Converts HSV to RGB.
372//
373// Input: H ∈ [0, 2π), S ∈ [0, 1], V ∈ [0, 1].
374// Output: R ∈ [0, 1], G ∈ [0, 1], B ∈ [0, 1].
375//
376// <https://en.wikipedia.org/wiki/HSL_and_HSV#HSV_to_RGB_alternative>
377fn hsv_to_rgb(hsv: vec3<f32>) -> vec3<f32> {
378    let n = vec3(5.0, 3.0, 1.0);
379    let k = (n + hsv.x / FRAC_PI_3) % 6.0;
380    return hsv.z - hsv.z * hsv.y * max(vec3(0.0), min(k, min(4.0 - k, vec3(1.0))));
381}

383// Converts RGB to HSV.
384//
385// Input: R ∈ [0, 1], G ∈ [0, 1], B ∈ [0, 1].
386// Output: H ∈ [0, 2π), S ∈ [0, 1], V ∈ [0, 1].
387//
388// <https://en.wikipedia.org/wiki/HSL_and_HSV#From_RGB>
389fn rgb_to_hsv(rgb: vec3<f32>) -> vec3<f32> {
390    let x_max = max(rgb.r, max(rgb.g, rgb.b));  // i.e. V
391    let x_min = min(rgb.r, min(rgb.g, rgb.b));
392    let c = x_max - x_min;  // chroma
393
394    var swizzle = vec3<f32>(0.0);
395    if (x_max == rgb.r) {
396        swizzle = vec3(rgb.gb, 0.0);
397    } else if (x_max == rgb.g) {
398        swizzle = vec3(rgb.br, 2.0);
399    } else {
400        swizzle = vec3(rgb.rg, 4.0);
401    }
402
403    let h = FRAC_PI_3 * fract(((swizzle.x - swizzle.y) / c + swizzle.z) / 6.0) * 6.0;
404
405    // Avoid division by zero.
406    var s = 0.0;
407    if (x_max > 0.0) {
408        s = c / x_max;
409    }
410
411    return vec3(h, s, x_max);
412}

417fn okhsl_toe_inv(x: f32) -> f32 {
418    let k_1 = 0.206;
419    let k_2 = 0.03;
420    let k_3 = (1.0 + k_1) / (1.0 + k_2);
421    return (x * x + k_1 * x) / (k_3 * (x + k_2));
422}

429// Polynomial approximation for mid-gamut ST values.
430// Derived by optimization to ensure S_mid < S_max and T_mid < T_max.
431fn okhsl_get_ST_mid(a_: f32, b_: f32) -> vec2<f32> {
432    let S = 0.11516993
433        + 1.0 / (7.4477897
434            + 4.1590123 * b_
435            + a_ * (-2.1955736
436                + 1.751984 * b_
437                + a_ * (-2.1370494 - 10.02301 * b_
438                    + a_ * (-4.2489457 + 5.387708 * b_ + 4.69891 * a_))));

441        + 1.0 / (1.6132032 - 0.6812438 * b_

443                + 0.9014812 * b_
444                + a_ * (-0.27087943
445                    + 0.6122399 * b_

451// Approximate max saturation using a polynomial, then refine with 1 step of Halley's method.
452// a and b must be normalized (a^2 + b^2 == 1).
453fn okhsl_compute_max_saturation(a: f32, b: f32) -> f32 {
454    var k0: f32 = 0.0; var k1: f32 = 0.0; var k2: f32 = 0.0;
455    var k3: f32 = 0.0; var k4: f32 = 0.0;
456    var wl: f32 = 0.0; var wm: f32 = 0.0; var ws: f32 = 0.0;
457
458    if (-1.8817033 * a - 0.8093649 * b > 1.0) {
459        // Red component limit
460        k0 = 1.1908628; k1 = 1.7657673; k2 = 0.5966264;
461        k3 = 0.755152; k4 = 0.5677124;
462        wl = 4.0767417; wm = -3.3077116; ws = 0.23096994;
463    } else if (1.8144411 * a - 1.1944528 * b > 1.0) {
464        // Green component limit
465        k0 = 0.73956515; k1 = -0.45954404; k2 = 0.08285427;
466        k3 = 0.1254107; k4 = 0.14503204;
467        wl = -1.268438; wm = 2.6097574; ws = -0.34131938;
468    } else {
469        // Blue component limit
470        k0 = 1.3573365; k1 = -0.00915799; k2 = -1.1513021;
471        k3 = -0.50559606; k4 = 0.00692167;
472        wl = -0.0041960863; wm = -0.7034186; ws = 1.7076147;
473    }
474
475    var S = k0 + k1 * a + k2 * b + k3 * a * a + k4 * a * b;
476
477    // This step is done once for faster rendering. 
478    // Most render targets will support 8 to 16 bit precision.
479    let k_l = 0.39633778 * a + 0.21580376 * b;
480    let k_m = -0.105561346 * a - 0.06385417 * b;
481    let k_s = -0.08948418 * a - 1.2914855 * b;
482    
483    let l_ = 1.0 + S * k_l;
484    let m_ = 1.0 + S * k_m;
485    let s_ = 1.0 + S * k_s;
486
487    let l = l_ * l_ * l_;
488    let m = m_ * m_ * m_;
489    let s = s_ * s_ * s_;
490
491    let l_dS = 3.0 * k_l * l_ * l_;
492    let m_dS = 3.0 * k_m * m_ * m_;
493    let s_dS = 3.0 * k_s * s_ * s_;
494
495    let l_dS2 = 6.0 * k_l * k_l * l_;
496    let m_dS2 = 6.0 * k_m * k_m * m_;
497    let s_dS2 = 6.0 * k_s * k_s * s_;
498
499    let f = wl * l + wm * m + ws * s;
500    let f1 = wl * l_dS + wm * m_dS + ws * s_dS;
501    let f2 = wl * l_dS2 + wm * m_dS2 + ws * s_dS2;
502
503    S = S - f * f1 / (f1 * f1 - 0.5 * f * f2);
504    return S;
505}

507fn okhsl_find_cusp(a: f32, b: f32) -> vec2<f32> {
508    let S_cusp = okhsl_compute_max_saturation(a, b);
509    let rgb_at_max = oklab_to_linear_rgb(vec3<f32>(1.0, S_cusp * a, S_cusp * b));
510    let L_cusp = pow(1.0 / max(rgb_at_max.r, max(rgb_at_max.g, rgb_at_max.b)), 1.0 / 3.0);
511    return vec2<f32>(L_cusp, L_cusp * S_cusp);
512}

514// Find intersection of the gamut boundary with the line from (L0, 0) to (L1, C1).
515// Uses 1 step of Halley's method.
516fn okhsl_find_gamut_intersection(a: f32, b: f32, L1: f32, C1: f32, L0: f32, cusp: vec2<f32>) -> f32 {
517    let cusp_L = cusp.x;
518    let cusp_C = cusp.y;
519    var t: f32;
520
521    if (((L1 - L0) * cusp_C - (cusp_L - L0) * C1) <= 0.0) {
522        // Lower half of the gamut
523        t = cusp_C * L0 / (C1 * cusp_L + cusp_C * (L0 - L1));
524    } else {
525        // Upper half: first intersect with triangle, then refine
526        t = cusp_C * (L0 - 1.0) / (C1 * (cusp_L - 1.0) + cusp_C * (L0 - L1));
527
528        let dL = L1 - L0;
529        let dC = C1;
530
531        // We cut down a step here for faster rendering.
532        // Most render targets will support 8 to 16 bit precision.
533
534        let k_l = 0.39633778 * a + 0.21580376 * b;
535        let k_m = -0.105561346 * a - 0.06385417 * b;
536        let k_s = -0.08948418 * a - 1.2914855 * b;
537
538        let l_dt = dL + dC * k_l;
539        let m_dt = dL + dC * k_m;
540        let s_dt = dL + dC * k_s;
541
542        for (var i = 0; i < 1; i = i + 1) {
543            let L = L0 * (1.0 - t) + t * L1;
544            let C = t * C1;
545
546            let l_ = L + C * k_l;
547            let m_ = L + C * k_m;
548            let s_ = L + C * k_s;
549
550            let l = l_ * l_ * l_;
551            let m = m_ * m_ * m_;
552            let s = s_ * s_ * s_;
553
554            let ldt = 3.0 * l_dt * l_ * l_;
555            let mdt = 3.0 * m_dt * m_ * m_;
556            let sdt = 3.0 * s_dt * s_ * s_;
557
558            let ldt2 = 6.0 * l_dt * l_dt * l_;
559            let mdt2 = 6.0 * m_dt * m_dt * m_;
560            let sdt2 = 6.0 * s_dt * s_dt * s_;
561
562            let r = 4.0767417 * l - 3.3077116 * m + 0.23096994 * s - 1.0;
563            let r1 = 4.0767417 * ldt - 3.3077116 * mdt + 0.23096994 * sdt;
564            let r2 = 4.0767417 * ldt2 - 3.3077116 * mdt2 + 0.23096994 * sdt2;
565
566            let u_r = r1 / (r1 * r1 - 0.5 * r * r2);
567            let t_r = select(3.40282347e+38, -r * u_r, u_r >= 0.0);
568
569            let g = -1.268438 * l + 2.6097574 * m - 0.34131938 * s - 1.0;
570            let g1 = -1.268438 * ldt + 2.6097574 * mdt - 0.34131938 * sdt;
571            let g2 = -1.268438 * ldt2 + 2.6097574 * mdt2 - 0.34131938 * sdt2;
572
573            let u_g = g1 / (g1 * g1 - 0.5 * g * g2);
574            let t_g = select(3.40282347e+38, -g * u_g, u_g >= 0.0);
575
576            let b_val = -0.0041960863 * l - 0.7034186 * m + 1.7076147 * s - 1.0;
577            let b1 = -0.0041960863 * ldt - 0.7034186 * mdt + 1.7076147 * sdt;
578            let b2 = -0.0041960863 * ldt2 - 0.7034186 * mdt2 + 1.7076147 * sdt2;
579
580            let u_b = b1 / (b1 * b1 - 0.5 * b_val * b2);
581            let t_b = select(3.40282347e+38, -b_val * u_b, u_b >= 0.0);
582
583            t = t + min(t_r, min(t_g, t_b));
584        }
585    }
586    return t;
587}

589// Returns C_0, C_mid, C_max for a given Oklab L and hue direction (a_, b_).
590fn okhsl_get_Cs(L: f32, a_: f32, b_: f32) -> vec3<f32> {
591    let cusp = okhsl_find_cusp(a_, b_);
592    let C_max = okhsl_find_gamut_intersection(a_, b_, L, 1.0, L, cusp);
593    let ST_max = okhsl_to_ST(cusp);
594
595    let k = C_max / min(L * ST_max.x, (1.0 - L) * ST_max.y);
596
597    let ST_mid = okhsl_get_ST_mid(a_, b_);
598    let C_a = L * ST_mid.x;
599    let C_b = (1.0 - L) * ST_mid.y;
600    let C_mid = 0.9 * k * sqrt(sqrt(1.0 / (1.0 / (C_a * C_a * C_a * C_a) + 1.0 / (C_b * C_b * C_b * C_b))));
601
602    let C_0_a = L * 0.4;
603    let C_0_b = (1.0 - L) * 0.8;
604    let C_0 = sqrt(1.0 / (1.0 / (C_0_a * C_0_a) + 1.0 / (C_0_b * C_0_b)));
605
606    return vec3<f32>(C_0, C_mid, C_max);
607}
