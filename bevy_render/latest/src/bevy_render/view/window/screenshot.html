1// This vertex shader will create a triangle that will cover the entire screen
2// with minimal effort, avoiding the need for a vertex buffer etc.
3@vertex
4fn vs_main(@builtin(vertex_index) in_vertex_index: u32) -> @builtin(position) vec4<f32> {
5    let x = f32((in_vertex_index & 1u) << 2u);
6    let y = f32((in_vertex_index & 2u) << 1u);
7    return vec4<f32>(x - 1.0, y - 1.0, 0.0, 1.0);
8}
