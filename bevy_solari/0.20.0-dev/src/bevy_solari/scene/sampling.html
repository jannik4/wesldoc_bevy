8fn power_heuristic(f: f32, g: f32) -> f32 {
9    return balance_heuristic(f * f, g * g);
10}
11
12fn balance_heuristic(f: f32, g: f32) -> f32 {
13    // Need to guard against NaNs since ReSTIR reservoirs can have UCW=0
14    if f == 0.0 {
15        return 0.0;
16    }
17    // When NaN, fall back to even weights of 0.5 : 0.5
18    let weight = 1.0 / (1.0 + (g / f));
19    return select(weight, 0.5, isnan(weight));
20}

22/// https://gpuopen.com/download/Bounded_VNDF_Sampling_for_Smith-GGX_Reflections.pdf (Listing 1)
23/// Result is invalid when output.z <= 0.0, and must be discarded
24fn sample_ggx_vndf(wi_tangent: vec3<f32>, roughness: f32, rng: ptr<function, u32>) -> vec3<f32> {
25    // Mirror BRDF case
26    if roughness <= MIRROR_ROUGHNESS_THRESHOLD {
27        return vec3(-wi_tangent.xy, wi_tangent.z);
28    }
29
30    let i = wi_tangent;
31    let rand = rand_vec2f(rng);
32    let i_std = normalize(vec3(i.xy * roughness, i.z));
33    let phi = PI_2 * rand.x;
34    let a = roughness;
35    let s = 1.0 + length(vec2(i.xy));
36    let a2 = a * a;
37    let s2 = s * s;
38    let k = (1.0 - a2) * s2 / (s2 + a2 * i.z * i.z);
39    let b = select(i_std.z, k * i_std.z, i.z > 0.0);
40    let z = fma(1.0 - rand.y, 1.0 + b, -b);
41    let sin_theta = sqrt(saturate(1.0 - z * z));
42    let o_std = vec3(sin_theta * cos(phi), sin_theta * sin(phi), z);
43    let m_std = i_std + o_std;
44    let m = normalize(vec3(m_std.xy * roughness, m_std.z));
45    return 2.0 * dot(i, m) * m - i;
46}

52/// https://gpuopen.com/download/Bounded_VNDF_Sampling_for_Smith-GGX_Reflections.pdf (Listing 2)
53fn ggx_vndf_pdf(wi_tangent: vec3<f32>, wo_tangent: vec3<f32>, roughness: f32) -> f32 {
54    // Mirror BRDF case
55    if roughness <= MIRROR_ROUGHNESS_THRESHOLD {
56        let mirror_wo = vec3(-wi_tangent.xy, wi_tangent.z);
57        if distance(mirror_wo, wo_tangent) < 0.0001 {
58            return bitcast<f32>(0x7F800000u); // INF
59        } else {
60            return 0.0;
61        }
62    }
63
64    let i = wi_tangent;
65    let o = wo_tangent;
66    let m = normalize(i + o);
67    let ndf = D_GGX(roughness, saturate(m.z));
68    let ai = roughness * i.xy;
69    let len2 = dot(ai, ai);
70    let t = sqrt(len2 + i.z * i.z);
71    var pdf: f32;
72    if i.z >= 0.0 {
73        let a = roughness;
74        let s = 1.0 + length(i.xy);
75        let a2 = a * a;
76        let s2 = s * s;
77        let k = (1.0 - a2) * s2 / (s2 + a2 * i.z * i.z);
78        pdf = ndf / (2.0 * (k * i.z + t));
79    } else {
80        pdf = ndf * (t - i.z) / (2.0 * len2);
81    }
82
83    return select(pdf, 0.0, isnan(pdf));
84}

124fn sample_random_light(ray_origin: vec3<f32>, origin_world_normal: vec3<f32>, rng: ptr<function, u32>) -> LightContribution {
125    let sample = generate_random_light_sample(rng);

131fn random_emissive_light_pdf(hit: ResolvedRayHitFull, ray_distance: f32, NdotV: f32) -> f32 {
132    let light_count = arrayLength(&light_sources);
133    let area_pdf = 1.0 / (f32(light_count) * f32(hit.triangle_count) * hit.triangle_area);
134    return area_pdf * (ray_distance * ray_distance) / NdotV;
135}

137fn sample_environment_map_light(world_direction: vec3<f32>) -> vec3<f32> {

147fn generate_random_light_sample(rng: ptr<function, u32>) -> GenerateRandomLightSampleResult {
148    let light_count = arrayLength(&light_sources);
149    let light_id = rand_range_u(light_count, rng);
150
151    let light_source = light_sources[light_id];
152
153    var triangle_id = 0u;
154    if light_source.kind != LIGHT_SOURCE_KIND_DIRECTIONAL {
155        let triangle_count = light_source.kind >> 1u;
156        triangle_id = rand_range_u(triangle_count, rng);
157    }
158
159    let seed = rand_u(rng);
160    let light_sample = LightSample((light_id << 16u) | triangle_id, seed);
161
162    var resolved_light_sample = resolve_light_sample(light_sample, light_source);
163    resolved_light_sample.inverse_pdf *= f32(light_count);
164
165    return GenerateRandomLightSampleResult(light_sample, resolved_light_sample);
166}

168fn resolve_light_sample(light_sample: LightSample, light_source: LightSource) -> ResolvedLightSample {
169    if light_source.kind == LIGHT_SOURCE_KIND_DIRECTIONAL {
170        let directional_light = directional_lights[light_source.id];
171
172        var direction_to_light: vec3<f32>;
173@if(!NO_DIRECTIONAL_LIGHT_SOFT_SHADOWS) {
174        // Sample a random direction within a cone whose base is the sun approximated as a disk
175        // https://www.realtimerendering.com/raytracinggems/unofficial_RayTracingGems_v1.9.pdf#0004286901.INDD%3ASec30%3A305
176        var rng = light_sample.seed;
177        let random = rand_vec2f(&rng);
178        let cos_theta = (1.0 - random.x) + random.x * directional_light.cos_theta_max;
179        let sin_theta = sqrt(1.0 - cos_theta * cos_theta);
180        let phi = random.y * PI_2;
181        let x = cos(phi) * sin_theta;
182        let y = sin(phi) * sin_theta;
183        direction_to_light = vec3(x, y, cos_theta);
184
185        // Rotate the ray so that the cone it was sampled from is aligned with the light direction
186        direction_to_light = orthonormalize(directional_light.direction_to_light) * direction_to_light;
187} @else {
188        direction_to_light = directional_light.direction_to_light;
189}
190
191        return ResolvedLightSample(
192        vec4(direction_to_light, 0.0),
193        -direction_to_light,
194        directional_light.luminance,
195        directional_light.inverse_pdf,
196    );
197} else {
198        let triangle_count = light_source.kind >> 1u;
199        let triangle_id = light_sample.light_id & 0xFFFFu;
200        let barycentrics = triangle_barycentrics(light_sample.seed);
201        let triangle_data = resolve_triangle_data_full(light_source.id, triangle_id, barycentrics);
202
203        return ResolvedLightSample(
204            vec4(triangle_data.world_position, 1.0),
205            triangle_data.world_normal,
206            triangle_data.material.emissive.rgb,
207            f32(triangle_count) * triangle_data.triangle_area,
208        );
209    }
210}

212fn calculate_resolved_light_contribution(resolved_light_sample: ResolvedLightSample, ray_origin: vec3<f32>, origin_world_normal: vec3<f32>) -> LightContribution {
213    let ray = resolved_light_sample.world_position.xyz - (resolved_light_sample.world_position.w * ray_origin);
214    let light_distance = length(ray);
215    let wi = ray / light_distance;
216
217    let cos_theta_light = saturate(dot(-wi, resolved_light_sample.world_normal));
218    let light_distance_squared = light_distance * light_distance;
219    let denominator = cos_theta_light / light_distance_squared;
220
221    let radiance = resolved_light_sample.radiance * denominator;
222    let inverse_solid_angle_pdf = resolved_light_sample.inverse_pdf * denominator;
223
224    return LightContribution(radiance, resolved_light_sample.inverse_pdf, inverse_solid_angle_pdf, wi, resolved_light_sample.world_position.w == 1.0);
225}

227fn trace_visibility(ray_origin: vec3<f32>, point: vec4<f32>) -> f32 {
228    var ray_direction = point.xyz;
229    var ray_t_max = RAY_T_MAX;
230
231    if point.w == 1.0 {
232        let ray = ray_direction - ray_origin;
233        let dist = length(ray);
234        ray_direction = ray / dist;
235        ray_t_max = dist - RAY_T_MIN;
236    }
237
238    if ray_t_max < RAY_T_MIN { return 0.0; }
239
240    let ray_hit = trace_ray(ray_origin, ray_direction, RAY_T_MIN, ray_t_max, RAY_FLAG_TERMINATE_ON_FIRST_HIT);
241    return f32(ray_hit.kind == RAY_QUERY_INTERSECTION_NONE);
242}

261/// https://www.realtimerendering.com/raytracinggems/unofficial_RayTracingGems_v1.9.pdf#0004286901.INDD%3ASec22%3A297
262fn triangle_barycentrics(seed: u32) -> vec3<f32> {
263    var rng = seed;
264    var barycentrics = rand_vec2f(&rng);
265    if barycentrics.x + barycentrics.y > 1.0 { barycentrics = 1.0 - barycentrics; }
266    return vec3(1.0 - barycentrics.x - barycentrics.y, barycentrics);
