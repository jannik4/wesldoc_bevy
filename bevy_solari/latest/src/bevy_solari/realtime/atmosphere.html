7fn calculate_light_contribution(resolved_light_sample: ResolvedLightSample, ray_origin: vec3<f32>, origin_world_normal: vec3<f32>) -> LightContribution {
8    var light_contribution = calculate_resolved_light_contribution(resolved_light_sample, ray_origin, origin_world_normal);
9    // Skip views without an atmosphere and lights behind the surface.
10    if atmosphere.outer_radius > 0.0 && resolved_light_sample.world_position.w == 0.0 && dot(light_contribution.wi, origin_world_normal) > 0.0 {
11        light_contribution.radiance *= directional_light_transmittance(
12            atmosphere,
13            atmosphere_transmittance_lut,
14            atmosphere_transmittance_sampler,
15            ray_origin,
16            light_contribution.wi,
17            // Light samples are points on the sun's disk, so they average to its visible fraction.
18            0.0,
19        );
20    }
21    return light_contribution;
22}
