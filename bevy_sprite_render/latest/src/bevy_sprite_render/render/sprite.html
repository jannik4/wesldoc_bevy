9    maths::affine3_to_square,

13import package::render::sprite_view_bindings::view;
14
15struct VertexInput {
16    @builtin(vertex_index) index: u32,
17    // NOTE: Instance-rate vertex buffer members prefixed with i_
18    // NOTE: i_model_transpose_colN are the 3 columns of a 3x4 matrix that is the transpose of the
19    // affine 4x3 model matrix.
20    @location(0) i_model_transpose_col0: vec4<f32>,
21    @location(1) i_model_transpose_col1: vec4<f32>,
22    @location(2) i_model_transpose_col2: vec4<f32>,
23    @location(3) i_color: vec4<f32>,
24    @location(4) i_uv_offset_scale: vec4<f32>,
25}
26
27struct VertexOutput {
28    @builtin(position) clip_position: vec4<f32>,
29    @location(0) uv: vec2<f32>,
30    @location(1) @interpolate(flat) color: vec4<f32>,
31};
32
33@vertex
34fn vertex(in: VertexInput) -> VertexOutput {
35    var out: VertexOutput;
36
37    let vertex_position = vec3<f32>(
38        f32(in.index & 0x1u),
39        f32((in.index & 0x2u) >> 1u),
40        0.0
41    );
42
43    out.clip_position = view.clip_from_world * affine3_to_square(mat3x4<f32>(
44        in.i_model_transpose_col0,
45        in.i_model_transpose_col1,
46        in.i_model_transpose_col2,
47    )) * vec4<f32>(vertex_position, 1.0);
48    out.uv = vec2<f32>(vertex_position.xy) * in.i_uv_offset_scale.zw + in.i_uv_offset_scale.xy;
49    out.color = in.i_color;
50
51    return out;
52}
53
54@group(1) @binding(0) var sprite_texture: texture_2d<f32>;
55@group(1) @binding(1) var sprite_sampler: sampler;
56
57@fragment
58fn fragment(in: VertexOutput) -> @location(0) vec4<f32> {
59    var color = in.color * textureSample(sprite_texture, sprite_sampler, in.uv);
60
61@if(TONEMAP_IN_SHADER)
62    color = tonemapping::tone_mapping(color, view.color_grading);
63
64@if(SRGB_OUTPUT)
65    color = vec4(linear_to_srgb(color.rgb), color.a);
66
67@if(OKLAB_OUTPUT)
68    color = vec4(linear_rgb_to_oklab(color.rgb), color.a);
69
70    return color;
71}
