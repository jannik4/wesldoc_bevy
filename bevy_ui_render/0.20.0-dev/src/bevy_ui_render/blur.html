1// Blur algorithms for blur regions behind UI nodes.
2//
3// Every pass shares the same uniform. `params` is interpreted per algorithm:
4//
5//   Box blur:    x = kernel radius in pixels, y = sample spacing multiplier
6//   Gaussian:    x = circle of confusion diameter in pixels, y = sigma multiplier
7//   Dual kawase: x = sample offset multiplier
8//   Bokeh:       x = kernel radius in pixels, y = 1 / kernel normalization factor

29struct ComputedBlurRegion {
30    min_x: f32,
31    max_x: f32,
32    min_y: f32,
33    max_y: f32,
34    border_radii_x: vec4<f32>,

36}

38// Whether a pixel falls inside any blur region, so that passes can skip the
39// expensive blur work for pixels that don't need it.
40fn is_blurred(position: vec2<f32>) -> bool {
41    return min_region_distance(position) <= 0.0;
42}
43
44// Returns the signed distance, in pixels, from `position` to the closest blur
45// region. Negative inside a region, positive outside.
46fn min_region_distance(position: vec2<f32>) -> f32 {
47    var min_distance = 1e30;
48    for (var i = 0u; i < blur_regions.current_regions_count; i++) {
49        let region = blur_regions.regions[i];
50        let center = vec2((region.max_x + region.min_x) * 0.5, (region.max_y + region.min_y) * 0.5);
51        let dims = vec2(region.max_x - region.min_x, region.max_y - region.min_y);
52        let half_smallest_dimension = min(dims.x, dims.y) * 0.5;
53
54        let distance = sd_rounded_box(
55            position - center,
56            dims,
57            min(region.border_radii_x, vec4(half_smallest_dimension)),
58            min(region.border_radii_y, vec4(half_smallest_dimension)),
59        );
60        min_distance = min(min_distance, distance);
61    }
62    return min_distance;
63}

65// ---------------------------------------------------------------------------
66// Gaussian blur
67// ---------------------------------------------------------------------------
68// Performs a single direction of the separable Gaussian blur kernel.
69//
70// * `frag_coord` is the screen-space pixel coordinate of the fragment.
71//
72// * `frag_offset` is the vector, in screen-space units, from one sample to the
73//   next: `vec2(1.0, 0.0)` for the horizontal pass and `vec2(0.0, 1.0)` for the
74//   vertical pass.
75fn gaussian_blur(frag_coord: vec4<f32>, frag_offset: vec2<f32>) -> vec4<f32> {
76    // The standard deviation as a fraction of the circle of confusion. Usually σ
77    // is half the radius, and the radius is half the CoC, giving the default 0.25.
78    let coc = blur_regions.params.x;
79    let sigma = coc * blur_regions.params.y;
80
81    // 1.5σ is a good, somewhat aggressive default for support—the number of
82    // texels on each side of the center that we process.
83    let support = i32(ceil(sigma * 1.5));
84    let uv = frag_coord.xy / vec2<f32>(textureDimensions(input_texture));
85    let offset = frag_offset / vec2<f32>(textureDimensions(input_texture));
86
87    // The probability density function of the Gaussian blur is (up to constant factors) `exp(-1 / 2σ² *
88    // x²). We precalculate the constant factor here to avoid having to
89    // calculate it in the inner loop.
90    let exp_factor = -1.0 / (2.0 * sigma * sigma);
91
92    // Accumulate samples on both sides of the current texel. Go two at a time,
93    // taking advantage of bilinear filtering.
94    var sum = textureSampleLevel(input_texture, texture_sampler, uv, 0.0).rgb;
95    var weight_sum = 1.0;
96    for (var i = 1; i <= support; i += 2) {
97        // This is a well-known trick to reduce the number of needed texture
98        // samples by a factor of two. We seek to accumulate two adjacent
99        // samples c₀ and c₁ with weights w₀ and w₁ respectively, with a single
100        // texture sample at a carefully chosen location. Observe that:
101        //
102        //     k ⋅ lerp(c₀, c₁, t) = w₀⋅c₀ + w₁⋅c₁
103        //
104        //                              w₁
105        //     if k = w₀ + w₁ and t = ───────
106        //                            w₀ + w₁
107        //
108        // Therefore, if we sample at a distance of t = w₁ / (w₀ + w₁) texels in
109        // between the two texel centers and scale by k = w₀ + w₁ afterward, we
110        // effectively evaluate w₀⋅c₀ + w₁⋅c₁ with a single texture lookup.
111        let w0 = exp(exp_factor * f32(i) * f32(i));
112        let w1 = exp(exp_factor * f32(i + 1) * f32(i + 1));
113        let uv_offset = offset * (f32(i) + w1 / (w0 + w1));
114        let weight = w0 + w1;
115
116        sum += (
117            textureSampleLevel(input_texture, texture_sampler, uv + uv_offset, 0.0).rgb +
118            textureSampleLevel(input_texture, texture_sampler, uv - uv_offset, 0.0).rgb
119        ) * weight;
120        weight_sum += weight * 2.0;
121    }
122
123    return vec4(sum / weight_sum, 1.0);
124}

142// ---------------------------------------------------------------------------
143// Box blur
144// ---------------------------------------------------------------------------
145// Performs a single direction of the separable box blur kernel: a plain average
146// of `2 * radius + 1` samples spaced `scale` pixels apart along `frag_offset`.
147fn box_blur(frag_coord: vec4<f32>, frag_offset: vec2<f32>) -> vec4<f32> {
148    let radius = i32(blur_regions.params.x);
149    let scale = blur_regions.params.y;
150
151    let uv = frag_coord.xy / vec2<f32>(textureDimensions(input_texture));
152    let offset = frag_offset * scale / vec2<f32>(textureDimensions(input_texture));
153
154    var sum = textureSampleLevel(input_texture, texture_sampler, uv, 0.0).rgb;
155    for (var i = 1; i <= radius; i++) {
156        sum += textureSampleLevel(input_texture, texture_sampler, uv + offset * f32(i), 0.0).rgb;
157        sum += textureSampleLevel(input_texture, texture_sampler, uv - offset * f32(i), 0.0).rgb;
158    }
159
160    return vec4(sum / f32(2 * radius + 1), 1.0);
161}

179// ---------------------------------------------------------------------------
180// Dual kawase blur
181// ---------------------------------------------------------------------------
182// The scene is downsampled through a chain of progressively half-resolution
183// textures and then upsampled back, blurring a little at every step. Because
184// most passes run at reduced resolution this produces very wide, smooth blurs
185// far cheaper than a single-pass kernel of equivalent width. Blur regions are
186// only applied in the final composite pass, which picks per pixel between the
187// blurred chain and the untouched scene.
188
189// Downsamples to half resolution: a center sample weighted 4 plus the four
190// diagonal neighbors. `half_pixel` of the destination equals one source texel.
191@fragment
192fn kawase_downsample(in: FullscreenVertexOutput) -> @location(0) vec4<f32> {
193    let offset = blur_regions.params.x;
194    let half_pixel = offset / vec2<f32>(textureDimensions(input_texture));
195
196    var sum = textureSampleLevel(input_texture, texture_sampler, in.uv, 0.0).rgb * 4.0;
197    sum += textureSampleLevel(input_texture, texture_sampler, in.uv - half_pixel, 0.0).rgb;
198    sum += textureSampleLevel(input_texture, texture_sampler, in.uv + half_pixel, 0.0).rgb;
199    sum += textureSampleLevel(input_texture, texture_sampler, in.uv + vec2(half_pixel.x, -half_pixel.y), 0.0).rgb;
200    sum += textureSampleLevel(input_texture, texture_sampler, in.uv - vec2(half_pixel.x, -half_pixel.y), 0.0).rgb;
201
202    return vec4(sum / 8.0, 1.0);
203}

205// Upsamples `source` to double resolution with the 8-tap dual kawase pattern.
206// `half_pixel` of the destination equals a quarter of a source texel.
207fn kawase_upsample_from(source: texture_2d<f32>, uv: vec2<f32>) -> vec4<f32> {
208    let offset = blur_regions.params.x;
209    let half_pixel = 0.25 * offset / vec2<f32>(textureDimensions(source));
210
211    var sum = textureSampleLevel(source, texture_sampler, uv + vec2(-half_pixel.x * 2.0, 0.0), 0.0).rgb;
212    sum += textureSampleLevel(source, texture_sampler, uv + vec2(-half_pixel.x, half_pixel.y), 0.0).rgb * 2.0;
213    sum += textureSampleLevel(source, texture_sampler, uv + vec2(0.0, half_pixel.y * 2.0), 0.0).rgb;
214    sum += textureSampleLevel(source, texture_sampler, uv + vec2(half_pixel.x, half_pixel.y), 0.0).rgb * 2.0;
215    sum += textureSampleLevel(source, texture_sampler, uv + vec2(half_pixel.x * 2.0, 0.0), 0.0).rgb;
216    sum += textureSampleLevel(source, texture_sampler, uv + vec2(half_pixel.x, -half_pixel.y), 0.0).rgb * 2.0;
217    sum += textureSampleLevel(source, texture_sampler, uv + vec2(0.0, -half_pixel.y * 2.0), 0.0).rgb;
218    sum += textureSampleLevel(source, texture_sampler, uv + vec2(-half_pixel.x, -half_pixel.y), 0.0).rgb * 2.0;
219
220    return vec4(sum / 12.0, 1.0);
221}
222
223@fragment
224fn kawase_upsample(in: FullscreenVertexOutput) -> @location(0) vec4<f32> {
225    return kawase_upsample_from(input_texture, in.uv);
226}

228// The final upsample back to full resolution, masked to the blur regions.
229// `input_texture` is the untouched scene and `aux_texture_a` is the
230// half-resolution blurred chain.
231@fragment
232fn kawase_composite(in: FullscreenVertexOutput) -> @location(0) vec4<f32> {
233    if !is_blurred(in.position.xy) {
234        return textureSampleLevel(input_texture, texture_sampler, in.uv, 0.0);
235    }
236    return kawase_upsample_from(aux_texture_a, in.uv);
237}

239// ---------------------------------------------------------------------------
240// Bokeh blur
241// ---------------------------------------------------------------------------
242// A separable convolution with a single-component complex Gaussian kernel
243//
244//     c(x) = exp(-a·x²) · (cos(b·x²) + i·sin(b·x²))
245//
246// whose 2D self-product has a magnitude close to a flat disc, mimicking the
247// aperture of a camera. Parameters are drawn from <http://yehar.com/blog/?p=1495>
248// via <https://github.com/mikepound/convolve/blob/master/complex_kernels.py>.
249//
250// The horizontal pass convolves each color channel with the complex kernel and
251// stores the complex responses across two textures. The vertical pass finishes
252// the separable convolution with complex multiplies and resolves the result to
253// a real color as `A·real + B·imag`, normalized by a factor computed on the CPU
254// (`bokeh_normalization` in `blur.rs`).
255//
256// These constants must stay in sync with their counterparts in `blur.rs`.
257const BOKEH_KERNEL_A: f32 = 0.862325;
258const BOKEH_KERNEL_B: f32 = 1.624835;
259const BOKEH_WEIGHT_REAL: f32 = 0.767583;
260const BOKEH_WEIGHT_IMAG: f32 = 1.862321;
261const BOKEH_KERNEL_SCALE: f32 = 1.4;
262
263// Evaluates the complex kernel at `x` texels from the center as (real, imaginary).
264fn bokeh_kernel(x: f32, radius: f32) -> vec2<f32> {
265    let t = x * BOKEH_KERNEL_SCALE / radius;
266    let t2 = t * t;
267    let magnitude = exp(-BOKEH_KERNEL_A * t2);
268    return vec2(magnitude * cos(BOKEH_KERNEL_B * t2), magnitude * sin(BOKEH_KERNEL_B * t2));
269}
270
271fn complex_multiply(a: vec2<f32>, b: vec2<f32>) -> vec2<f32> {
272    return vec2(a.x * b.x - a.y * b.y, a.x * b.y + a.y * b.x);
273}

275struct BokehHorizontalOutput {
276    // The complex response of the red and green channels: (R.re, R.im, G.re, G.im).
277    @location(0) red_green: vec4<f32>,
278    // The complex response of the blue channel: (B.re, B.im, 0, 0).
279    @location(1) blue: vec4<f32>,
280}
281
282@fragment
283fn bokeh_horizontal(in: FullscreenVertexOutput) -> BokehHorizontalOutput {
284    var out: BokehHorizontalOutput;
285    out.red_green = vec4(0.0);
286    out.blue = vec4(0.0);
287
288    // The vertical pass only reads this texture within `radius` pixels of a blur
289    // region, so everything further away can be skipped.
290    let radius = blur_regions.params.x;
291    if min_region_distance(in.position.xy) > radius {
292        return out;
293    }
294
295    let texel = 1.0 / vec2<f32>(textureDimensions(input_texture));
296    let support = i32(radius);
297    for (var x = -support; x <= support; x++) {
298        let kernel = bokeh_kernel(f32(x), radius);
299        let color = textureSampleLevel(
300            input_texture, texture_sampler, in.uv + vec2(f32(x), 0.0) * texel, 0.0).rgb;
301        // The input color is real-valued, so this is a scalar-complex product.
302        out.red_green += vec4(color.r * kernel, color.g * kernel);
303        out.blue += vec4(color.b * kernel, 0.0, 0.0);
304    }
305
306    return out;
307}

309@fragment
310fn bokeh_vertical(in: FullscreenVertexOutput) -> @location(0) vec4<f32> {
311    if !is_blurred(in.position.xy) {
312        return textureSampleLevel(input_texture, texture_sampler, in.uv, 0.0);
313    }
314
315    let radius = blur_regions.params.x;
316    let inverse_normalization = blur_regions.params.y;
317    let texel = 1.0 / vec2<f32>(textureDimensions(aux_texture_a));
318
319    var red = vec2(0.0);
320    var green = vec2(0.0);
321    var blue = vec2(0.0);
322
323    let support = i32(radius);
324    for (var y = -support; y <= support; y++) {
325        let kernel = bokeh_kernel(f32(y), radius);
326        let uv = in.uv + vec2(0.0, f32(y)) * texel;
327        let red_green = textureSampleLevel(aux_texture_a, texture_sampler, uv, 0.0);
328        let blue_sample = textureSampleLevel(aux_texture_b, texture_sampler, uv, 0.0);
329        red += complex_multiply(red_green.xy, kernel);
330        green += complex_multiply(red_green.zw, kernel);
331        blue += complex_multiply(blue_sample.xy, kernel);
332    }
333
334    // Resolve the complex accumulators to a real color.
335    let weights = vec2(BOKEH_WEIGHT_REAL, BOKEH_WEIGHT_IMAG);
336    let color = vec3(dot(red, weights), dot(green, weights), dot(blue, weights))
337        * inverse_normalization;
338
339    // The kernel has small negative lobes, so the reconstruction rings into
340    // negative, out-of-gamut values around hard HDR edges. Desaturate toward the
341    // (non-negative) luminance just enough to lift the most-negative channel to 
342    // zero. This preserves luminance and keeps the hue stable. 
343
344    // The constants for calculating luminance are from the BT.709 standard.
345    let bt709 : vec3<f32> = vec3<f32>(0.2126, 0.7152, 0.0722);
346    
347    let luma = max(dot(color, bt709), 0.0);
348    let min_channel = min(color.r, min(color.g, color.b));
349    var resolved = color;
350    if min_channel < 0.0 {
351        // Solve luma + t * (min_channel - luma) = 0 for the blend toward gray.
352        let t = luma / (luma - min_channel);
353        resolved = mix(vec3(luma), color, t);
354    }
355    // Guard against any residual negatives from floating-point error.
356    return vec4(max(resolved, vec3(0.0)), 1.0);
